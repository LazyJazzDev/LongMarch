423 LM_DEVICE_FUNC Eigen::Matrix3<Real> DihedralAngle<Real>::SubHessian(const InputType &V, int subdim) const {
424   Eigen::Vector3<Real> e0 = V.col(2) - V.col(1);
425   Eigen::Vector3<Real> e1 = V.col(0) - V.col(2);
426   Eigen::Vector3<Real> e2 = V.col(0) - V.col(1);
427   Eigen::Vector3<Real> e1_tilda = V.col(3) - V.col(2);
428   Eigen::Vector3<Real> e2_tilda = V.col(3) - V.col(1);
429 
430   Real e0_norm_sqr = e0.squaredNorm();
431   Real e1_norm_sqr = e1.squaredNorm();
432   Real e2_norm_sqr = e2.squaredNorm();
433   Real e1_tilda_norm_sqr = e1_tilda.squaredNorm();
434   Real e2_tilda_norm_sqr = e2_tilda.squaredNorm();
435 
436   Real e0_norm = sqrt(e0_norm_sqr);
437   Real e1_norm = sqrt(e1_norm_sqr);
438   Real e2_norm = sqrt(e2_norm_sqr);
439   Real e1_tilda_norm = sqrt(e1_tilda_norm_sqr);
440   Real e2_tilda_norm = sqrt(e2_tilda_norm_sqr);
441 
442   Eigen::Vector3<Real> m0 = -(e1 - e1.dot(e0) * e0 / e0_norm_sqr).normalized();
443   Eigen::Vector3<Real> m1 = (e2 - e2.dot(e1) * e1 / e1_norm_sqr).normalized();
444   Eigen::Vector3<Real> m2 = (e1 - e1.dot(e2) * e2 / e2_norm_sqr).normalized();
445 
446   Eigen::Vector3<Real> m0_tilda = -(e1_tilda - e1_tilda.dot(e0) * e0 / e0_norm_sqr).normalized();
447   Eigen::Vector3<Real> m1_tilda = (e2_tilda - e2_tilda.dot(e1_tilda) * e1_tilda / e1_tilda_norm_sqr).normalized();
448   Eigen::Vector3<Real> m2_tilda = (e1_tilda - e1_tilda.dot(e2_tilda) * e2_tilda / e2_tilda_norm_sqr).normalized();
449 
450   Real h0 = -m0.dot(e1);
451   Real h1 = m1.dot(e2);
452   Real h2 = m2.dot(e1);
453 
454   Real h0_tilda = -m0_tilda.dot(e1_tilda);
455   Real h1_tilda = m1_tilda.dot(e2_tilda);
456   Real h2_tilda = m2_tilda.dot(e1_tilda);
457 
458   Eigen::Vector3<Real> n = e2.cross(e1).normalized();
459   Eigen::Vector3<Real> n_tilda = e1_tilda.cross(e2_tilda).normalized();
460 
461   Real cos_a0 = e1.dot(e2) / (e1_norm * e2_norm);
462   Real cos_a1 = e2.dot(e0) / (e2_norm * e0_norm);
463   Real cos_a2 = -e0.dot(e1) / (e0_norm * e1_norm);
464 
465   Real cos_a0_tilda = e1_tilda.dot(e2_tilda) / (e1_tilda_norm * e2_tilda_norm);
466   Real cos_a1_tilda = e2_tilda.dot(e0) / (e2_tilda_norm * e0_norm);
467   Real cos_a2_tilda = -e0.dot(e1_tilda) / (e0_norm * e1_tilda_norm);
468 
469   switch (subdim) {
470     case 0:
471       return -S(Q0);
472     case 1:
473       return S(P11) - N0 + S(P11_tilda) - N0_tilda;
474     case 2:
475       return S(P22) - N0 + S(P22_tilda) - N0_tilda;
476     case 3:
477       return -S(Q0_tilda);
478     default:
479       return Eigen::Matrix3<Real>::Zero();
480   }
481 }

829   float e0_norm = sqrt(e0_norm_sqr);
830   float e2_norm = sqrt(e2_norm_sqr);
831   Vector3<float> m0 = -(e1 - e1.dot(e0) * e0 / e0_norm_sqr).normalized();
832   Vector3<float> m2 = (e1 - e1.dot(e2) * e2 / e2_norm_sqr).normalized();
833   float h2 = m2.dot(e1);
834   Vector3<float> n = e2.cross(e1).normalized();
835   float cos_a1 = e2.dot(e0) / (e2_norm * e0_norm);
836 
837   jacobian = cos_a1 / h2 * n;
838   hessian = S(P22) - N0;
839 
840   e1 = V.col(3) - V.col(2);
841   e2 = V.col(3) - V.col(1);
842   e1_norm_sqr = e1.squaredNorm();
843   e2_norm_sqr = e2.squaredNorm();
844   e2_norm = sqrt(e2_norm_sqr);
845   m0 = -(e1 - e1.dot(e0) * e0 / e0_norm_sqr).normalized();
846   m2 = (e1 - e1.dot(e2) * e2 / e2_norm_sqr).normalized();
847   h2 = m2.dot(e1);
848   n = e1.cross(e2).normalized();
849   cos_a1 = e2.dot(e0) / (e2_norm * e0_norm);
850 
851   jacobian += cos_a1 / h2 * n;
852   hessian += S(P22) - N0;
853 }
