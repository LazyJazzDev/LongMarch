code/grassland/physics/diff_kernel/dk_geometry_sdf.cpp · 271 行 · 查看版本源码 ↗

职责与设计

七种解析距离核与导数；本文件实现接口行为。

  • 点、线、线段返回无符号距离；球、胶囊、立方体、平面带符号。
  • CubeSDF.size 是半边长；PlaneSDF.normal 和 LineSDF.direction 要求单位向量才能具有距离语义。
  • ValidInput 并非都检查奇点；线核始终 true，但查询点在线上时导数有除零。
扩展语法核对记录

诊断均为template class显式实例化。已核对每条float/double实例化均有独立对象记录，算法说明位于对应模板类和方法；不是新算法或缺失实现。

159 template <typename Real>
160 LM_DEVICE_FUNC Eigen::Matrix<Real, 1, 3> CubeSDF<Real>::Jacobian(const InputType &p) const {
161   Eigen::Vector3<Real> p0 = p - center;
162   Eigen::Vector3<Real> q = p0.cwiseAbs() - Eigen::Vector3<Real>(size, size, size);
163   if (q.maxCoeff() > 0) {
164     q = q.cwiseMax(Eigen::Vector3<Real>(0, 0, 0));
165     // multiply by sign to get the correct direction
166     for (int i = 0; i < 3; i++) {
167       if (p0(i) < 0) {
168         q(i) = -q(i);
169       }
170     }
171     return q.normalized().transpose();
172   } else {
173     // Find max component
174     int max_idx = 0;
175     for (int i = 1; i < 3; i++) {
176       if (q(i) > q(max_idx)) {
177         max_idx = i;
178       }
179     }
180     Eigen::RowVector3<Real> result = Eigen::RowVector3<Real>::Zero();
181     result(max_idx) = 1;
182     if (p0(max_idx) < 0) {
183       result(max_idx) = -1;
184     }
185     return result;
186   }

189 template <typename Real>
190 LM_DEVICE_FUNC HessianTensor<Real, 1, 3> CubeSDF<Real>::Hessian(const InputType &v) const {
191   HessianTensor<Real, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> H;
192   Eigen::Vector3<Real> v_diff = v - center;
193   Eigen::Vector3<Real> p = v_diff;
194   H.m[0] = Eigen::Matrix3<Real>::Identity();
195   for (int dim = 0; dim < 3; dim++) {
196     if (p(dim) < -size) {
197       p(dim) = -size;
198     } else if (p(dim) > size) {
199       p(dim) = size;
200     } else {
201       H.m[0](dim, dim) = 0;
202     }
203   }
204   v_diff = v_diff - p;
205   Real v_diff_norm = v_diff.norm();
206   if (v_diff_norm > Eps<Real>()) {
207     v_diff = v_diff / v_diff_norm;
208     H.m[0] = (H.m[0] - v_diff * v_diff.transpose()) / v_diff_norm;
209     return H;
210   } else {
211     return {};
212   }
213 }
