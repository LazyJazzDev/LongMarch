职责与设计

基础可微函数与组合器；本文件定义类型与调用接口。

  • 所有导数遵循列主序展开，组合方向为 f2(f1(x))。
  • 归一化、范数、atan2 的 ValidInput 使用 100×Eps，行列式的对数要求 det>0。
  • MultiplyConstant 与 PlusConstant 继承内核输入约束，RightMultiplyMatrix 逐输出列组合导数。

339     H(0, 0, 1) = H(0, 1, 0) = (v[0] * v[0] - v[1] * v[1]) / (len2 * len2);
340     H(0, 1, 1) = 2.0 * v[0] * v[1] / (len2 * len2);

346 struct CrossNorm {
347   typedef Real Scalar;
348   typedef Eigen::Matrix<Real, 3, 2> InputType;
349   typedef Eigen::Matrix<Real, 1, 1> OutputType;
350 
351   LM_DEVICE_FUNC bool ValidInput(const InputType &F) const {
352     return F.col(0).cross(F.col(1)).norm() > Eps<Real>() * 100;
353   }
354 
355   LM_DEVICE_FUNC OutputType operator()(const InputType &F) const {
356     return OutputType{F.col(0).cross(F.col(1)).norm()};
357   }
358 
359   LM_DEVICE_FUNC Eigen::Matrix<Real, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> Jacobian(
360       const InputType &F) const {
361     Eigen::Matrix<Real, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> J;
362     Eigen::Vector3<Real> n = F.col(0).cross(F.col(1)).normalized();
363     J.block(0, 0, 1, 3) = n.transpose() * -Skew3<Real>(F.col(1));
364     J.block(0, 3, 1, 3) = n.transpose() * Skew3<Real>(F.col(0));
365     return J;
366   }
367 
368   LM_DEVICE_FUNC HessianTensor<Real, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> Hessian(
369       const InputType &F) const {
370     VecLength<Real> length;
371     HessianTensor<Real, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> H;
372     Eigen::Matrix<Real, 3, 6> J;
373     J.block(0, 0, 3, 3) = -Skew3<Real>(F.col(1));
374     J.block(0, 3, 3, 3) = Skew3<Real>(F.col(0));
375     Eigen::Vector3<Real> Fz = F.col(0).cross(F.col(1));
376     Eigen::Vector3<Real> n = Fz.normalized();
377     H.m[0].block(0, 3, 3, 3) = -Skew3<Real>(n);
378     H.m[0].block(3, 0, 3, 3) = Skew3<Real>(n);
379     H.m[0] += J.transpose() * length.Hessian(Fz).m[0] * J;
380     return H;
381   }

440 template <typename Func, typename MatrixType>
441 struct RightMultiplyMatrix {
442   typedef typename Func::Scalar Scalar;
443   typedef typename Func::InputType InputType;
444   typedef typename Eigen::Matrix<Scalar, Func::OutputType::RowsAtCompileTime, MatrixType::ColsAtCompileTime> OutputType;
445 
446   LM_DEVICE_FUNC bool ValidInput(const InputType &v) const {
447     return f.ValidInput(v);
448   }
449 
450   LM_DEVICE_FUNC OutputType operator()(const InputType &v) const {
451     // m by n * n by k
452     return f(v) * s;
453   }
454 
455   LM_DEVICE_FUNC Eigen::Matrix<Scalar, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> Jacobian(
456       const InputType &v) const {
457     JacobianType<RightMultiplyMatrix> J;
458     constexpr int m = Func::OutputType::RowsAtCompileTime;
459     constexpr int n = Func::OutputType::ColsAtCompileTime;
460     constexpr int k = MatrixType::ColsAtCompileTime;
461     constexpr int input_width = InputType::SizeAtCompileTime;
462     J.setZero();
463     auto J_f = f.Jacobian(v);
464     for (int i = 0; i < k; i++) {
465       for (int j = 0; j < n; j++) {
466         J.block(i * m, 0, m, input_width) += s(j, i) * J_f.block(j * m, 0, m, input_width);
467       }
468     }
469     return J;
470   }
471 
472   LM_DEVICE_FUNC HessianTensor<Scalar, OutputType::SizeAtCompileTime, InputType::SizeAtCompileTime> Hessian(
473       const InputType &v) const {
474     HessianType<RightMultiplyMatrix> H;
475     constexpr int m = Func::OutputType::RowsAtCompileTime;
476     constexpr int n = Func::OutputType::ColsAtCompileTime;
477     constexpr int k = MatrixType::ColsAtCompileTime;
478     constexpr int input_width = InputType::SizeAtCompileTime;
479     auto H_f = f.Hessian(v);
480     for (int i = 0; i < k; i++) {
481       for (int j = 0; j < n; j++) {
482         for (int c = 0; c < m; c++) {
483           H.m[i * m + c] += s(j, i) * H_f.m[j * m + c];
484         }
485       }
486     }
487     return H;
488   }
489 
490   Func f{};
491   MatrixType s{};
492 };
