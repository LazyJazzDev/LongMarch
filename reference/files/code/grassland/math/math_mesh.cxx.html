职责与设计

OBJ 读写、顶点重建、法线与 MikkTSpace 切线生成

  • MergeVertices 按所有存在的属性精确比较，重新编号并去掉重复索引的三角形。
  • LoadObjFile 不读取材质或顶点色，且没有清空已有 colors_；复用带颜色网格读取不同顶点数文件有属性长度风险。
  • GenerateNormals 按面角加权；任意两相邻面法线点积小于阈值便把该顶点按面拆开。退化面和孤立点不保证有效法线。
  • Sphere 生成单位球；Transformed 复制网格，位置作仿射变换、法线作逆转置、切线作线性变换。

181   for (size_t s = 0; s < shapes.size(); s++) {
182     size_t index_offset = 0;
183     // Loop over faces(polygon)
184     for (size_t f = 0; f < shapes[s].mesh.num_face_vertices.size(); f++) {
185       size_t fv = size_t(shapes[s].mesh.num_face_vertices[f]);
186 
187       // Loop over vertices in the face.
188       for (size_t v = 0; v < fv; v++) {
189         Vector3<Scalar> position{};
190         Vector3<Scalar> normal{};
191         Vector2<Scalar> tex_coord{};
192 
193         // access to vertex
194         tinyobj::index_t idx = shapes[s].mesh.indices[index_offset + v];
195         tinyobj::real_t vx = attrib.vertices[3 * size_t(idx.vertex_index) + 0];
196         tinyobj::real_t vy = attrib.vertices[3 * size_t(idx.vertex_index) + 1];
197         tinyobj::real_t vz = attrib.vertices[3 * size_t(idx.vertex_index) + 2];
198         position = Vector3<Scalar>(vx, vy, vz);
199 
200         // Check if `normal_index` is zero or positive. negative = no normal
201         // data
202         if (idx.normal_index >= 0) {
203           tinyobj::real_t nx = attrib.normals[3 * size_t(idx.normal_index) + 0];
204           tinyobj::real_t ny = attrib.normals[3 * size_t(idx.normal_index) + 1];
205           tinyobj::real_t nz = attrib.normals[3 * size_t(idx.normal_index) + 2];
206           normal = Vector3<Scalar>(nx, ny, nz);
207           have_normal = true;
208         }
209 
210         // Check if `texcoord_index` is zero or positive. negative = no texcoord
211         // data
212         if (idx.texcoord_index >= 0) {
213           tinyobj::real_t tx = attrib.texcoords[2 * size_t(idx.texcoord_index) + 0];
214           tinyobj::real_t ty = attrib.texcoords[2 * size_t(idx.texcoord_index) + 1];
215           tex_coord = Vector2<Scalar>(tx, ty);
216           have_texcoord = true;
217         }
218 
219         positions.push_back(position);
220         normals.push_back(normal);
221         tex_coords.push_back(tex_coord);
222 
223         // Optional: vertex colors
224         // tinyobj::real_t red   = attrib.colors[3*size_t(idx.vertex_index)+0];
225         // tinyobj::real_t green = attrib.colors[3*size_t(idx.vertex_index)+1];
226         // tinyobj::real_t blue  = attrib.colors[3*size_t(idx.vertex_index)+2];
227         if (v >= 2) {
228           indices.push_back(global_index_offset);
229           indices.push_back(global_index_offset + v - 1);
230           indices.push_back(global_index_offset + v);
231         }
232       }
233 
234       index_offset += fv;
235       global_index_offset += fv;
236 
237       // per-face material
238       // shapes[s].mesh.material_ids[f];
239     }
240   }
241 
242   positions_ = positions;
243   if (have_normal)
244     normals_ = normals;
245   else
246     normals_.clear();
247   if (have_texcoord)
248     tex_coords_ = tex_coords;
249   else
250     tex_coords_.clear();
251   tangents_.clear();
252   signals_.clear();
253   indices_ = indices;
254 
255   num_vertices_ = positions_.size();
256   num_indices_ = indices.size();
257 
258   MergeVertices();
259 
260   return 0;
261 }

343 template <typename Scalar>
344 int Mesh<Scalar>::GenerateNormals(Scalar merging_threshold) {
345   std::vector<std::vector<Vector3<Scalar>>> vertex_normals(num_vertices_);
346   std::vector<Vector3<Scalar>> weighted_normals(num_vertices_, Vector3<Scalar>::Zero());
347   std::vector<Scalar> weights(num_vertices_, 0);
348   std::vector<bool> merge(num_vertices_, true);
349 
350   for (size_t i = 0; i < num_indices_; i += 3) {
351     Vector3<Scalar> v0 = positions_[indices_[i]];
352     Vector3<Scalar> v1 = positions_[indices_[i + 1]];
353     Vector3<Scalar> v2 = positions_[indices_[i + 2]];
354     Vector3<Scalar> normal = (v1 - v0).cross(v2 - v0);
355     if (normal.norm() == 0.0) {
356       continue;
357     }
358     normal.normalize();
359     Scalar angles[3];
360     angles[0] = acos((v1 - v0).normalized().dot((v2 - v0).normalized()));
361     angles[1] = acos((v2 - v1).normalized().dot((v0 - v1).normalized()));
362     angles[2] = acos((v0 - v2).normalized().dot((v1 - v2).normalized()));
363     for (size_t j = 0; j < 3; j++) {
364       vertex_normals[indices_[i + j]].push_back(normal);
365       weighted_normals[indices_[i + j]] += angles[j] * normal;
366       weights[indices_[i + j]] += angles[j];
367     }
368   }
369 
370   std::vector<Vector3<Scalar>> new_positions = positions_;
371   std::vector<Vector3<Scalar>> new_normals(num_vertices_);
372   std::vector<Vector2<Scalar>> new_tex_coords = tex_coords_;
373   std::vector<Vector3<Scalar>> new_colors = colors_;
374   std::vector<uint32_t> split_count(num_vertices_, 0);
375   std::vector<uint32_t> new_indices;
376 
377   for (size_t i = 0; i < num_vertices_; i++) {
378     for (size_t j = 0; j < vertex_normals[i].size(); j++) {
379       for (size_t k = j + 1; k < vertex_normals[i].size(); k++) {
380         if (vertex_normals[i][j].dot(vertex_normals[i][k]) < merging_threshold) {
381           merge[i] = false;
382           break;
383         }
384       }
385       if (!merge[i]) {
386         break;
387       }
388     }
389 
390     if (merge[i]) {
391       if (weights[i] > Eps<Scalar>()) {
392         new_normals[i] = (weighted_normals[i] / weights[i]).normalized();
393       }
394     }
395   }
396 
397   for (size_t i = 0; i < num_indices_; i++) {
398     uint32_t vi = indices_[i];
399     if (merge[vi]) {
400       new_indices.push_back(vi);
401     } else {
402       if (split_count[vi] == 0) {
403         new_normals[vi] = vertex_normals[vi][0];
404         new_indices.push_back(vi);
405       } else {
406         new_positions.push_back(positions_[vi]);
407         new_normals.push_back(vertex_normals[vi][split_count[vi]]);
408         if (!tex_coords_.empty()) {
409           new_tex_coords.push_back(tex_coords_[vi]);
410         }
411         if (!colors_.empty()) {
412           new_colors.push_back(colors_[vi]);
413         }
414         new_indices.push_back(new_positions.size() - 1);
415       }
416       split_count[vi]++;
417     }
418   }
419 
420   positions_ = new_positions;
421   normals_ = new_normals;
422   tex_coords_ = new_tex_coords;
423   colors_ = new_colors;
424   indices_ = new_indices;
425 
426   tangents_.clear();
427   signals_.clear();
428   num_vertices_ = positions_.size();
429   num_indices_ = indices_.size();
430 
431   return 0;
432 }

441 template <typename Scalar>
442 int Mesh<Scalar>::GenerateTangents() {
443   SplitVertices();
444   if (tex_coords_.empty()) {
445     InitializeTexCoords();
446   }
447   if (normals_.empty()) {
448     GenerateNormals();
449   }
450 
451   SMikkTSpaceInterface mikkt_space_interface{};
452 
453   mikkt_space_interface.m_getNumFaces = [](const SMikkTSpaceContext *context) -> int {
454     auto mesh = reinterpret_cast<const Mesh *>(context->m_pUserData);
455     return mesh->NumIndices() / 3;
456   };
457 
458   mikkt_space_interface.m_getNumVerticesOfFace = [](const SMikkTSpaceContext *context, const int face) { return 3; };
459 
460   mikkt_space_interface.m_getPosition = [](const SMikkTSpaceContext *context, float position[], const int face,
461                                            const int vertex) {
462     auto mesh = reinterpret_cast<const Mesh *>(context->m_pUserData);
463     auto positions = mesh->positions_[face * 3 + vertex];
464     position[0] = positions[0];
465     position[1] = positions[1];
466     position[2] = positions[2];
467   };
468 
469   mikkt_space_interface.m_getNormal = [](const SMikkTSpaceContext *context, float normal[], const int face,
470                                          const int vertex) {
471     auto mesh = reinterpret_cast<const Mesh *>(context->m_pUserData);
472     auto normals = mesh->normals_[face * 3 + vertex];
473     normal[0] = normals[0];
474     normal[1] = normals[1];
475     normal[2] = normals[2];
476   };
477 
478   mikkt_space_interface.m_getTexCoord = [](const SMikkTSpaceContext *context, float tex_coord[], const int face,
479                                            const int vertex) {
480     auto mesh = reinterpret_cast<const Mesh *>(context->m_pUserData);
481     auto tex_coords = mesh->tex_coords_[face * 3 + vertex];
482     tex_coord[0] = tex_coords[0];
483     tex_coord[1] = tex_coords[1];
484   };
485 
486   mikkt_space_interface.m_setTSpaceBasic = [](const SMikkTSpaceContext *context, const float tangent[],
487                                               const float sign, const int face, const int vertex) {
488     auto mesh = reinterpret_cast<Mesh *>(context->m_pUserData);
489     auto &tangents = mesh->tangents_[face * 3 + vertex];
490     tangents[0] = tangent[0];
491     tangents[1] = tangent[1];
492     tangents[2] = tangent[2];
493     mesh->signals_[face * 3 + vertex] = sign;
494   };
495 
496   SMikkTSpaceContext context{};
497   context.m_pInterface = &mikkt_space_interface;
498   context.m_pUserData = reinterpret_cast<void *>(this);
499 
500   tangents_.resize(num_vertices_);
501   signals_.resize(num_vertices_);
502 
503   if (genTangSpaceDefault(&context) == false) {
504     return -1;
505   }
506 
507   return MergeVertices();
508 }

510 template <typename Scalar>
511 Mesh<Scalar> Mesh<Scalar>::Sphere(int precision_lon, int precision_lat) {
512   if (precision_lat == -1) {
513     precision_lat = precision_lon * 2;
514   }
515   std::vector<glm::vec<3, Scalar>> sphere_positions;
516   std::vector<uint32_t> sphere_indices;
517   sphere_positions.push_back(glm::vec3{0.0, 1.0, 0.0});
518   for (int j = 1; j < precision_lon; ++j) {
519     for (int i = 0; i < precision_lat; ++i) {
520       Scalar theta = glm::radians(Scalar(360.0) * (i + (j & 1) * Scalar(0.5)) / precision_lat);
521       Scalar phi = glm::radians(Scalar(180.0) * j / precision_lon);
522       sphere_positions.push_back(
523           glm::vec<3, Scalar>{glm::sin(phi) * glm::cos(theta), glm::cos(phi), glm::sin(phi) * glm::sin(theta)});
524     }
525   }
526   sphere_positions.push_back(glm::vec<3, Scalar>{0.0, -1.0, 0.0});
527   for (int j = 0; j < precision_lon; ++j) {
528     for (int i = 0; i < precision_lat; ++i) {
529       int i_1 = (i - 1 + precision_lat) % precision_lat;
530       int i0 = i;
531       int i1 = (i + 1) % precision_lat;
532       int v00 = (j - 1) * precision_lat + i0 + 1;
533       int v01 = (j - 1) * precision_lat + i1 + 1;
534       int v10 = j * precision_lat + i0 + 1;
535       int v11 = j * precision_lat + i1 + 1;
536       if (j & 1) {
537         v00 = (j - 1) * precision_lat + i_1 + 1;
538         v01 = (j - 1) * precision_lat + i0 + 1;
539       }
540       if (j == 0) {
541         v00 = 0;
542         v01 = 0;
543       }
544       if (j == precision_lon - 1) {
545         v10 = sphere_positions.size() - 1;
546         v11 = sphere_positions.size() - 1;
547       } else {
548         sphere_indices.push_back(v11);
549         sphere_indices.push_back(v10);
550         sphere_indices.push_back(v01);
551       }
552 
553       if (j) {
554         sphere_indices.push_back(v00);
555         sphere_indices.push_back(v01);
556         sphere_indices.push_back(v10);
557       }
558     }
559   }
560   Mesh sphere_mesh(sphere_positions.size(), sphere_indices.size(), sphere_indices.data(),
561                    reinterpret_cast<Vector3<Scalar> *>(sphere_positions.data()),
562                    reinterpret_cast<Vector3<Scalar> *>(sphere_positions.data()), nullptr, nullptr);
563   return sphere_mesh;
564 }

576 template <typename Scalar>
577 Mesh<Scalar> Mesh<Scalar>::Transformed(const Matrix<Scalar, 3, 4> &transform) const {
578   Mesh<Scalar> mesh = *this;
579   Matrix<Scalar, 3, 3> affine_transform = transform.block(0, 0, 3, 3);
580   Matrix<Scalar, 3, 3> normal_transform = affine_transform.inverse().transpose();
581   for (size_t i = 0; i < mesh.num_vertices_; i++) {
582     mesh.positions_[i] = (affine_transform * mesh.positions_[i]) + transform.col(3);
583     if (!mesh.normals_.empty()) {
584       mesh.normals_[i] = (normal_transform * mesh.normals_[i]).normalized();
585     }
586     if (!mesh.tangents_.empty()) {
587       mesh.tangents_[i] = (affine_transform * mesh.tangents_[i]).normalized();
588     }
589   }
590   return mesh;
591 }
