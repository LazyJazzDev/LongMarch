84                                        const Vector3<Real> &v0,
85                                        const Vector3<Real> &v1,

89   Vector3<Real> ds0 = v1 - v0;
90   Vector3<Real> ds1 = v2 - v0;
91   Vector3<Real> dp = p - v0;
92   // clang-format off
93   // d(u, v) = p - v0 - u * ds0 - v * ds1
94   // E = d^2 = dp^2 - 2 * u * dp dot ds0 - 2 * v * dp dot ds1 + 2 * uv * ds0 dot ds1 + u^2 * ds0^2 + v^2 * ds1^2
95   // = a + 2bu + 2cv + du^2 + ev^2 + 2fuv
96   // dE/du = 2b + 2u * d + 2v * f = 0
97   // dE/dv = 2c + 2v * e + 2u * f = 0
98   // 2d * u + 2f * v = -2b
99   // 2f * u + 2e * v = -2c
100   // du + fv = -b
101   // fu + ev = -c
102   // clang-format on
103   Real b = dp.dot(ds0);
104   Real c = dp.dot(ds1);
105   Real d = ds0.squaredNorm();
106   Real e = ds1.squaredNorm();
107   Real f = ds0.dot(ds1);
108   // Ax = b
109   Matrix<Real, 2, 2> A;
110   A << d, f, f, e;
111   A = A.inverse().eval();
112   u = A(0, 0) * b + A(0, 1) * c;
113   v = A(1, 0) * b + A(1, 1) * c;
114   return (dp - u * ds0 - v * ds1).norm();
115 }

231                                      const Vector3<Real> &p1,
232                                      const Vector3<Real> &q0,
233                                      const Vector3<Real> &q1,
234                                      Real &u,
235                                      Real &v) {
236   Vector3<Real> pq = p0 - q0;
237   Vector3<Real> dp = p1 - p0;
238   Vector3<Real> dq = q1 - q0;
239   // clang-format off
240   // d(u, v) = (p0 + u * dp) - (q0 + v * dq) = pq + u * dp - v * dq
241   // E = d^2 = pq^2 + 2u * pq dot dp - 2v * pq dot dq + u^2 * dp^2 - 2uv * dp dot dq + v^2 * dq^2
242   // = a + 2bu + 2cv + du^2 + ev^2 + 2fuv
243   // dE/du = 2b + 2du + 2fv = 0
244   // dE/dv = 2c + 2fu + 2ev = 0
245   // clang-format on
246   Real a = pq.squaredNorm();
247   Real b = dp.dot(pq);
248   Real c = -dq.dot(pq);
249   Real d = dp.squaredNorm();
250   Real e = dq.squaredNorm();
251   Real f = -dp.dot(dq);
252   // Ax = b
253   Matrix2<Real> A;
254   A << d, f, f, e;
255   if (A.determinant() < Eps<Real>() * Eps<Real>()) {
256     u = 0;
257     return DistancePointLine(p0, q0, q1, v);
258   }

261   u = A(0, 0) * b + A(0, 1) * c;
262   v = A(1, 0) * b + A(1, 1) * c;
263   return (pq + u * dp - v * dq).norm();
