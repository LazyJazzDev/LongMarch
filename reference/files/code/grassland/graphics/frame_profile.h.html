职责与设计

可选 CPU/GPU 帧性能记录与 RAII scope。

  • FrameProfile::active 为 thread_local，嵌套 Begin 不允许；每帧先 Begin，录制 scope，WaitGPU 后 Finish。
  • GPU 查询只实现 Vulkan，固定 64 个时间戳槽即最多 32 个区间；CPU-only 可用于其他后端。
  • CPU/GPU scope 是含子区间的累计值，不可把嵌套值相加作为总耗时。

16 // Opt-in, single-thread frame profiling. GPU results must be read after WaitGPU.
17 // CPU scopes may nest: inclusive CPU and GPU durations must not be added together.
18 class FrameProfile {
19  public:
20   explicit FrameProfile(graphics::Core *core, bool gpu_timestamps = true) {
21     device_name = core->DeviceName();
22     if (!gpu_timestamps)
23       return;
24 #if defined(LONGMARCH_VULKAN_ENABLED)
25     auto vk = dynamic_cast<graphics::backend::VulkanCore *>(core);
26     if (!vk)
27       throw std::runtime_error("frame GPU profiling currently requires Vulkan; use --profile-cpu-only for Metal");
28     device_ = vk->Handle();
29     const auto properties = grassland::graphics::backend::vulkan::GetPhysicalDeviceProperties(vk->PhysicalDevice());
30     device_name = properties.deviceName;
31     period_ = properties.limits.timestampPeriod;
32     bits_ = grassland::graphics::backend::vulkan::GetQueueFamilyProperties(
33                 vk->PhysicalDevice())[grassland::graphics::backend::vulkan::GraphicsFamilyIndex(vk->PhysicalDevice())]
34                 .timestampValidBits;
35     if (!bits_ || period_ <= 0)
36       throw std::runtime_error("GPU queue does not support timestamps");
37     VkQueryPoolCreateInfo info{};
38     info.sType = VK_STRUCTURE_TYPE_QUERY_POOL_CREATE_INFO;
39     info.queryType = VK_QUERY_TYPE_TIMESTAMP;
40     info.queryCount = kQueries;
41     if (vkCreateQueryPool(device_, &info, nullptr, &pool_) != VK_SUCCESS)
42       throw std::runtime_error("could not create timestamp query pool");
43 #else
44     throw std::runtime_error("frame GPU profiling currently requires Vulkan; use --profile-cpu-only for Metal");
45 #endif
46   }
47 
48   ~FrameProfile() {
49     if (active == this)
50       active = nullptr;
51 #if defined(LONGMARCH_VULKAN_ENABLED)
52     if (pool_) {
53       vkDeviceWaitIdle(device_);
54       vkDestroyQueryPool(device_, pool_, nullptr);
55     }
56 #endif
57   }
58 
59   FrameProfile(const FrameProfile &) = delete;
60   FrameProfile &operator=(const FrameProfile &) = delete;
61 
62   void Begin(bool gpu_timestamps = true) {
63     if (active)
64       throw std::runtime_error("frame profile already active");
65     cpu_ms.clear();
66     gpu_ms.clear();
67     counters.clear();
68     names_.clear();
69     timestamps_enabled_ = gpu_timestamps;
70 #if defined(LONGMARCH_VULKAN_ENABLED)
71     counters["gpu_timestamps"] = pool_ && timestamps_enabled_;
72 #endif
73     active = this;
74   }
75 
76   void Finish() {
77     active = nullptr;
78 #if defined(LONGMARCH_VULKAN_ENABLED)
79     std::vector<uint64_t> values(names_.size() * 2);
80     if (!values.empty() && vkGetQueryPoolResults(device_, pool_, 0, values.size(), values.size() * sizeof(uint64_t),
81                                                  values.data(), sizeof(uint64_t), VK_QUERY_RESULT_64_BIT) != VK_SUCCESS)
82       throw std::runtime_error("GPU timestamps unavailable; Finish requires GPU completion");
83     const uint64_t mask = bits_ == 64 ? ~uint64_t(0) : (uint64_t(1) << bits_) - 1;
84     for (size_t i = 0; i < names_.size(); ++i)
85       gpu_ms[names_[i]] += ((values[i * 2 + 1] - values[i * 2]) & mask) * period_ / 1e6;
86 #endif
87   }
88 
89   int BeginGpu(graphics::CommandContext *commands, const std::string &name) {
90 #if defined(LONGMARCH_VULKAN_ENABLED)
91     if (!pool_ || !timestamps_enabled_)
92       return -1;
93     auto vk = dynamic_cast<graphics::backend::VulkanCommandContext *>(commands);
94     if (!vk || (names_.size() + 1) * 2 > kQueries)
95       throw std::runtime_error("invalid GPU profiling scope");
96     const int index = names_.size() * 2;
97     names_.push_back(name);
98     vk->CmdTimestamp(pool_, index, index == 0 ? kQueries : 0);
99     return index;
100 #else
101     return 0;
102 #endif
103   }
104 
105   void EndGpu(graphics::CommandContext *commands, int index) {
106 #if defined(LONGMARCH_VULKAN_ENABLED)
107     if (index < 0)
108       return;
109     dynamic_cast<graphics::backend::VulkanCommandContext *>(commands)->CmdTimestamp(pool_, index + 1);
110 #endif
111   }
112

114   std::map<std::string, double> cpu_ms, gpu_ms;
115   std::map<std::string, uint64_t> counters;
116   std::string device_name;
117 
118  private:
119   static constexpr uint32_t kQueries = 64;
120   std::vector<std::string> names_;
121   bool timestamps_enabled_{true};
122 #if defined(LONGMARCH_VULKAN_ENABLED)
123   VkDevice device_{};
124   VkQueryPool pool_{};
125   double period_{};
126   uint32_t bits_{};
127 #endif
128 };
