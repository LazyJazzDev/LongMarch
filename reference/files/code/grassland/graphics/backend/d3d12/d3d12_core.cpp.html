职责与设计

设备工厂、逐帧提交同步、共享staging与DXR构建。；本文件实现接口行为。

  • Submit先上传动态buffer，再编译命令、恢复普通资源GENERIC_READ、Present(0,0)、signal fence；轮转后等待可复用帧槽并执行回调。
  • WaitGPU等当前fence值并清全部完成回调；析构仅关闭event，不隐式等待GPU。
  • Create*通常成功返回0，原生错误可能抛异常；BLAS/TLAS创建忽略内部Build HRESULT，不能仅凭0判断资源有效。
  • DXR payload上限2048字节、attribute16字节、递归深度31；SBT record32字节、table64字节对齐，无local root参数。
  • CUDA共享fence导入返回码未检查，初始化查询cudaDeviceProp使用adapter索引而非匹配出的CUDA索引；共享semaphore/handle清理也不完整。
扩展语法核对记录

CUDA条件编译把else与后续普通fence分支分开，C++解析器报告孤立else。已核对InitializeLogicalDevice完整函数范围及CUDA/非CUDA两条路径，原文全部保留。

330     command_context->resource_descriptor_gpu_base_ =
331         resource_descriptor_heaps_[current_frame_]->GetGPUDescriptorHandleForHeapStart();
332   }
333 
334   if (command_context->sampler_descriptor_count_) {
335     command_context->sampler_descriptor_size_ =
336         device_.Get()->GetDescriptorHandleIncrementSize(D3D12_DESCRIPTOR_HEAP_TYPE_SAMPLER);
337     command_context->sampler_descriptor_base_ =
338         sampler_descriptor_heaps_[current_frame_]->GetCPUDescriptorHandleForHeapStart();
339     command_context->sampler_descriptor_gpu_base_ =
340         sampler_descriptor_heaps_[current_frame_]->GetGPUDescriptorHandleForHeapStart();
341   }
342 
343   if (!rtv_descriptor_heaps_[current_frame_] ||
344       rtv_descriptor_heaps_[current_frame_]->GetDesc().NumDescriptors < command_context->rtv_index_.size()) {
345     rtv_descriptor_heaps_[current_frame_] =
346         CreateNativeDescriptorHeap(device_.Get(), D3D12_DESCRIPTOR_HEAP_TYPE_RTV, command_context->rtv_index_.size());
347   }
348 
349   if (!dsv_descriptor_heaps_[current_frame_] ||
350       dsv_descriptor_heaps_[current_frame_]->GetDesc().NumDescriptors < command_context->dsv_index_.size()) {
351     dsv_descriptor_heaps_[current_frame_] =
352         CreateNativeDescriptorHeap(device_.Get(), D3D12_DESCRIPTOR_HEAP_TYPE_DSV, command_context->dsv_index_.size());
353   }
354 
355   for (auto &[resource, index] : command_context->rtv_index_) {
356     auto rtv_handle = RTVDescriptorHandle(index);
357     device_.Get()->CreateRenderTargetView(resource, nullptr, rtv_handle);
358   }
359 
360   for (auto &[resource, index] : command_context->dsv_index_) {
361     auto dsv_handle = DSVDescriptorHandle(index);
362     device_.Get()->CreateDepthStencilView(resource, nullptr, dsv_handle);
363   }
364 
365   BindDescriptorHeaps(command_list);
366 
367   for (auto &command : command_context->commands_) {
368     command->CompileCommand(command_context, command_list);
369   }
370 
371   for (auto &[image, state] : command_context->resource_states_) {
372     if (state != D3D12_RESOURCE_STATE_GENERIC_READ) {
373       CD3DX12_RESOURCE_BARRIER barrier =
374           CD3DX12_RESOURCE_BARRIER::Transition(image, state, D3D12_RESOURCE_STATE_GENERIC_READ);
375       command_list->ResourceBarrier(1, &barrier);
376     }
377   }
378 
379   command_list->Close();
380 
381   command_context->dsv_index_.clear();
382   command_context->rtv_index_.clear();
383 
384   ID3D12CommandList *command_lists[] = {command_list};
385   QueueWaitFence(command_queue_.Get());
386   command_queue_.Get()->ExecuteCommandLists(1, command_lists);
387 
388   for (auto window : command_context->windows_) {
389     window->SwapChain()->Present(0, 0);
390   }
391 
392   SignalFence(command_queue_.Get());
393   in_flight_values_[current_frame_] = fence_value_;
394 
395   post_execute_functions_[current_frame_] = p_command_context->GetPostExecutionCallbacks();
396 
397   current_frame_ = (current_frame_ + 1) % FramesInFlight();
398   WaitForFence(std::max(in_flight_values_[current_frame_], transfer_wait_value));
399 
400   for (auto &function : post_execute_functions_[current_frame_]) {
401     function();
402   }
403   post_execute_functions_[current_frame_].clear();
404 
405   return 0;
406 }

434

442       D3D12CreateDevice(adapter_.Get(), D3D_FEATURE_LEVEL_11_0, IID_PPV_ARGS(native_device.GetAddressOf())),
443       "Failed to create D3D12 device");
444   D3D12_FEATURE_DATA_FEATURE_LEVELS feature_levels{};
445   const D3D_FEATURE_LEVEL requested_levels[] = {D3D_FEATURE_LEVEL_12_1, D3D_FEATURE_LEVEL_12_0, D3D_FEATURE_LEVEL_11_1,

451           native_device->CheckFeatureSupport(D3D12_FEATURE_FEATURE_LEVELS, &feature_levels, sizeof(feature_levels)))) {
452     feature_levels.MaxSupportedFeatureLevel = D3D_FEATURE_LEVEL_11_0;
453   }
454   if (DebugEnabled()) {
455     ComPtr<ID3D12InfoQueue> info_queue;
456     if (SUCCEEDED(native_device.As(&info_queue))) {
457       info_queue->SetBreakOnSeverity(D3D12_MESSAGE_SEVERITY_CORRUPTION, true);
458       info_queue->SetBreakOnSeverity(D3D12_MESSAGE_SEVERITY_ERROR, true);
459     }
460   }
461   device_ = native_device;
462   d3d12::ThrowIfFailed(device_.As(&dxr_device_), "Failed to query DXR device");
463   device_->CheckFeatureSupport(D3D12_FEATURE_D3D12_OPTIONS1, &d3d12_options1_, sizeof(d3d12_options1_));
464 
465   device_name_ = NativeAdapterName(adapter_.Get());
466   ray_tracing_support_ = NativeAdapterSupportsRayTracing(adapter_.Get());
467 
468   D3D12_COMMAND_QUEUE_DESC queue_desc{};
469   queue_desc.Type = D3D12_COMMAND_LIST_TYPE_DIRECT;
470   d3d12::ThrowIfFailed(device_.Get()->CreateCommandQueue(&queue_desc, IID_PPV_ARGS(command_queue_.GetAddressOf())),
471                        "Failed to create graphics queue");
472   command_allocators_.resize(FramesInFlight());
473   command_lists_.resize(FramesInFlight());
474   resource_descriptor_heaps_.resize(FramesInFlight());
475   sampler_descriptor_heaps_.resize(FramesInFlight());
476   rtv_descriptor_heaps_.resize(FramesInFlight());
477   dsv_descriptor_heaps_.resize(FramesInFlight());
478   post_execute_functions_.resize(FramesInFlight());
479   for (int i = 0; i < FramesInFlight(); i++) {
480     resource_descriptor_heaps_[i] =
481         CreateNativeDescriptorHeap(device_.Get(), D3D12_DESCRIPTOR_HEAP_TYPE_CBV_SRV_UAV, 64);
482     sampler_descriptor_heaps_[i] = CreateNativeDescriptorHeap(device_.Get(), D3D12_DESCRIPTOR_HEAP_TYPE_SAMPLER, 64);
483 
484     CreateCommandRecord(device_.Get(), D3D12_COMMAND_LIST_TYPE_DIRECT, command_allocators_[i], command_lists_[i]);
485   }
486 
487   CreateCommandRecord(device_.Get(), D3D12_COMMAND_LIST_TYPE_DIRECT, single_time_allocator_, single_time_command_list_);
488 
489   d3d12::ThrowIfFailed(
490       device_.Get()->CreateCommandQueue(&queue_desc, IID_PPV_ARGS(transfer_command_queue_.GetAddressOf())),
491       "Failed to create transfer queue");
492   CreateCommandRecord(device_.Get(), D3D12_COMMAND_LIST_TYPE_DIRECT, transfer_allocator_, transfer_command_list_);
493 
494   blit_pipeline_.Initialize(device_.Get());
495 
496 #if defined(LONGMARCH_CUDA_RUNTIME)
497   cuda_device_ = NativeAdapterCUDADeviceIndex(adapter_.Get());
498   cudaDeviceProp device_prop{};
499   cudaGetDeviceProperties(&device_prop, device_index);
500   cuda_device_node_mask_ = device_prop.luidDeviceNodeMask;
501 
502   if (cuda_device_ >= 0) {
503     d3d12::ThrowIfFailed(device_.Get()->CreateFence(1, D3D12_FENCE_FLAG_SHARED, IID_PPV_ARGS(fence_.GetAddressOf())),
504                          "Failed to create shared fence");
505     cudaExternalSemaphoreHandleDesc externalSemaphoreHandleDesc{};
506     WindowsSecurityAttributes windowsSecurityAttributes;
507     LPCWSTR name = nullptr;
508     HANDLE sharedHandle;
509     externalSemaphoreHandleDesc.type = cudaExternalSemaphoreHandleTypeD3D12Fence;
510     device_.Get()->CreateSharedHandle(fence_.Get(), &windowsSecurityAttributes, GENERIC_ALL, name, &sharedHandle);
511     externalSemaphoreHandleDesc.handle.win32.handle = (void *)sharedHandle;
512     externalSemaphoreHandleDesc.flags = 0;
513 
514     int current_cuda_device;
515     cudaGetDevice(&current_cuda_device);
516     cudaSetDevice(cuda_device_);
517     cudaImportExternalSemaphore(&cuda_semaphore_, &externalSemaphoreHandleDesc);
518     cudaSetDevice(current_cuda_device);
519   } else
520 #endif
521   {
522     d3d12::ThrowIfFailed(device_.Get()->CreateFence(1, D3D12_FENCE_FLAG_NONE, IID_PPV_ARGS(fence_.GetAddressOf())),
523                          "Failed to create fence");
524   }
525   fence_event_ = CreateEvent(nullptr, FALSE, FALSE, nullptr);
526   if (!fence_event_) {
527     throw std::runtime_error("Failed to create fence event");
528   }
529   in_flight_values_.resize(FramesInFlight(), 0);
530 
531   return 0;
532 }

809   CD3DX12_STATE_OBJECT_DESC pipeline_desc(D3D12_STATE_OBJECT_TYPE_RAYTRACING_PIPELINE);
810   auto lib_ray_gen = pipeline_desc.CreateSubobject<CD3DX12_DXIL_LIBRARY_SUBOBJECT>();
811   auto ray_gen_code = D3D12_SHADER_BYTECODE{ray_gen_shader->data.data(), ray_gen_shader->data.size()};
812   lib_ray_gen->SetDXILLibrary(&ray_gen_code);
813   lib_ray_gen->DefineExport(L"RayGenMain", StringToWString(ray_gen_shader->entry_point).c_str());
814 
815   for (size_t i = 0; i < miss_shaders.size(); i++) {
816     auto &miss_shader = miss_shaders[i];
817     auto lib_miss = pipeline_desc.CreateSubobject<CD3DX12_DXIL_LIBRARY_SUBOBJECT>();
818     auto miss_code = D3D12_SHADER_BYTECODE{miss_shader->data.data(), miss_shader->data.size()};
819     lib_miss->SetDXILLibrary(&miss_code);
820     lib_miss->DefineExport((L"MissMain" + std::to_wstring(i)).c_str(),
821                            StringToWString(miss_shader->entry_point).c_str());
822   }
823 
824   for (size_t i = 0; i < hit_groups.size(); i++) {
825     auto &hit_group = hit_groups[i];
826     auto obj_hit_group = pipeline_desc.CreateSubobject<CD3DX12_HIT_GROUP_SUBOBJECT>();
827 
828     auto lib_rchit = pipeline_desc.CreateSubobject<CD3DX12_DXIL_LIBRARY_SUBOBJECT>();
829     auto rchit_code =
830         D3D12_SHADER_BYTECODE{hit_group.closest_hit_shader->data.data(), hit_group.closest_hit_shader->data.size()};
831     lib_rchit->SetDXILLibrary(&rchit_code);
832     lib_rchit->DefineExport((L"ClosestHitMain" + std::to_wstring(i)).c_str(),
833                             StringToWString(hit_group.closest_hit_shader->entry_point).c_str());
834     obj_hit_group->SetClosestHitShaderImport((L"ClosestHitMain" + std::to_wstring(i)).c_str());
835 
836     if (hit_group.intersection_shader) {
837       auto lib_rint = pipeline_desc.CreateSubobject<CD3DX12_DXIL_LIBRARY_SUBOBJECT>();
838       auto rint_code =
839           D3D12_SHADER_BYTECODE{hit_group.intersection_shader->data.data(), hit_group.intersection_shader->data.size()};
840       lib_rint->SetDXILLibrary(&rint_code);
841       lib_rint->DefineExport((L"IntersectionMain" + std::to_wstring(i)).c_str(),
842                              StringToWString(hit_group.intersection_shader->entry_point).c_str());
843       obj_hit_group->SetIntersectionShaderImport((L"IntersectionMain" + std::to_wstring(i)).c_str());
844     }
845 
846     if (hit_group.any_hit_shader) {
847       auto lib_rahit = pipeline_desc.CreateSubobject<CD3DX12_DXIL_LIBRARY_SUBOBJECT>();
848       auto rahit_code =
849           D3D12_SHADER_BYTECODE{hit_group.any_hit_shader->data.data(), hit_group.any_hit_shader->data.size()};
850       lib_rahit->SetDXILLibrary(&rahit_code);
851       lib_rahit->DefineExport((L"AnyHitMain" + std::to_wstring(i)).c_str(),
852                               StringToWString(hit_group.any_hit_shader->entry_point).c_str());
853       obj_hit_group->SetAnyHitShaderImport((L"AnyHitMain" + std::to_wstring(i)).c_str());
854     }
855 
856     obj_hit_group->SetHitGroupExport((L"HitGroup" + std::to_wstring(i)).c_str());
857     obj_hit_group->SetHitGroupType(hit_group.procedure ? D3D12_HIT_GROUP_TYPE_PROCEDURAL_PRIMITIVE
858                                                        : D3D12_HIT_GROUP_TYPE_TRIANGLES);
859   }
860 
861   for (size_t i = 0; i < callable_shaders.size(); i++) {
862     auto &callable_shader = callable_shaders[i];
863     auto lib_callable = pipeline_desc.CreateSubobject<CD3DX12_DXIL_LIBRARY_SUBOBJECT>();
864     auto callable_code = D3D12_SHADER_BYTECODE{callable_shader->data.data(), callable_shader->data.size()};
865     lib_callable->SetDXILLibrary(&callable_code);
866     lib_callable->DefineExport((L"CallableMain" + std::to_wstring(i)).c_str(),
867                                StringToWString(callable_shader->entry_point).c_str());
868   }
869 
870   auto shader_config = pipeline_desc.CreateSubobject<CD3DX12_RAYTRACING_SHADER_CONFIG_SUBOBJECT>();
871   shader_config->Config(512 * sizeof(float), 4 * sizeof(float));
872 
873   auto global_root_signature = pipeline_desc.CreateSubobject<CD3DX12_GLOBAL_ROOT_SIGNATURE_SUBOBJECT>();
874   global_root_signature->SetRootSignature(root_signature);
875 
876   auto pipeline_config = pipeline_desc.CreateSubobject<CD3DX12_RAYTRACING_PIPELINE_CONFIG_SUBOBJECT>();
877   pipeline_config->Config(31);
878 
879   RETURN_IF_FAILED_HR(dxr_device_->CreateStateObject(pipeline_desc, IID_PPV_ARGS(result.GetAddressOf())),
880                       "failed to create ray tracing pipeline.");
881 
882   return S_OK;
883 }

885 HRESULT D3D12Core::CreateShaderTable(ID3D12StateObject *pipeline,
886                                      const std::vector<int32_t> &miss_shader_indices,
887                                      const std::vector<int32_t> &hit_group_indices,
888                                      const std::vector<int32_t> &callable_shader_indices,
889                                      ComPtr<ID3D12Resource> &result,
890                                      D3D12_GPU_VIRTUAL_ADDRESS &miss_shader_offset,
891                                      D3D12_GPU_VIRTUAL_ADDRESS &hit_group_offset,
892                                      D3D12_GPU_VIRTUAL_ADDRESS &callable_shader_offset) const {
893   ComPtr<ID3D12StateObjectProperties> pipeline_properties;
894   RETURN_IF_FAILED_HR(pipeline->QueryInterface(IID_PPV_ARGS(&pipeline_properties)),
895                       "failed to get pipeline properties.");
896 
897   const UINT shader_idenfitier_size = D3D12_SHADER_IDENTIFIER_SIZE_IN_BYTES;
898   const UINT shader_record_size =
899       d3d12::SizeAlignTo(shader_idenfitier_size, D3D12_RAYTRACING_SHADER_RECORD_BYTE_ALIGNMENT);
900   const UINT shader_table_alignment = D3D12_RAYTRACING_SHADER_TABLE_BYTE_ALIGNMENT;
901 
902   D3D12_GPU_VIRTUAL_ADDRESS accum_offset = 0;
903 
904   miss_shader_offset = accum_offset = d3d12::SizeAlignTo(accum_offset + shader_record_size * 1, shader_table_alignment);
905 
906   hit_group_offset = accum_offset =
907       d3d12::SizeAlignTo(accum_offset + shader_record_size * miss_shader_indices.size(), shader_table_alignment);
908 
909   callable_shader_offset = accum_offset =
910       d3d12::SizeAlignTo(accum_offset + shader_record_size * hit_group_indices.size(), shader_table_alignment);
911 
912   accum_offset =
913       d3d12::SizeAlignTo(accum_offset + shader_record_size * callable_shader_indices.size(), shader_table_alignment);
914 
915   ComPtr<ID3D12Resource> buffer;
916   RETURN_IF_FAILED_HR(d3d12::CreateBuffer(device_.Get(), accum_offset, D3D12_HEAP_TYPE_UPLOAD,
917                                           D3D12_RESOURCE_STATE_GENERIC_READ, D3D12_RESOURCE_FLAG_NONE, buffer),
918                       "failed to create shader binding table buffer.");
919   uint8_t *data = nullptr;
920   RETURN_IF_FAILED_HR(buffer->Map(0, nullptr, reinterpret_cast<void **>(&data)),
921                       "failed to map shader binding table buffer.");
922 
923   void *shader_idenfitier = pipeline_properties->GetShaderIdentifier(L"RayGenMain");
924   std::memcpy(data, shader_idenfitier, shader_idenfitier_size);
925 
926   for (size_t i = 0; i < miss_shader_indices.size(); i++) {
927     auto miss_id = miss_shader_indices[i];
928     shader_idenfitier = pipeline_properties->GetShaderIdentifier((L"MissMain" + std::to_wstring(miss_id)).c_str());
929     std::memcpy(data + miss_shader_offset + i * shader_record_size, shader_idenfitier, shader_idenfitier_size);
930   }
931 
932   for (size_t i = 0; i < hit_group_indices.size(); i++) {
933     auto hit_group_id = hit_group_indices[i];
934     shader_idenfitier = pipeline_properties->GetShaderIdentifier((L"HitGroup" + std::to_wstring(hit_group_id)).c_str());
935     std::memcpy(data + hit_group_offset + i * shader_record_size, shader_idenfitier, shader_idenfitier_size);
936   }
937 
938   for (size_t i = 0; i < callable_shader_indices.size(); i++) {
939     auto callable_id = callable_shader_indices[i];
940     shader_idenfitier =
941         pipeline_properties->GetShaderIdentifier((L"CallableMain" + std::to_wstring(callable_id)).c_str());
942     std::memcpy(data + callable_shader_offset + i * shader_record_size, shader_idenfitier, shader_idenfitier_size);
943   }
944 
945   buffer->Unmap(0, nullptr);
946 
947   result = std::move(buffer);
948 
949   return S_OK;
950 }

952 ID3D12Resource *D3D12Core::RequestScratchBuffer(size_t size) {
953   if (!scratch_buffer_ || scratch_buffer_->GetDesc().Width < size) {
954     scratch_buffer_.Reset();
955     d3d12::CreateBuffer(device_.Get(), size, D3D12_HEAP_TYPE_DEFAULT, D3D12_RESOURCE_STATE_COMMON,
956                         D3D12_RESOURCE_FLAG_ALLOW_UNORDERED_ACCESS, scratch_buffer_);
957   }
958   return scratch_buffer_.Get();
959 }
