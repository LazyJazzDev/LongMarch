code/grassland/graphics/backend/d3d12/d3d12_program.cpp · 224 行 · 查看版本源码 ↗

职责与设计

图形、计算和DXR管线布局及Finalize。；本文件实现接口行为。

  • 每个资源绑定建立单个descriptor range，register space为添加顺序，base register为0。
  • 图形管线单sample、CCW正面、默认triangle拓扑类型，深度格式有效才开启深度，stencil关闭；最多8颜色附件由调用方保证。
  • Compute Finalize未检查CreateComputePipelineState HRESULT；DXR Finalize也未传播Core返回HRESULT。
  • DXR必须先设置raygen；Shader blob全为借用指针。SBT索引须指向已注册shader/group，未做范围检查。

18   for (size_t i = 0; i < descriptor_ranges_.size(); i++) {

83 void D3D12Program::BindShader(Shader *shader, ShaderType type) {
84   D3D12Shader *d3d12_shader = dynamic_cast<D3D12Shader *>(shader);
85   if (d3d12_shader) {
86     switch (type) {
87       case SHADER_TYPE_VERTEX:
88         pipeline_state_desc_.VS = d3d12_shader->Bytecode();
89         break;
90       case SHADER_TYPE_PIXEL:
91         pipeline_state_desc_.PS = d3d12_shader->Bytecode();
92         break;
93       case SHADER_TYPE_GEOMETRY:
94         pipeline_state_desc_.GS = d3d12_shader->Bytecode();
95         break;
96     }
97   } else {
98     throw std::runtime_error("Invalid shader object, expected D3D12Shader");
99   }
100 }

102 void D3D12Program::Finalize() {
103   FinalizeRootSignature();
104 
105   pipeline_state_desc_.InputLayout.pInputElementDescs = input_attributes_.data();
106   pipeline_state_desc_.InputLayout.NumElements = static_cast<UINT>(input_attributes_.size());
107   pipeline_state_desc_.pRootSignature = root_signature_.Get();
108 
109   d3d12::ThrowIfFailed(
110       core_->Device()->CreateGraphicsPipelineState(&pipeline_state_desc_, IID_PPV_ARGS(pipeline_state_.GetAddressOf())),
111       "Failed to create graphics pipeline state");
112 }

176 void D3D12RayTracingProgram::AddHitGroup(HitGroup hit_group) {
177   d3d12::HitGroup d3d_hit_group;
178   D3D12Shader *d3d12_closest_hit_shader = dynamic_cast<D3D12Shader *>(hit_group.closest_hit_shader);
179   assert(d3d12_closest_hit_shader != nullptr);
180   d3d_hit_group.closest_hit_shader = &d3d12_closest_hit_shader->CompiledBlob();
181   D3D12Shader *d3d12_any_hit_shader = dynamic_cast<D3D12Shader *>(hit_group.any_hit_shader);
182   if (d3d12_any_hit_shader) {
183     d3d_hit_group.any_hit_shader = &d3d12_any_hit_shader->CompiledBlob();
184   }
185 
186   D3D12Shader *d3d12_intersection_shader = dynamic_cast<D3D12Shader *>(hit_group.intersection_shader);
187   if (d3d12_intersection_shader) {
188     d3d_hit_group.intersection_shader = &d3d12_intersection_shader->CompiledBlob();
189   }
190   d3d_hit_group.procedure = hit_group.procedure;
191   hit_groups_.emplace_back(d3d_hit_group);
192 }

196   assert(shader != nullptr);
197   callable_shaders_.emplace_back(&shader->CompiledBlob());

200 void D3D12RayTracingProgram::Finalize(const std::vector<int32_t> &miss_shader_indices,
201                                       const std::vector<int32_t> &hit_group_indices,
202                                       const std::vector<int32_t> &callable_shader_indices) {
203   FinalizeRootSignature();
204 
205   core_->CreateRayTracingPipeline(root_signature_.Get(), raygen_shader_, miss_shaders_, hit_groups_, callable_shaders_,
206                                   pipeline_);
207   core_->CreateShaderTable(pipeline_.Get(), miss_shader_indices, hit_group_indices, callable_shader_indices,
208                            shader_table_, miss_offset_, hit_group_offset_, callable_offset_);
209   miss_count_ = miss_shader_indices.size();
210   hit_group_count_ = hit_group_indices.size();
211   callable_count_ = callable_shader_indices.size();
212 }
213 
214 void D3D12RayTracingProgram::Finalize() {
215   std::vector<int32_t> miss_shader_indices(miss_shaders_.size());
216   std::iota(miss_shader_indices.begin(), miss_shader_indices.end(), 0);
217   std::vector<int32_t> hit_group_indices(hit_groups_.size());
218   std::iota(hit_group_indices.begin(), hit_group_indices.end(), 0);
219   std::vector<int32_t> callable_shader_indices(callable_shaders_.size());
220   std::iota(callable_shader_indices.begin(), callable_shader_indices.end(), 0);
221   Finalize(miss_shader_indices, hit_group_indices, callable_shader_indices);
222 }
