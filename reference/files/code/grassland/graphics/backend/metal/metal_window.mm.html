metal_window.mm

code/grassland/graphics/backend/metal/metal_window.mm · 207 行 · 查看版本源码 ↗

职责与设计

Cocoa CAMetalLayer、SDR/EDR呈现和兼容ImGui。；Objective-C++实现。

  • ConfigurePresentation构建内嵌全屏三角MSL，采样输入纹理；SDR/HDR转换由上层准备，本呈现shader不做tone map。
  • SetHDR失败返回-1；Window::SetHDR失败时尝试恢复旧presentation；捕获异常路径只报告错误。
  • Present按framebuffer尺寸设置drawable，最小化或nextDrawable空时跳过；先图像后ImGui叠加，HDR下ImGuiLinearColors临时转换颜色。
  • CloseWindow先WaitGPU、关闭ImGui、从NSView解除layer，再释放原生window。
扩展语法核对记录

诊断均为Objective-C消息发送和__bridge cast；已核对构造、亮度查询、HDR、关闭、ImGui与Present方法均被提取，内嵌MSL另外逐对象登记。没有独立Objective-C类定义被遗漏。

33 DisplayBrightness MetalWindow::QueryDisplayBrightness() const {
34   DisplayBrightness result;
35   if (!GLFWWindow())
36     return result;
37   auto screen = [glfwGetCocoaWindow(GLFWWindow()) screen];
38   if (screen) {
39     // EDR linear 1.0 already follows the system's reference white.
40     result.reference_white_known = true;
41     result.hdr_reference_white_scale = 1.0f;
42     result.hdr_headroom = screen.maximumExtendedDynamicRangeColorComponentValue;
43     result.hdr_enabled = screen.maximumPotentialExtendedDynamicRangeColorComponentValue > 1.0;
44   }
45   return result;
46 }

48 void MetalWindow::ConfigurePresentation(bool enable_hdr) {
49   MetalPool pool;
50   const auto format = enable_hdr ? MTL::PixelFormatRGBA16Float : MTL::PixelFormatBGRA8Unorm;
51   const char *source = R"(
52 #include <metal_stdlib>
53 using namespace metal;
54 struct Vertex { float4 position [[position]]; float2 uv; };
55 vertex Vertex present_vertex(uint i [[vertex_id]]) {
56   float2 p = float2((i << 1) & 2, i & 2);
57   return {float4(p * float2(2, -2) + float2(-1, 1), 0, 1), p};
58 }
59 fragment float4 present_fragment(Vertex v [[stage_in]], texture2d<float> image [[texture(0)]]) {
60   constexpr sampler s(coord::normalized, address::clamp_to_edge, filter::linear);
61   return image.sample(s, v.uv);
62 }
63 )";
64   NS::Error *error = nullptr;
65   auto library =
66       NS::TransferPtr(core_->Device()->newLibrary(NS::String::string(source, NS::UTF8StringEncoding), nullptr, &error));
67   MetalCheck(library.get(), error, "presentation shaders");
68   auto vertex = NS::TransferPtr(library->newFunction(NS::String::string("present_vertex", NS::UTF8StringEncoding)));
69   auto fragment = NS::TransferPtr(library->newFunction(NS::String::string("present_fragment", NS::UTF8StringEncoding)));
70   auto descriptor = NS::TransferPtr(MTL::RenderPipelineDescriptor::alloc()->init());
71   descriptor->setVertexFunction(vertex.get());
72   descriptor->setFragmentFunction(fragment.get());
73   descriptor->colorAttachments()->object(0)->setPixelFormat(format);
74   auto pipeline = NS::TransferPtr(core_->Device()->newRenderPipelineState(descriptor.get(), &error));
75   MetalCheck(pipeline.get(), error, "presentation pipeline");
76   core_->WaitGPU();
77   auto colorspace = CGColorSpaceCreateWithName(enable_hdr ? kCGColorSpaceExtendedLinearSRGB : kCGColorSpaceSRGB);
78   if (!colorspace)
79     throw std::runtime_error("Failed to create Metal presentation color space");
80   auto native_layer = (CAMetalLayer *)layer_.get();
81   native_layer.pixelFormat = (MTLPixelFormat)format;
82   native_layer.colorspace = colorspace;
83   native_layer.wantsExtendedDynamicRangeContent = enable_hdr;
84   CGColorSpaceRelease(colorspace);
85   pipeline_ = std::move(pipeline);
86 }

88 int MetalWindow::SetHDR(bool enable_hdr) {
89   try {
90     MetalPool pool;
91     if (!GLFWWindow())
92       return -1;
93     if (enable_hdr_ == enable_hdr)
94       return 0;
95     const bool previous = enable_hdr_;
96     ConfigurePresentation(enable_hdr);
97     if (Window::SetHDR(enable_hdr) != 0) {
98       ConfigurePresentation(previous);
99       return -1;
100     }
101     if (enable_hdr) {
102       auto screen = [glfwGetCocoaWindow(GLFWWindow()) screen];
103       LogInfo("Metal HDR enabled: linear sRGB, RGBA16Float; display {} EDR headroom {:.2f}, potential {:.2f}",
104               screen ? screen.localizedName.UTF8String : "unknown",
105               double(screen.maximumExtendedDynamicRangeColorComponentValue),
106               double(screen.maximumPotentialExtendedDynamicRangeColorComponentValue));
107       if (screen.maximumPotentialExtendedDynamicRangeColorComponentValue <= 1.0)
108         LogWarning("Current display has no HDR headroom; HDR output will be displayed within its SDR range");
109     } else {
110       LogInfo("Metal HDR disabled: sRGB, BGRA8Unorm");
111     }
112     return 0;
113   } catch (const std::exception &error) {
114     LogError("Failed to change Metal HDR presentation: {}", error.what());
115     return -1;
116   }
117 }

178 void MetalWindow::Present(MTL::CommandBuffer *command, MetalImage *image) {
179   MetalPool pool;
180   GetDisplayBrightness();
181   int width, height;
182   glfwGetFramebufferSize(GLFWWindow(), &width, &height);
183   if (width <= 0 || height <= 0)
184     return;
185   layer_->setDrawableSize(CGSizeMake(width, height));
186   auto drawable = layer_->nextDrawable();
187   if (!drawable)
188     return;
189   auto pass = MTL::RenderPassDescriptor::renderPassDescriptor();
190   auto attachment = pass->colorAttachments()->object(0);
191   attachment->setTexture(drawable->texture());
192   attachment->setLoadAction(MTL::LoadActionDontCare);
193   attachment->setStoreAction(MTL::StoreActionStore);
194   auto encoder = command->renderCommandEncoder(pass);
195   encoder->setRenderPipelineState(pipeline_.get());
196   encoder->setFragmentTexture(image->Handle(), 0);
197   encoder->drawPrimitives(MTL::PrimitiveTypeTriangle, NS::UInteger(0), NS::UInteger(3));
198   if (imgui_) {
199     ImGui::SetCurrentContext(imgui_);
200     ImGuiLinearColors linear_ui(enable_hdr_);
201     ImGui_ImplMetal_RenderDrawData(ImGui::GetDrawData(), (__bridge id<MTLCommandBuffer>)command,
202                                    (__bridge id<MTLRenderCommandEncoder>)encoder);
203   }
204   encoder->endEncoding();
205   command->presentDrawable(drawable);
206 }
