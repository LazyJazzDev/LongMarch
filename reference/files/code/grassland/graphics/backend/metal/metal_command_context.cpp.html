metal_command_context.cpp

code/grassland/graphics/backend/metal/metal_command_context.cpp · 346 行 · 查看版本源码 ↗

职责与设计

Metal encoder录制、延迟render encoder创建与argument-buffer绑定。；本文件实现接口行为。

  • 资源按graphics/compute各自map暂存，实际Draw/Dispatch才验证已使用set及数量并分配argument buffer；BufferRange校验边界，未约束shader只读范围。
  • 原生AS只接受compute bind point，显式useResource TLAS/BLAS并捕获具体代资源到完成回调；sampler也通过回调保留。
  • Draw要求render pass和program，attachmentless pass必须有正viewport；viewport/scissor在同context跨pass保留。
  • compute dispatch后插入Buffer|Texture屏障；render pass内部禁止dispatch/copy/clear。索引固定uint32；CmdPresent结束当前encoder并录制呈现。
  • 部分dynamic_cast直接解引用而未检空，跨后端输入不安全；提交后进一步录制没有逐方法保护，应视context为一次性。

92 void MetalCommandContext::BindStage(MetalStage &stage,
93                                     const std::vector<MetalBinding> &layout,
94                                     BindPoint point,
95                                     bool vertex) {
96   MetalPool pool;
97   for (auto &[slot, encoder] : stage.arguments) {
98     auto it = resources_[point].find(slot);
99     if (it == resources_[point].end())
100       throw std::runtime_error("unbound Metal resource set " + std::to_string(slot));
101     auto &resources = it->second;
102     auto &binding = layout.at(slot);
103     auto argument =
104         NS::TransferPtr(core_->Device()->newBuffer(encoder->encodedLength(), MTL::ResourceStorageModeShared));
105     MetalCheck(argument.get(), nullptr, "argument buffer allocation");
106     encoder->setArgumentBuffer(argument.get(), 0);
107     MTL::ResourceUsage usage = MTL::ResourceUsageRead;
108     if (binding.type == RESOURCE_TYPE_WRITABLE_IMAGE || binding.type == RESOURCE_TYPE_WRITABLE_STORAGE_BUFFER)
109       usage |= MTL::ResourceUsageWrite;
110     auto use = [&](MTL::Resource *resource) {
111       if (point == BIND_POINT_COMPUTE)
112         compute_->useResource(resource, usage);
113       else
114         render_->useResource(resource, usage, vertex ? MTL::RenderStageVertex : MTL::RenderStageFragment);
115     };
116     size_t count = resources.buffers.size() + resources.images.size() + resources.samplers.size() +
117                    (resources.acceleration_structure ? 1 : 0);
118     if (count != binding.count)
119       throw std::runtime_error("Metal resource count mismatch at set " + std::to_string(slot));
120     if (resources.acceleration_structure) {
121       if (binding.type != RESOURCE_TYPE_ACCELERATION_STRUCTURE)
122         throw std::invalid_argument("Metal AS bound to a non-AS resource slot");
123       auto structure = dynamic_cast<MetalAccelerationStructure *>(resources.acceleration_structure);
124       encoder->setAccelerationStructure(structure->Handle(), 0);
125       use(structure->Handle());
126       for (const auto &child : structure->Children())
127         use(child.get());
128       // Capture the exact AS generation and dependencies until this submission completes.
129       PushPostExecutionCallback([storage = NS::RetainPtr(structure->Handle()), children = structure->Children()] {});
130     } else if (binding.type == RESOURCE_TYPE_ACCELERATION_STRUCTURE) {
131       throw std::invalid_argument("Metal AS slot requires an acceleration structure");
132     }
133     for (size_t i = 0; i < resources.buffers.size(); ++i) {
134       auto &range = resources.buffers[i];
135       auto buffer = dynamic_cast<MetalBuffer *>(range.buffer);
136       if (!buffer || range.offset > buffer->Size() || range.size > buffer->Size() - range.offset)
137         throw std::invalid_argument("invalid Metal buffer range");
138       encoder->setBuffer(buffer->Handle(), range.offset, i);
139       use(buffer->Handle());
140     }
141     for (size_t i = 0; i < resources.images.size(); ++i) {
142       auto image = dynamic_cast<MetalImage *>(resources.images[i]);
143       if (!image)
144         throw std::invalid_argument("expected MetalImage");
145       encoder->setTexture(image->Handle(), i);
146       use(image->Handle());
147     }
148     for (size_t i = 0; i < resources.samplers.size(); ++i) {
149       auto sampler = dynamic_cast<MetalSampler *>(resources.samplers[i]);
150       if (!sampler)
151         throw std::invalid_argument("expected MetalSampler");
152       encoder->setSamplerState(sampler->state.get(), i);
153       // Samplers are indirect objects, not MTL::Resources. Retain through completion.
154       PushPostExecutionCallback([state = sampler->state]() {});
155     }
156     if (point == BIND_POINT_COMPUTE)
157       compute_->setBuffer(argument.get(), 0, slot);
158     else if (vertex)
159       render_->setVertexBuffer(argument.get(), 0, slot);
160     else
161       render_->setFragmentBuffer(argument.get(), 0, slot);
162   }
163 }

165 void MetalCommandContext::CmdDispatch(uint32_t x, uint32_t y, uint32_t z) {
166   if (!compute_program_)
167     throw std::runtime_error("no compute program bound");
168   if (render_pass_)
169     throw std::runtime_error("compute dispatch inside render pass");
170   MetalPool pool;
171   if (!compute_)
172     compute_ = NS::RetainPtr(command_->computeCommandEncoder());
173   compute_->setComputePipelineState(compute_program_->pipeline.get());
174   BindStage(compute_program_->stage, compute_program_->bindings, BIND_POINT_COMPUTE);
175   compute_->dispatchThreadgroups(MTL::Size(x, y, z), compute_program_->stage.threads);
176   // BVH sorting/build/refit and traversal dispatches share storage buffers.
177   compute_->memoryBarrier(MTL::BarrierScopeBuffers | MTL::BarrierScopeTextures);
178 }

180 void MetalCommandContext::CmdBeginRendering(const std::vector<Image *> &colors, Image *depth) {
181   if (render_pass_)
182     throw std::runtime_error("nested Metal render pass");
183   EndEncoder();
184   MetalPool pool;
185   auto pass = MTL::RenderPassDescriptor::renderPassDescriptor();
186   for (size_t i = 0; i < colors.size(); ++i) {
187     auto attachment = pass->colorAttachments()->object(i);
188     attachment->setTexture(dynamic_cast<MetalImage *>(colors[i])->Handle());
189     attachment->setLoadAction(MTL::LoadActionLoad);
190     attachment->setStoreAction(MTL::StoreActionStore);
191   }
192   if (depth) {
193     pass->depthAttachment()->setTexture(dynamic_cast<MetalImage *>(depth)->Handle());
194     pass->depthAttachment()->setLoadAction(MTL::LoadActionLoad);
195     pass->depthAttachment()->setStoreAction(MTL::StoreActionStore);
196   }
197   render_pass_ = NS::RetainPtr(pass);
198   attachmentless_ = colors.empty() && !depth;
199 }

245 void MetalCommandContext::PrepareDraw() {
246   if (!render_pass_ || !program_)
247     throw std::runtime_error("draw needs a render pass and program");
248   if (!render_) {
249     MetalPool pool;
250     // Delay encoder creation until the viewport is known. A pass that only
251     // writes storage images still needs a rasterization extent on Metal.
252     if (attachmentless_) {
253       if (!has_viewport_ || viewport_.width <= 0 || viewport_.height <= 0)
254         throw std::runtime_error("attachmentless Metal pass requires a viewport");
255       render_pass_->setRenderTargetWidth(NS::UInteger(viewport_.originX + viewport_.width));
256       render_pass_->setRenderTargetHeight(NS::UInteger(viewport_.originY + viewport_.height));
257       render_pass_->setDefaultRasterSampleCount(1);
258     }
259     render_ = NS::RetainPtr(command_->renderCommandEncoder(render_pass_.get()));
260     MetalCheck(render_.get(), nullptr, "renderCommandEncoder");
261     // Grassland viewport/scissor state persists across passes in a context.
262     if (has_viewport_)
263       render_->setViewport(viewport_);
264     if (has_scissor_)
265       render_->setScissorRect(scissor_);
266   }
267   render_->setRenderPipelineState(program_->pipeline.get());
268   render_->setDepthStencilState(program_->depth_state.get());
269   render_->setCullMode(static_cast<MTL::CullMode>(program_->cull));
270   render_->setFrontFacingWinding(MTL::WindingCounterClockwise);
271   for (auto &[slot, range] : vertices_) {
272     auto buffer = dynamic_cast<MetalBuffer *>(range.buffer);
273     if (!buffer)
274       throw std::invalid_argument("expected Metal vertex buffer");
275     render_->setVertexBuffer(buffer->Handle(), range.offset, 16 + slot);
276   }
277 
278   BindStage(program_->vertex_stage, program_->bindings, BIND_POINT_GRAPHICS, true);
279   BindStage(program_->fragment_stage, program_->bindings, BIND_POINT_GRAPHICS, false);
280 }

300 void MetalCommandContext::CmdClearImage(Image *image, const ClearValue &value) {
301   if (render_pass_)
302     throw std::runtime_error("clear inside render pass");
303   EndEncoder();
304   MetalPool pool;
305   auto metal = dynamic_cast<MetalImage *>(image);
306   auto pass = MTL::RenderPassDescriptor::renderPassDescriptor();
307   if (IsDepthFormat(image->Format())) {
308     pass->depthAttachment()->setTexture(metal->Handle());
309     pass->depthAttachment()->setClearDepth(value.depth.depth);
310     pass->depthAttachment()->setLoadAction(MTL::LoadActionClear);
311     pass->depthAttachment()->setStoreAction(MTL::StoreActionStore);
312   } else {
313     auto attachment = pass->colorAttachments()->object(0);
314     attachment->setTexture(metal->Handle());
315     auto c = value.color;
316     attachment->setClearColor(MTL::ClearColor(c.r, c.g, c.b, c.a));
317     attachment->setLoadAction(MTL::LoadActionClear);
318     attachment->setStoreAction(MTL::StoreActionStore);
319   }
320   command_->renderCommandEncoder(pass)->endEncoding();
321 }
