职责与设计

AS handle、VMA存储与设备地址所有权，提供TLAS更新。；本文件实现接口行为。

  • 实例数量相同请求UPDATE，否则BUILD；helper容量不足也降为BUILD。上传实例至少分配一条并16字节对齐。
  • Update返回0/-1并刷新device address；未标记类型阻止对BLAS调用，未先WaitGPU，旧AS替换前需调用方同步。

30 int VulkanAccelerationStructure::UpdateInstances(const std::vector<RayTracingInstance> &instances) {
31   std::vector<VkAccelerationStructureInstanceKHR> vk_instances;
32   vk_instances.reserve(instances.size());
33   for (const auto &instance : instances)
34     vk_instances.emplace_back(RayTracingInstanceToVkAccelerationStructureInstanceKHR(instance));
35   VkBuffer instances_buffer = VK_NULL_HANDLE;
36   VmaAllocation instances_allocation = VK_NULL_HANDLE;
37   VkDeviceSize data_size = sizeof(VkAccelerationStructureInstanceKHR) * std::max(size_t(1), instances.size());
38   VkResult result = core_->CreateBuffer(
39       data_size,
40       VK_BUFFER_USAGE_ACCELERATION_STRUCTURE_BUILD_INPUT_READ_ONLY_BIT_KHR | VK_BUFFER_USAGE_SHADER_DEVICE_ADDRESS_BIT,
41       VMA_MEMORY_USAGE_CPU_TO_GPU, 0, 16, &instances_buffer, &instances_allocation);
42   if (result != VK_SUCCESS)
43     return -1;
44   void *mapped = nullptr;
45   result = vmaMapMemory(core_->Allocator(), instances_allocation, &mapped);
46   if (result != VK_SUCCESS) {
47     vmaDestroyBuffer(core_->Allocator(), instances_buffer, instances_allocation);
48     return -1;
49   }
50   if (!instances.empty())
51     std::memcpy(mapped, vk_instances.data(), vk_instances.size() * sizeof(vk_instances[0]));
52   vmaUnmapMemory(core_->Allocator(), instances_allocation);
53   VkAccelerationStructureGeometryKHR geometry{VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_GEOMETRY_KHR};
54   geometry.geometryType = VK_GEOMETRY_TYPE_INSTANCES_KHR;
55   geometry.flags = VK_GEOMETRY_OPAQUE_BIT_KHR;
56   geometry.geometry.instances.sType = VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_GEOMETRY_INSTANCES_DATA_KHR;
57   geometry.geometry.instances.arrayOfPointers = VK_FALSE;
58   geometry.geometry.instances.data.deviceAddress = core_->BufferAddress(instances_buffer);
59   auto mode = num_instance_ == instances.size() ? VK_BUILD_ACCELERATION_STRUCTURE_MODE_UPDATE_KHR
60                                                 : VK_BUILD_ACCELERATION_STRUCTURE_MODE_BUILD_KHR;
61   result = vulkan::BuildAccelerationStructure(
62       core_, geometry, VK_ACCELERATION_STRUCTURE_TYPE_TOP_LEVEL_KHR,
63       VK_BUILD_ACCELERATION_STRUCTURE_PREFER_FAST_TRACE_BIT_KHR | VK_BUILD_ACCELERATION_STRUCTURE_ALLOW_UPDATE_BIT_KHR,
64       mode, static_cast<uint32_t>(instances.size()), core_->GraphicsCommandPool(), core_->GraphicsQueue(), &as_,
65       &buffer_, &allocation_, &buffer_size_);
66   vmaDestroyBuffer(core_->Allocator(), instances_buffer, instances_allocation);
67   if (result != VK_SUCCESS)
68     return -1;
69   num_instance_ = static_cast<int>(instances.size());
70   VkAccelerationStructureDeviceAddressInfoKHR address_info{
71       VK_STRUCTURE_TYPE_ACCELERATION_STRUCTURE_DEVICE_ADDRESS_INFO_KHR};
72   address_info.accelerationStructure = as_;
73   device_address_ = core_->Procedures().vkGetAccelerationStructureDeviceAddressKHR(core_->Handle(), &address_info);
74   return 0;
75 }
