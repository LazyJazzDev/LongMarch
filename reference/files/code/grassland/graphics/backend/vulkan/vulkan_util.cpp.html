职责与设计

统一格式、shader/resource/bind point到Vulkan枚举转换。；本文件实现接口行为。

  • 未知格式UNDEFINED；未知shader stage ALL，未知resource MAX_ENUM，非法bind point抛异常。
  • RayTracingInstance复制3×4行主序transform、位字段及借用BLAS地址；AS cast未检空。

197       return VK_BLEND_OP_MAX;
198   }
199   return VK_BLEND_OP_ADD;
200 }

202 VkPipelineColorBlendAttachmentState BlendStateToVkPipelineColorBlendAttachmentState(const BlendState &state) {
203   VkPipelineColorBlendAttachmentState attachment{};
204   attachment.blendEnable = state.blend_enable;
205   attachment.srcColorBlendFactor = BlendFactorToVkBlendFactor(state.src_color);
206   attachment.dstColorBlendFactor = BlendFactorToVkBlendFactor(state.dst_color);
207   attachment.colorBlendOp = BlendOpToVkBlendOp(state.color_op);
208   attachment.srcAlphaBlendFactor = BlendFactorToVkBlendFactor(state.src_alpha);
209   attachment.dstAlphaBlendFactor = BlendFactorToVkBlendFactor(state.dst_alpha);
210   attachment.alphaBlendOp = BlendOpToVkBlendOp(state.alpha_op);
211   attachment.colorWriteMask =
212       VK_COLOR_COMPONENT_R_BIT | VK_COLOR_COMPONENT_G_BIT | VK_COLOR_COMPONENT_B_BIT | VK_COLOR_COMPONENT_A_BIT;
213   return attachment;
214 }
