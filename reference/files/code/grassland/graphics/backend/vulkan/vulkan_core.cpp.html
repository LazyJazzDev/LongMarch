职责与设计

统一资源工厂、设备选择、提交/呈现/完成回调与CUDA同步。；本文件实现接口行为。

  • 静态图像创建后同步转GENERAL；Slang使用spirv_1_5及dx-layout。
  • descriptor pool按每类与set数倍增，帧槽复用时reset；提交先上传动态Buffer，再加ALL_COMMANDS内存barrier。
  • 等待窗口image semaphore的stage设为COLOR_ATTACHMENT_OUTPUT，但Present路径首先transfer clear/blit；这不是完整验证过的transfer等待协议。
  • Submit末尾等待transfer_queue idle，而它实际是同一graphics queue，因此当前提交也被等待完成。
  • WaitGPU只等graphics queue并执行回调，未检查idle返回码；析构device idle但不执行剩余post callbacks。

253 int VulkanCore::SubmitCommandContext(CommandContext *p_command_context) {
254   VulkanCommandContext *command_context = dynamic_cast<VulkanCommandContext *>(p_command_context);
255 
256   auto &pool = descriptor_pools_[current_frame_];
257   auto pool_size = descriptor_pool_sizes_[current_frame_];
258   auto max_sets = descriptor_pool_max_sets_[current_frame_];
259   bool update_pool = false;
260   for (auto &[type, count] : command_context->required_pool_size_.descriptor_type_count) {
261     if (!pool_size.descriptor_type_count.count(type) || pool_size.descriptor_type_count[type] < count) {
262       uint32_t type_count = pool_size.descriptor_type_count[type];
263       if (!type_count) {
264         type_count = 1;
265       }
266       while (type_count < count) {
267         type_count *= 2;
268       }
269       pool_size.descriptor_type_count[type] = type_count;
270       update_pool = true;
271     }
272   }
273   while (max_sets < command_context->required_set_count_) {
274     max_sets *= 2;
275     update_pool = true;
276   }
277   if (update_pool) {
278     vkDestroyDescriptorPool(device_, pool, nullptr);
279     this->CreateDescriptorPool(pool_size.ToVkDescriptorPoolSize(), max_sets, &pool);
280     descriptor_pool_sizes_[current_frame_] = pool_size;
281     descriptor_pool_max_sets_[current_frame_] = max_sets;
282   } else {
283     vkResetDescriptorPool(device_, pool, 0);
284   }
285   current_descriptor_pool_ = pool;
286 
287   for (auto &window : command_context->windows_) {
288     window->AcquireNextImage();
289   }
290 
291   if (command_context->dynamic_buffers_.size()) {
292     vkResetCommandBuffer(transfer_command_buffer_, 0);
293     VkCommandBufferBeginInfo begin_info = {};
294     begin_info.sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_BEGIN_INFO;
295     begin_info.flags = VK_COMMAND_BUFFER_USAGE_ONE_TIME_SUBMIT_BIT;
296 
297     vkBeginCommandBuffer(transfer_command_buffer_, &begin_info);
298     for (auto &buffer : command_context->dynamic_buffers_) {
299       buffer->TransferData(transfer_command_buffer_);
300     }
301     vkEndCommandBuffer(transfer_command_buffer_);
302 
303     VkCommandBuffer command_buffers[] = {transfer_command_buffer_};
304 
305     VkSubmitInfo submit_info = {};
306     submit_info.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO;
307     submit_info.commandBufferCount = 1;
308     submit_info.pCommandBuffers = command_buffers;
309 
310 #if defined(LONGMARCH_CUDA_RUNTIME)
311     VkTimelineSemaphoreSubmitInfo timeline_info = {};
312     uint64_t wait_value;
313     uint64_t signal_value;
314     VkPipelineStageFlags wait_stages[] = {VK_PIPELINE_STAGE_ALL_COMMANDS_BIT};
315     if (cuda_device_ >= 0) {
316       wait_value = cuda_synchronization_value_++;
317       signal_value = cuda_synchronization_value_;
318       timeline_info.sType = VK_STRUCTURE_TYPE_TIMELINE_SEMAPHORE_SUBMIT_INFO;
319       timeline_info.waitSemaphoreValueCount = 1;
320       timeline_info.pWaitSemaphoreValues = &wait_value;
321       timeline_info.signalSemaphoreValueCount = 1;
322       timeline_info.pSignalSemaphoreValues = &signal_value;
323       timeline_info.pNext = submit_info.pNext;
324       submit_info.pNext = &timeline_info;
325       submit_info.signalSemaphoreCount = 1;
326       submit_info.pSignalSemaphores = &cuda_synchronization_semaphore_;
327       submit_info.pWaitDstStageMask = wait_stages;
328       submit_info.waitSemaphoreCount = 1;
329       submit_info.pWaitSemaphores = &cuda_synchronization_semaphore_;
330     }
331 #endif
332 
333     vulkan::ThrowIfFailed(vkQueueSubmit(transfer_queue_, 1, &submit_info, nullptr), "Submit Vulkan uploads");
334   }
335 
336   VkCommandBuffer command_buffer = command_buffers_[current_frame_];
337   vkResetCommandBuffer(command_buffer, 0);
338   VkCommandBufferBeginInfo begin_info{};
339   begin_info.sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_BEGIN_INFO;
340   begin_info.flags = VK_COMMAND_BUFFER_USAGE_ONE_TIME_SUBMIT_BIT;
341   if (vkBeginCommandBuffer(command_buffer, &begin_info) != VK_SUCCESS) {
342     return -1;
343   }
344 
345   // Queue order alone does not make copied uniforms/instances visible to shaders.
346   // This also orders static uploads and previous dispatches before this context.
347   VkMemoryBarrier upload_barrier{};
348   upload_barrier.sType = VK_STRUCTURE_TYPE_MEMORY_BARRIER;
349   upload_barrier.srcAccessMask = VK_ACCESS_MEMORY_WRITE_BIT;
350   upload_barrier.dstAccessMask = VK_ACCESS_MEMORY_READ_BIT | VK_ACCESS_MEMORY_WRITE_BIT;
351   vkCmdPipelineBarrier(command_buffer, VK_PIPELINE_STAGE_ALL_COMMANDS_BIT, VK_PIPELINE_STAGE_ALL_COMMANDS_BIT, 0, 1,
352                        &upload_barrier, 0, nullptr, 0, nullptr);
353 
354   for (auto &command : command_context->commands_) {
355     command->CompileCommand(command_context, command_buffer);
356   }
357 
358   for (auto [image, state] : command_context->image_states_) {
359     vulkan::TransitImageLayout(command_buffer, image, state.layout, VK_IMAGE_LAYOUT_GENERAL, state.stage,
360                                VK_PIPELINE_STAGE_ALL_COMMANDS_BIT, state.access, VK_ACCESS_MEMORY_READ_BIT,
361                                state.aspect);
362   }
363   command_context->image_states_.clear();
364 
365   vkEndCommandBuffer(command_buffer);
366 
367   std::vector<VkSemaphore> wait_semaphores;
368   std::vector<VkSemaphore> signal_semaphores;
369 
370   std::vector<VkPipelineStageFlags> wait_stages{};
371   for (auto &window : command_context->windows_) {
372     wait_semaphores.push_back(window->ImageAvailableSemaphore());
373     wait_stages.push_back(VK_PIPELINE_STAGE_COLOR_ATTACHMENT_OUTPUT_BIT);
374     signal_semaphores.push_back(window->RenderFinishSemaphore());
375   }
376 
377   VkFence fence = in_flight_fences_[current_frame_];
378   vkResetFences(this->Handle(), 1, &fence);
379 
380   VkSubmitInfo submit_info{};
381   submit_info.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO;
382   submit_info.commandBufferCount = 1;
383   submit_info.pCommandBuffers = &command_buffer;
384 
385 #if defined(LONGMARCH_CUDA_RUNTIME)
386   VkTimelineSemaphoreSubmitInfo timeline_info = {};
387   std::vector<uint64_t> wait_values(wait_semaphores.size() + 1, 0);
388   std::vector<uint64_t> signal_values(signal_semaphores.size() + 1, 0);
389   if (cuda_device_ >= 0) {
390     wait_values[wait_semaphores.size()] = cuda_synchronization_value_++;
391     signal_values[signal_semaphores.size()] = cuda_synchronization_value_;
392     timeline_info.sType = VK_STRUCTURE_TYPE_TIMELINE_SEMAPHORE_SUBMIT_INFO;
393     timeline_info.waitSemaphoreValueCount = wait_values.size();
394     timeline_info.pWaitSemaphoreValues = wait_values.data();
395     timeline_info.signalSemaphoreValueCount = signal_values.size();
396     timeline_info.pSignalSemaphoreValues = signal_values.data();
397     timeline_info.pNext = submit_info.pNext;
398     submit_info.pNext = &timeline_info;
399     wait_semaphores.push_back(cuda_synchronization_semaphore_);
400     wait_stages.push_back(VK_PIPELINE_STAGE_ALL_COMMANDS_BIT);
401     signal_semaphores.push_back(cuda_synchronization_semaphore_);
402   }
403 #endif
404 
405   if (!wait_semaphores.empty()) {
406     submit_info.waitSemaphoreCount = wait_semaphores.size();
407     submit_info.pWaitSemaphores = wait_semaphores.data();
408     submit_info.pWaitDstStageMask = wait_stages.data();
409     submit_info.signalSemaphoreCount = signal_semaphores.size();
410     submit_info.pSignalSemaphores = signal_semaphores.data();
411   }
412 
413   vulkan::ThrowIfFailed(vkQueueSubmit(graphics_queue_, 1, &submit_info, fence), "Submit Vulkan rendering");
414 
415   for (auto &window : command_context->windows_) {
416     window->Present();
417   }
418 
419   post_execute_functions_[current_frame_] = p_command_context->GetPostExecutionCallbacks();
420 
421   current_frame_ = (current_frame_ + 1) % FramesInFlight();
422   fence = in_flight_fences_[current_frame_];
423   vulkan::ThrowIfFailed(vkWaitForFences(this->Handle(), 1, &fence, VK_TRUE, std::numeric_limits<uint64_t>::max()),
424                         "Wait for Vulkan rendering");
425 
426   vulkan::ThrowIfFailed(vkQueueWaitIdle(transfer_queue_), "Wait for Vulkan uploads");
427 
428   for (auto &callback : post_execute_functions_[current_frame_]) {
429     callback();
430   }
431   post_execute_functions_[current_frame_].clear();
432 
433   return 0;
434 }

463 int VulkanCore::InitializeLogicalDevice(int device_index) {
464   std::vector<VkPhysicalDevice> physical_devices = vulkan::EnumerateNativePhysicalDevices(instance_);
465 
466   if (device_index < 0 || device_index >= physical_devices.size()) {
467     return -1;
468   }
469 
470   auto physical_device = physical_devices[device_index];
471 
472 #if defined(LONGMARCH_CUDA_RUNTIME)
473   cuda_device_ = vulkan::GetCUDADeviceIndex(physical_device);
474 #endif
475 
476   grassland::graphics::backend::vulkan::DeviceFeatureRequirement device_feature_requirement{};
477   device_feature_requirement.enable_raytracing_extension = vulkan::SupportRayTracing(physical_device);
478   device_feature_requirement.enable_rayquery_extension = vulkan::SupportRayQuery(physical_device);
479   auto create_info = device_feature_requirement.GenerateRecommendedDeviceCreateInfo(physical_device);
480   if (instance_hint_.IsEnabledExtension(VK_KHR_SURFACE_EXTENSION_NAME) &&
481       vulkan::IsExtensionSupported(physical_device, VK_KHR_SWAPCHAIN_EXTENSION_NAME)) {
482     create_info.AddExtension(VK_KHR_SWAPCHAIN_EXTENSION_NAME);
483   }
484 #if defined(LONGMARCH_CUDA_RUNTIME)
485   if (cuda_device_ >= 0) {
486     create_info.AddExtension(VK_KHR_EXTERNAL_MEMORY_EXTENSION_NAME);
487     create_info.AddExtension(VK_KHR_EXTERNAL_SEMAPHORE_EXTENSION_NAME);
488     create_info.AddExtension(VK_KHR_TIMELINE_SEMAPHORE_EXTENSION_NAME);
489 #ifdef _WIN64
490     create_info.AddExtension(VK_KHR_EXTERNAL_MEMORY_WIN32_EXTENSION_NAME);
491     create_info.AddExtension(VK_KHR_EXTERNAL_SEMAPHORE_WIN32_EXTENSION_NAME);
492 #else
493     create_info.AddExtension(VK_KHR_EXTERNAL_MEMORY_FD_EXTENSION_NAME);
494     create_info.AddExtension(VK_KHR_EXTERNAL_SEMAPHORE_FD_EXTENSION_NAME);
495 #endif /* _WIN64 */
496     VkPhysicalDeviceTimelineSemaphoreFeatures physical_device_timeline_semaphore_features{};
497     physical_device_timeline_semaphore_features.sType = VK_STRUCTURE_TYPE_PHYSICAL_DEVICE_TIMELINE_SEMAPHORE_FEATURES;
498     physical_device_timeline_semaphore_features.timelineSemaphore = VK_TRUE;
499     create_info.AddFeature(physical_device_timeline_semaphore_features);
500   }
501 #endif
502   auto native_create_info =
503       create_info.CompileVkDeviceCreateInfo(instance_hint_.enable_validation_layers, physical_device);
504   VkDevice native_device = VK_NULL_HANDLE;
505   if (vkCreateDevice(physical_device, &native_create_info, nullptr, &native_device) != VK_SUCCESS) {
506     return -1;
507   }
508   InitializeNativeDevice(instance_hint_.app_info.apiVersion, physical_device, create_info,
509                          device_feature_requirement.GetVmaAllocatorCreateFlags(), native_device);
510   memory_properties_ = vulkan::GetPhysicalDeviceMemoryProperties(physical_device);
511 
512   device_name_ = vulkan::GetPhysicalDeviceProperties(physical_device).deviceName;
513   ray_tracing_support_ = vulkan::SupportRayTracing(physical_device);
514 
515   VkCommandPoolCreateInfo pool_info{VK_STRUCTURE_TYPE_COMMAND_POOL_CREATE_INFO};
516   pool_info.flags = VK_COMMAND_POOL_CREATE_RESET_COMMAND_BUFFER_BIT;
517   pool_info.queueFamilyIndex = vulkan::GraphicsFamilyIndex(physical_device);
518   vulkan::ThrowIfFailed(vkCreateCommandPool(native_device, &pool_info, nullptr, &graphics_command_pool_),
519                         "failed to create graphics command pool");
520   // Upload buffers use exclusive sharing and are consumed by the graphics queue.
521   pool_info.queueFamilyIndex = vulkan::GraphicsFamilyIndex(physical_device);
522   vulkan::ThrowIfFailed(vkCreateCommandPool(native_device, &pool_info, nullptr, &transfer_command_pool_),
523                         "failed to create transfer command pool");
524 
525   vkGetDeviceQueue(native_device, vulkan::GraphicsFamilyIndex(physical_device), 0, &graphics_queue_);
526   vkGetDeviceQueue(native_device, vulkan::GraphicsFamilyIndex(physical_device), 0, &transfer_queue_);
527 
528   in_flight_fences_.resize(FramesInFlight());
529   command_buffers_.resize(FramesInFlight());
530 
531   descriptor_pools_.resize(FramesInFlight());
532   descriptor_pool_sizes_.resize(FramesInFlight());
533   descriptor_pool_max_sets_.resize(FramesInFlight(), 1);
534 
535   post_execute_functions_.resize(FramesInFlight());
536 
537   for (int i = 0; i < FramesInFlight(); i++) {
538     VkFenceCreateInfo fence_info{VK_STRUCTURE_TYPE_FENCE_CREATE_INFO};
539     fence_info.flags = VK_FENCE_CREATE_SIGNALED_BIT;
540     vulkan::ThrowIfFailed(vkCreateFence(this->Handle(), &fence_info, nullptr, &in_flight_fences_[i]),
541                           "Failed to create in-flight fence");
542     VkCommandBufferAllocateInfo allocate_info{VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO};
543     allocate_info.commandPool = graphics_command_pool_;
544     allocate_info.level = VK_COMMAND_BUFFER_LEVEL_PRIMARY;
545     allocate_info.commandBufferCount = 1;
546     vulkan::ThrowIfFailed(vkAllocateCommandBuffers(native_device, &allocate_info, &command_buffers_[i]),
547                           "failed to allocate graphics command buffer");
548 
549     this->CreateDescriptorPool({{VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER, 1}}, 1, &descriptor_pools_[i]);
550     descriptor_pool_sizes_[i] = vulkan::DescriptorPoolSize(VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER, 1);
551   }
552   VkCommandBufferAllocateInfo transfer_allocate_info{VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO};
553   transfer_allocate_info.commandPool = transfer_command_pool_;
554   transfer_allocate_info.level = VK_COMMAND_BUFFER_LEVEL_PRIMARY;
555   transfer_allocate_info.commandBufferCount = 1;
556   vulkan::ThrowIfFailed(vkAllocateCommandBuffers(native_device, &transfer_allocate_info, &transfer_command_buffer_),
557                         "failed to allocate transfer command buffer");
558 
559 #if defined(LONGMARCH_CUDA_RUNTIME)
560 
561   if (cuda_device_ >= 0) {
562     // Create a timeline semaphore
563     auto handle_type = vulkan::GetDefaultExternalSemaphoreHandleType();
564     VkSemaphoreCreateInfo semaphore_create_info{};
565     semaphore_create_info.sType = VK_STRUCTURE_TYPE_SEMAPHORE_CREATE_INFO;
566     VkExportSemaphoreCreateInfoKHR export_semaphore_create_info{};
567     export_semaphore_create_info.sType = VK_STRUCTURE_TYPE_EXPORT_SEMAPHORE_CREATE_INFO;
568 
569     VkSemaphoreTypeCreateInfo export_semaphore_create_info_timeline{};
570     export_semaphore_create_info_timeline.sType = VK_STRUCTURE_TYPE_SEMAPHORE_TYPE_CREATE_INFO;
571     export_semaphore_create_info_timeline.initialValue = 0;
572     export_semaphore_create_info_timeline.semaphoreType = VK_SEMAPHORE_TYPE_TIMELINE;
573     export_semaphore_create_info.pNext = &export_semaphore_create_info_timeline;
574     export_semaphore_create_info.handleTypes = handle_type;
575 
576     semaphore_create_info.pNext = &export_semaphore_create_info;
577     vulkan::ThrowIfFailed(
578         vkCreateSemaphore(this->Handle(), &semaphore_create_info, nullptr, &cuda_synchronization_semaphore_),
579         "Failed to create CUDA synchronization semaphore");
580 
581     cudaExternalSemaphoreHandleDesc external_semaphore_handle_desc = {};
582 
583     if (handle_type & VK_EXTERNAL_SEMAPHORE_HANDLE_TYPE_OPAQUE_WIN32_BIT) {
584       external_semaphore_handle_desc.type = cudaExternalSemaphoreHandleTypeTimelineSemaphoreWin32;
585     } else if (handle_type & VK_EXTERNAL_SEMAPHORE_HANDLE_TYPE_OPAQUE_WIN32_KMT_BIT) {
586       external_semaphore_handle_desc.type = cudaExternalSemaphoreHandleTypeTimelineSemaphoreWin32;
587     } else if (handle_type & VK_EXTERNAL_SEMAPHORE_HANDLE_TYPE_OPAQUE_FD_BIT) {
588       external_semaphore_handle_desc.type = cudaExternalSemaphoreHandleTypeTimelineSemaphoreFd;
589     } else {
590       throw std::runtime_error("Unknown handle type requested!");
591     }
592 
593 #ifdef _WIN64
594     external_semaphore_handle_desc.handle.win32.handle = (HANDLE)GetSemaphoreHandle(cuda_synchronization_semaphore_);
595 #else
596     external_semaphore_handle_desc.handle.fd = (int)(uintptr_t)GetSemaphoreHandle(cuda_synchronization_semaphore_);
597 #endif
598 
599     external_semaphore_handle_desc.flags = 0;
600 
601     int current_cuda_device;
602     cudaGetDevice(&current_cuda_device);
603     cudaSetDevice(cuda_device_);
604 
605     cudaImportExternalSemaphore(&cuda_external_semaphore_, &external_semaphore_handle_desc);
606     cudaSetDevice(current_cuda_device);
607   }
608 
609 #endif
610 
611   return 0;
612 }
