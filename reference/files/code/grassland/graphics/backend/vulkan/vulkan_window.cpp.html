vulkan_window.cpp

code/grassland/graphics/backend/vulkan/vulkan_window.cpp · 460 行 · 查看版本源码 ↗

职责与设计

surface、swapchain、HDR切换、semaphore和兼容ImGui。；本文件实现接口行为。

  • HDR只接受RGBA16F/scRGB线性或两种10bit/PQ组合，不静默回退SDR，并要求目标格式BLIT_DST。
  • Rebuild尝试用oldSwapchain创建，失败后重建旧模式；恢复再失败则标记presentation_failed，禁止后续Acquire/Present。
  • semaphore数组按初始交换链image数分配，但按Core CurrentFrame索引；FramesInFlight须不超过数组长度，重建不调整semaphore数组。
  • Acquire把SUBOPTIMAL/OUT_OF_DATE直接视为异常；Present才对这两种结果Rebuild。
  • HDRFramebuffer只首次创建，必须与当前合成图像保持一致；Rebuild清缓存。

12   swap_chain_image_views_.clear();
13   swap_chain_images_.clear();
14   if (swap_chain_) {
15     vkDestroySwapchainKHR(core_->Handle(), swap_chain_, nullptr);
16     swap_chain_ = VK_NULL_HANDLE;
17   }
18 }

20 VkResult VulkanWindow::CreatePresentationSwapchain(VkSurfaceFormatKHR surface_format,
21                                                    VkSwapchainKHR *result,
22                                                    VkSwapchainKHR old_swapchain) {
23   VkDevice device = core_->Handle();
24   auto physical_device = core_->PhysicalDevice();
25   auto support = vulkan::QuerySwapChainSupport(physical_device, surface_);
26   auto present_mode = vulkan::ChooseSwapPresentMode(support.presentModes);
27   swap_chain_extent_ = vulkan::ChooseSwapExtent(support.capabilities, GLFWWindow());
28   uint32_t image_count = support.capabilities.minImageCount + 1;
29   if (support.capabilities.maxImageCount && image_count > support.capabilities.maxImageCount) {
30     image_count = support.capabilities.maxImageCount;
31   }
32   VkSwapchainCreateInfoKHR create_info{VK_STRUCTURE_TYPE_SWAPCHAIN_CREATE_INFO_KHR};
33   create_info.surface = surface_;
34   create_info.minImageCount = image_count;
35   create_info.imageFormat = surface_format.format;
36   create_info.imageColorSpace = surface_format.colorSpace;
37   create_info.imageExtent = swap_chain_extent_;
38   create_info.imageArrayLayers = 1;
39   create_info.imageUsage = VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT | VK_IMAGE_USAGE_TRANSFER_DST_BIT;
40   uint32_t families[] = {vulkan::GraphicsFamilyIndex(physical_device),
41                          vulkan::PresentFamilyIndex(physical_device, surface_)};
42   if (families[0] != families[1]) {
43     create_info.imageSharingMode = VK_SHARING_MODE_CONCURRENT;
44     create_info.queueFamilyIndexCount = 2;
45     create_info.pQueueFamilyIndices = families;
46   } else {
47     create_info.imageSharingMode = VK_SHARING_MODE_EXCLUSIVE;
48   }
49   create_info.preTransform = support.capabilities.currentTransform;
50   create_info.compositeAlpha = VK_COMPOSITE_ALPHA_OPAQUE_BIT_KHR;
51   create_info.presentMode = present_mode;
52   create_info.clipped = VK_TRUE;
53   create_info.oldSwapchain = old_swapchain;
54   return vkCreateSwapchainKHR(device, &create_info, nullptr, result);
55 }

57 void VulkanWindow::LoadSwapChainImages() {
58   VkDevice device = core_->Handle();
59   uint32_t image_count = 0;
60   vulkan::ThrowIfFailed(vkGetSwapchainImagesKHR(device, swap_chain_, &image_count, nullptr),
61                         "Failed to query Vulkan swap chain images");
62   swap_chain_images_.resize(image_count);
63   vulkan::ThrowIfFailed(vkGetSwapchainImagesKHR(device, swap_chain_, &image_count, swap_chain_images_.data()),
64                         "Failed to get Vulkan swap chain images");
65   swap_chain_image_views_.resize(image_count);
66   for (size_t i = 0; i < image_count; ++i) {
67     VkImageViewCreateInfo view_info{VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO};
68     view_info.image = swap_chain_images_[i];
69     view_info.viewType = VK_IMAGE_VIEW_TYPE_2D;
70     view_info.format = swap_chain_format_;
71     view_info.subresourceRange.aspectMask = VK_IMAGE_ASPECT_COLOR_BIT;
72     view_info.subresourceRange.levelCount = 1;
73     view_info.subresourceRange.layerCount = 1;
74     vulkan::ThrowIfFailed(vkCreateImageView(device, &view_info, nullptr, &swap_chain_image_views_[i]),
75                           "Failed to create Vulkan swap chain image view");
76   }
77 }

79 void VulkanWindow::CreateSwapChain() {
80   const auto format = SelectSurfaceFormat(enable_hdr_);
81   if (!format)
82     throw std::runtime_error("No supported Vulkan presentation format");
83   vulkan::ThrowIfFailed(CreatePresentationSwapchain(*format, &swap_chain_, VK_NULL_HANDLE),
84                         "Failed to create Vulkan swap chain");
85   surface_format_ = *format;
86   swap_chain_format_ = format->format;
87   LoadSwapChainImages();
88 }

90 VulkanWindow::VulkanWindow(VulkanCore *core,
91                            int width,
92                            int height,
93                            const std::string &title,
94                            bool fullscreen,
95                            bool resizable,
96                            bool enable_hdr)
97     : Window(width, height, title, fullscreen, resizable, enable_hdr),
98       core_(core) {
99   vulkan::ThrowIfFailed(glfwCreateWindowSurface(core_->Instance(), GLFWWindow(), nullptr, &surface_),
100                         "Failed to create window surface");
101   CreateSwapChain();
102   image_available_semaphores_.resize(swap_chain_images_.size());
103   render_finish_semaphores_.resize(swap_chain_images_.size());

106     vulkan::ThrowIfFailed(vkCreateSemaphore(core_->Handle(), &semaphore_info, nullptr, &image_available_semaphores_[i]),
107                           "Failed to create image available semaphore");
108     vulkan::ThrowIfFailed(vkCreateSemaphore(core_->Handle(), &semaphore_info, nullptr, &render_finish_semaphores_[i]),
109                           "Failed to create render finish semaphore");
110   }
111   vkGetDeviceQueue(core_->Handle(), vulkan::PresentFamilyIndex(core_->PhysicalDevice(), surface_), 0, &present_queue_);
112   FramebufferResizeEvent().RegisterCallback([this](int width, int height) {
113     if (width > 0 && height > 0 && Rebuild() != 0)
114       LogError("Failed to resize Vulkan presentation");
115   });
116 }

124 void VulkanWindow::CloseWindow() {
125   core_->WaitGPU();
126   if (hdr_framebuffer_ != VK_NULL_HANDLE) {
127     vkDestroyFramebuffer(core_->Handle(), hdr_framebuffer_, nullptr);
128     hdr_framebuffer_ = VK_NULL_HANDLE;
129   }
130   if (imgui_assets_.context) {
131     TerminateImGui();
132   }
133   for (VkSemaphore semaphore : image_available_semaphores_) {
134     vkDestroySemaphore(core_->Handle(), semaphore, nullptr);
135   }
136   image_available_semaphores_.clear();
137   for (VkSemaphore semaphore : render_finish_semaphores_) {
138     vkDestroySemaphore(core_->Handle(), semaphore, nullptr);
139   }
140   render_finish_semaphores_.clear();
141   DestroySwapChain();
142   vkDestroySurfaceKHR(core_->Instance(), surface_, nullptr);
143   surface_ = VK_NULL_HANDLE;
144   Window::CloseWindow();
145 }

147 int VulkanWindow::SetHDR(bool enable_hdr) {
148   if (!GLFWWindow() || presentation_failed_)
149     return -1;
150   if (enable_hdr == enable_hdr_)
151     return 0;
152   if (!SelectSurfaceFormat(enable_hdr)) {
153     LogWarning("Requested Vulkan HDR/SDR presentation mode is unavailable");
154     return -1;
155   }
156   const bool previous = enable_hdr_;
157   enable_hdr_ = enable_hdr;
158   try {
159     if (Rebuild() != 0) {
160       enable_hdr_ = previous;
161       return -1;
162     }
163     return Window::SetHDR(enable_hdr);
164   } catch (const std::exception &error) {
165     enable_hdr_ = previous;
166     LogError("Failed to change Vulkan HDR presentation: {}", error.what());
167     return -1;
168   }
169 }

171 std::optional<VkSurfaceFormatKHR> VulkanWindow::ChooseHDRSurfaceFormat(const std::vector<VkSurfaceFormatKHR> &formats) {
172   for (const auto &candidate :
173        {VkSurfaceFormatKHR{VK_FORMAT_R16G16B16A16_SFLOAT, VK_COLOR_SPACE_EXTENDED_SRGB_LINEAR_EXT},
174         VkSurfaceFormatKHR{VK_FORMAT_A2B10G10R10_UNORM_PACK32, VK_COLOR_SPACE_HDR10_ST2084_EXT},
175         VkSurfaceFormatKHR{VK_FORMAT_A2R10G10B10_UNORM_PACK32, VK_COLOR_SPACE_HDR10_ST2084_EXT}}) {
176     for (const auto &format : formats) {
177       if (format.format == candidate.format && format.colorSpace == candidate.colorSpace)
178         return format;
179     }
180   }
181   return std::nullopt;
182 }

184 std::optional<VkSurfaceFormatKHR> VulkanWindow::SelectSurfaceFormat(bool hdr) const {
185   const auto support = vulkan::QuerySwapChainSupport(core_->PhysicalDevice(), surface_);
186   if (support.formats.empty())
187     return std::nullopt;
188   auto format = hdr ? ChooseHDRSurfaceFormat(support.formats)
189                     : std::optional<VkSurfaceFormatKHR>(vulkan::ChooseSwapSurfaceFormat(
190                           support.formats, VK_FORMAT_R8G8B8A8_UNORM, VK_COLOR_SPACE_SRGB_NONLINEAR_KHR));
191   if (!format || (!hdr && format->colorSpace != VK_COLOR_SPACE_SRGB_NONLINEAR_KHR))
192     return std::nullopt;
193   VkFormatProperties properties{};
194   vkGetPhysicalDeviceFormatProperties(core_->PhysicalDevice(), format->format, &properties);
195   if (!(properties.optimalTilingFeatures & VK_FORMAT_FEATURE_BLIT_DST_BIT))
196     return std::nullopt;
197   return format;
198 }

225 int VulkanWindow::Rebuild() {
226   if (presentation_failed_)
227     return -1;
228   core_->WaitGPU();
229   const auto format = SelectSurfaceFormat(enable_hdr_);
230   if (!format)
231     return -1;
232   const auto previous_format = surface_format_;
233   const bool previous_hdr = previous_format.colorSpace != VK_COLOR_SPACE_SRGB_NONLINEAR_KHR;
234   VkSwapchainKHR next = VK_NULL_HANDLE;
235   const auto status = CreatePresentationSwapchain(*format, &next, swap_chain_);
236   if (hdr_framebuffer_ != VK_NULL_HANDLE) {
237     vkDestroyFramebuffer(core_->Handle(), hdr_framebuffer_, nullptr);
238     hdr_framebuffer_ = VK_NULL_HANDLE;
239   }
240   if (imgui_assets_.context) {
241     DestroyImGuiFramebuffers();
242   }
243   DestroySwapChain();
244   bool recovered = false;
245   if (status != VK_SUCCESS) {
246     if (next != VK_NULL_HANDLE)
247       vkDestroySwapchainKHR(core_->Handle(), next, nullptr);
248     enable_hdr_ = previous_hdr;
249     if (CreatePresentationSwapchain(previous_format, &next, VK_NULL_HANDLE) != VK_SUCCESS) {
250       presentation_failed_ = true;
251       LogError("Vulkan presentation disabled: could not recreate the previous swapchain mode");
252       return -1;
253     }
254     recovered = true;
255   }
256   swap_chain_ = next;
257   surface_format_ = recovered ? previous_format : *format;
258   swap_chain_format_ = surface_format_.format;
259   LoadSwapChainImages();
260   if (imgui_assets_.context) {
261     ImGui::SetCurrentContext(imgui_assets_.context);
262     if (imgui_assets_.render_pass_format != ImGuiFormat()) {
263       ImGui_ImplVulkan_Shutdown();
264       ImGui_ImplGlfw_Shutdown();
265 
266       ImGui::DestroyContext(imgui_assets_.context);
267       DestroyImGuiRenderPass();
268       SetupImGuiContext();
269     }
270 
271     BuildImGuiFramebuffers();
272   }
273   return recovered ? -1 : 0;
274 }

276 void VulkanWindow::Present() {
277   if (presentation_failed_)
278     throw std::runtime_error("Vulkan presentation unavailable after swapchain recovery failed");
279   VkSemaphore render_finish_semaphore = RenderFinishSemaphore();
280 
281   VkSwapchainKHR swap_chain = swap_chain_;
282 
283   VkPresentInfoKHR presentInfo{};
284   presentInfo.sType = VK_STRUCTURE_TYPE_PRESENT_INFO_KHR;
285   presentInfo.waitSemaphoreCount = 1;
286   presentInfo.pWaitSemaphores = &render_finish_semaphore;
287   presentInfo.swapchainCount = 1;
288   presentInfo.pSwapchains = &swap_chain;
289   presentInfo.pImageIndices = &image_index_;
290 
291   auto result = vkQueuePresentKHR(present_queue_, &presentInfo);
292   if (result == VK_ERROR_OUT_OF_DATE_KHR || result == VK_SUBOPTIMAL_KHR) {
293     if (Rebuild() != 0)
294       throw std::runtime_error("Failed to rebuild Vulkan swapchain after presentation");
295   } else if (result != VK_SUCCESS) {
296     throw std::runtime_error("Failed to present swap chain image");
297   }
298 }

425   }
