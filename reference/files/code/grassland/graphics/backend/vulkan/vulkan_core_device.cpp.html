职责与设计

VMA资源工厂、dynamic rendering管线、原生AS和SBT构建。；本文件实现接口行为。

  • 原生CreateImage/Buffer成功才写输出；ImageView失败会清理image allocation。
  • 三角BLAS使用float3、uint32索引及16字节对齐identity变换Buffer，maxVertex当前赋num_vertex而非num_vertex-1。
  • RT pipeline递归深度固定1；SBT record按handle alignment、表与地址按base alignment。
  • SBT查询group_count按索引数组总长度计算，复制时却用注册shader数量计算偏移；重复/子集重排可使计数与真实pipeline groups不符。未检查索引与GetHandles返回值。

414       VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
415       nullptr,
416       0,
417       VK_SHADER_STAGE_RAYGEN_BIT_KHR,

419       ray_gen_shader->EntryPointRef().c_str(),
420       nullptr,
421   });
422

432       miss_group_ci.intersectionShader = VK_SHADER_UNUSED_KHR;

438         0,

443     });
444   }

452     hit_group_ci.generalShader = VK_SHADER_UNUSED_KHR;
453     hit_group_ci.closestHitShader = VK_SHADER_UNUSED_KHR;

455     hit_group_ci.intersectionShader = VK_SHADER_UNUSED_KHR;
456 
457     hit_group_ci.closestHitShader = shader_stage_create_infos.size();

459         VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
460         nullptr,
461         0,
462         VK_SHADER_STAGE_CLOSEST_HIT_BIT_KHR,
463         hit_group.closest_hit_shader->ModuleHandle(),
464         hit_group.closest_hit_shader->EntryPointRef().c_str(),
465         nullptr,
466     });
467 
468     if (hit_group.any_hit_shader) {

471           VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
472           nullptr,
473           0,
474           VK_SHADER_STAGE_ANY_HIT_BIT_KHR,

476           hit_group.any_hit_shader->EntryPointRef().c_str(),

478       });
479     }

485           VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
486           nullptr,
487           0,
488           VK_SHADER_STAGE_INTERSECTION_BIT_KHR,
489           hit_group.intersection_shader->ModuleHandle(),
490           hit_group.intersection_shader->EntryPointRef().c_str(),
491           nullptr,
492       });
493     }
494 
495     shader_groups.push_back(hit_group_ci);
496   }
497 
498   for (auto callable_shader : callable_shaders) {
499     // Ray miss group
500     {
501       VkRayTracingShaderGroupCreateInfoKHR callable_group_ci{};
502       callable_group_ci.sType = VK_STRUCTURE_TYPE_RAY_TRACING_SHADER_GROUP_CREATE_INFO_KHR;
503       callable_group_ci.type = VK_RAY_TRACING_SHADER_GROUP_TYPE_GENERAL_KHR;

505       callable_group_ci.closestHitShader = VK_SHADER_UNUSED_KHR;
506       callable_group_ci.anyHitShader = VK_SHADER_UNUSED_KHR;
507       callable_group_ci.intersectionShader = VK_SHADER_UNUSED_KHR;
508       shader_groups.push_back(callable_group_ci);
509     }
510     shader_stage_create_infos.push_back({
511         VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO,
512         nullptr,
513         0,
514         VK_SHADER_STAGE_CALLABLE_BIT_KHR,
515         callable_shader->ModuleHandle(),
516         callable_shader->EntryPointRef().c_str(),
517         nullptr,
518     });
519   }
520 
521   VkRayTracingPipelineCreateInfoKHR ray_tracing_pipeline_create_info{};
522   ray_tracing_pipeline_create_info.sType = VK_STRUCTURE_TYPE_RAY_TRACING_PIPELINE_CREATE_INFO_KHR;
523   ray_tracing_pipeline_create_info.stageCount = shader_stage_create_infos.size();
524   ray_tracing_pipeline_create_info.pStages = shader_stage_create_infos.data();
525   ray_tracing_pipeline_create_info.groupCount = shader_groups.size();
526   ray_tracing_pipeline_create_info.pGroups = shader_groups.data();
527   ray_tracing_pipeline_create_info.maxPipelineRayRecursionDepth = 1;
528   ray_tracing_pipeline_create_info.layout = pipeline_layout;
529   ray_tracing_pipeline_create_info.basePipelineHandle = VK_NULL_HANDLE;
530 
531   VkPipeline pipeline;
532   RETURN_IF_FAILED_VK(procedures_.vkCreateRayTracingPipelinesKHR(device_, VK_NULL_HANDLE, VK_NULL_HANDLE, 1,
533                                                                  &ray_tracing_pipeline_create_info, nullptr, &pipeline),
534                       "failed to create ray tracing pipeline!");
535 
536   *pp_pipeline = pipeline;
537 
538   return VK_SUCCESS;
539 }

554 
555   VkPhysicalDeviceRayTracingPipelinePropertiesKHR ray_tracing_pipeline_properties =
556       vulkan::GetPhysicalDeviceRayTracingPipelineProperties(physical_device_);
557 
558   const uint32_t handle_size = ray_tracing_pipeline_properties.shaderGroupHandleSize;
559   const uint32_t handle_size_aligned = aligned_size(ray_tracing_pipeline_properties.shaderGroupHandleSize,
560                                                     ray_tracing_pipeline_properties.shaderGroupHandleAlignment);
561   const uint32_t base_alignment = ray_tracing_pipeline_properties.shaderGroupBaseAlignment;
562   const uint32_t group_count =
563       1 + miss_shader_indices.size() + hit_group_indices.size() + callable_shader_indices.size();
564   VkDeviceSize raygen_shader_offset = 0;
565   VkDeviceSize miss_shader_offset = 0;
566   VkDeviceSize hit_group_offset = 0;
567   VkDeviceSize callable_shader_offset = 0;
568   VkDeviceSize sbt_size = 0;
569   miss_shader_offset = sbt_size = aligned_size(sbt_size + handle_size_aligned * 1, base_alignment);
570   hit_group_offset = sbt_size =
571       aligned_size(sbt_size + handle_size_aligned * miss_shader_indices.size(), base_alignment);
572   callable_shader_offset = sbt_size =
573       aligned_size(sbt_size + handle_size_aligned * hit_group_indices.size(), base_alignment);
574   sbt_size = aligned_size(sbt_size + handle_size_aligned * callable_shader_indices.size(), base_alignment);
575   const VkBufferUsageFlags sbt_buffer_usage_flags = VK_BUFFER_USAGE_SHADER_BINDING_TABLE_BIT_KHR |
576                                                     VK_BUFFER_USAGE_TRANSFER_SRC_BIT |
577                                                     VK_BUFFER_USAGE_SHADER_DEVICE_ADDRESS_BIT;
578 
579   // Ray_gen
580   // Create binding table buffers for each shader type
581   VkBuffer buffer = VK_NULL_HANDLE;
582   VmaAllocation allocation = VK_NULL_HANDLE;
583   RETURN_IF_FAILED_VK(CreateBuffer(sbt_size, sbt_buffer_usage_flags, VMA_MEMORY_USAGE_CPU_TO_GPU, 0, base_alignment,
584                                    &buffer, &allocation),
585                       "Failed to allocate shader binding table");
586   VkDeviceAddress buffer_address = BufferAddress(buffer);
587 
588   // Copy the pipeline's shader handles into a host buffer
589   std::vector<uint8_t> shader_handle_storage(handle_size * group_count);
590   procedures_.vkGetRayTracingShaderGroupHandlesKHR(device_, pipeline, 0, group_count, handle_size * group_count,
591                                                    shader_handle_storage.data());
592 
593   // Copy the shader handles from the host buffer to the binding tables
594   void *mapped = nullptr;
595   RETURN_IF_FAILED_VK(vmaMapMemory(allocator_, allocation, &mapped), "Failed to map shader binding table");
596   auto *data = static_cast<uint8_t *>(mapped);
597   std::memcpy(data + raygen_shader_offset, shader_handle_storage.data(), handle_size);
598   auto data_head = data + miss_shader_offset;
599   for (auto miss_shader_index : miss_shader_indices) {
600     std::memcpy(data_head, shader_handle_storage.data() + handle_size * (miss_shader_index + 1), handle_size);
601     data_head += handle_size_aligned;
602   }
603   data_head = data + hit_group_offset;
604   for (auto hit_group_index : hit_group_indices) {
605     std::memcpy(data_head, shader_handle_storage.data() + handle_size * (hit_group_index + miss_shader_count + 1),
606                 handle_size);
607     data_head += handle_size_aligned;
608   }
609   data_head = data + callable_shader_offset;
610   for (auto callable_shader_index : callable_shader_indices) {
611     std::memcpy(
612         data_head,
613         shader_handle_storage.data() + handle_size * (callable_shader_index + hit_group_count + miss_shader_count + 1),
614         handle_size);
615     data_head += handle_size_aligned;
616   }
617   vmaUnmapMemory(allocator_, allocation);
618 
619   *out_buffer = buffer;
620   *out_allocation = allocation;
621   *raygen_address = buffer_address + raygen_shader_offset;
622   *miss_address = buffer_address + miss_shader_offset;
623   *hit_address = buffer_address + hit_group_offset;
624   *callable_address = buffer_address + callable_shader_offset;
625 
626   return VK_SUCCESS;
627 }
