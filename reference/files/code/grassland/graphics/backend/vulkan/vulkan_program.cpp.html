vulkan_program.cpp

code/grassland/graphics/backend/vulkan/vulkan_program.cpp · 247 行 · 查看版本源码 ↗

职责与设计

每slot单binding的layout、图形/compute/RT管线及SBT。；本文件实现接口行为。

  • FinalizePipelineLayout检查maxBoundDescriptorSets、storage buffers和sampled images逐stage上限，其他descriptor限制依赖原生验证。
  • BindShader为追加stage而非同类替换；Shader module及entry字符串须活到Finalize。
  • 图形Finalize忽略CreatePipeline结果；Compute检查错误并抛异常；RT忽略pipeline结果但检查SBT创建。
  • 头文件VulkanProgram有一个raygen/miss/closest-hit构造声明，当前cpp没有该重载定义；使用VulkanRayTracingProgram对应实现。

29 void VulkanProgramBase::AddResourceBindingImpl(ResourceType type, int count) {
30   VkDescriptorSetLayoutBinding binding = {};
31   binding.binding = 0;
32   binding.descriptorType = ResourceTypeToVkDescriptorType(type);
33   binding.descriptorCount = count;
34   binding.stageFlags = VK_SHADER_STAGE_ALL;
35   VkDescriptorSetLayoutCreateInfo info{VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_CREATE_INFO};
36   info.bindingCount = 1;
37   info.pBindings = &binding;
38   VkDescriptorSetLayout layout = VK_NULL_HANDLE;
39   vulkan::ThrowIfFailed(vkCreateDescriptorSetLayout(core_->Handle(), &info, nullptr, &layout),
40                         "Failed to create Vulkan descriptor set layout");
41   descriptor_set_layouts_.push_back(layout);
42   descriptor_bindings_.push_back(binding);
43 }
44 
45 void VulkanProgramBase::FinalizePipelineLayout() {
46   // Bindings currently use VK_SHADER_STAGE_ALL and ordinary descriptor sets.
47   // Reject oversized scenes even when validation is disabled.
48   const auto limits = vulkan::GetPhysicalDeviceProperties(core_->PhysicalDevice()).limits;
49   uint64_t storage_buffers = 0, sampled_images = 0;
50   for (const auto &binding : descriptor_bindings_) {
51     if (binding.descriptorType == VK_DESCRIPTOR_TYPE_STORAGE_BUFFER)
52       storage_buffers += binding.descriptorCount;
53     if (binding.descriptorType == VK_DESCRIPTOR_TYPE_SAMPLED_IMAGE)
54       sampled_images += binding.descriptorCount;
55   }
56   if (descriptor_set_layouts_.size() > limits.maxBoundDescriptorSets ||
57       storage_buffers > limits.maxPerStageDescriptorStorageBuffers ||
58       sampled_images > limits.maxPerStageDescriptorSampledImages)
59     throw std::runtime_error("pipeline exceeds Vulkan descriptor limits: " + std::to_string(storage_buffers) +
60                              " storage buffers (limit " + std::to_string(limits.maxPerStageDescriptorStorageBuffers) +
61                              "), " + std::to_string(sampled_images) + " sampled images (limit " +
62                              std::to_string(limits.maxPerStageDescriptorSampledImages) + "), " +
63                              std::to_string(descriptor_set_layouts_.size()) + " sets (limit " +
64                              std::to_string(limits.maxBoundDescriptorSets) + ")");
65   std::vector<VkDescriptorSetLayout> descriptor_set_layouts;
66   descriptor_set_layouts.reserve(descriptor_set_layouts_.size());
67   descriptor_set_layouts = descriptor_set_layouts_;
68   VkPipelineLayoutCreateInfo layout_info{VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
69   layout_info.setLayoutCount = static_cast<uint32_t>(descriptor_set_layouts.size());
70   layout_info.pSetLayouts = descriptor_set_layouts.data();
71   vulkan::ThrowIfFailed(vkCreatePipelineLayout(core_->Handle(), &layout_info, nullptr, &pipeline_layout_),
72                         "Failed to create Vulkan pipeline layout");
73 }

145 void VulkanComputeProgram::Finalize() {
146   FinalizePipelineLayout();
147   VkComputePipelineCreateInfo pipeline_create_info = {};
148   pipeline_create_info.sType = VK_STRUCTURE_TYPE_COMPUTE_PIPELINE_CREATE_INFO;
149   pipeline_create_info.layout = pipeline_layout_;
150   pipeline_create_info.stage.sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO;
151   pipeline_create_info.stage.stage = VK_SHADER_STAGE_COMPUTE_BIT;
152   pipeline_create_info.stage.module = compute_shader_->ModuleHandle();
153   pipeline_create_info.stage.pName = compute_shader_->EntryPointRef().c_str();
154   pipeline_create_info.stage.pSpecializationInfo = nullptr;
155   const VkResult result =
156       vkCreateComputePipelines(core_->Handle(), VK_NULL_HANDLE, 1, &pipeline_create_info, nullptr, &pipeline_);
157   if (result != VK_SUCCESS)
158     throw std::runtime_error("failed to create Vulkan compute pipeline: " + std::to_string(result));
159 }

197 void VulkanRayTracingProgram::AddHitGroup(HitGroup hit_group) {
198   vulkan::HitGroup vk_hit_group;
199   auto vk_closest_hit_shader = dynamic_cast<VulkanShader *>(hit_group.closest_hit_shader);
200   vk_hit_group.closest_hit_shader = vk_closest_hit_shader;
201   assert(vk_hit_group.closest_hit_shader != nullptr);
202   auto vk_any_hit_shader = dynamic_cast<VulkanShader *>(hit_group.any_hit_shader);
203   if (vk_any_hit_shader) {
204     vk_hit_group.any_hit_shader = vk_any_hit_shader;
205   }
206 
207   auto vk_intersection_shader = dynamic_cast<VulkanShader *>(hit_group.intersection_shader);
208   if (vk_intersection_shader) {
209     vk_hit_group.intersection_shader = vk_intersection_shader;
210   }
211   vk_hit_group.procedure = hit_group.procedure;
212   hit_groups_.emplace_back(std::move(vk_hit_group));
213 }

216   auto vk_callable_shader = dynamic_cast<VulkanShader *>(callable_shader);
217   assert(vk_callable_shader != nullptr);
218   callable_shaders_.emplace_back(vk_callable_shader);
219 }
220 
221 void VulkanRayTracingProgram::Finalize(const std::vector<int32_t> &miss_shader_indices,
222                                        const std::vector<int32_t> &hit_group_indices,
223                                        const std::vector<int32_t> &callable_shader_indices) {
224   FinalizePipelineLayout();
225   core_->CreateRayTracingPipeline(pipeline_layout_, raygen_shader_, miss_shaders_, hit_groups_, callable_shaders_,
226                                   &pipeline_);
227   vulkan::ThrowIfFailed(
228       core_->CreateShaderBindingTable(pipeline_, miss_shaders_.size(), hit_groups_.size(), miss_shader_indices,
229                                       hit_group_indices, callable_shader_indices, &sbt_buffer_, &sbt_allocation_,
230                                       &raygen_address_, &miss_address_, &hit_address_, &callable_address_),
231       "Failed to create Vulkan shader binding table");
232   miss_shader_count_ = miss_shader_indices.size();
233   hit_group_count_ = hit_group_indices.size();
234   callable_shader_count_ = callable_shader_indices.size();
235 }
236 
237 void VulkanRayTracingProgram::Finalize() {
238   std::vector<int32_t> miss_shader_indices(miss_shaders_.size());
239   std::iota(miss_shader_indices.begin(), miss_shader_indices.end(), 0);
240   std::vector<int32_t> hit_group_indices(hit_groups_.size());
241   std::iota(hit_group_indices.begin(), hit_group_indices.end(), 0);
242   std::vector<int32_t> callable_shader_indices(callable_shaders_.size());
243   std::iota(callable_shader_indices.begin(), callable_shader_indices.end(), 0);
244   Finalize(miss_shader_indices, hit_group_indices, callable_shader_indices);
245 }
