职责与设计

设备feature、扩展、队列family与选择评分。；本文件实现接口行为。

  • GetDeviceLocalMemorySize返回最后一个DEVICE_LOCAL heap的大小，不是所有heap总和。
  • RayQuery同时检查4个扩展和query/AS/address feature；RayTracing只查rayTracingPipeline feature。
  • family选择取第一个符合条件者，失败返回uint32(-1)；不会优先专用transfer/compute队列。

112 uint64_t Evaluate(VkPhysicalDevice physical_device) {
113   uint64_t score = 0;
114   VkPhysicalDeviceProperties properties = GetPhysicalDeviceProperties(physical_device);
115   VkPhysicalDeviceFeatures features = GetPhysicalDeviceFeatures(physical_device);
116   VkPhysicalDeviceMemoryProperties memory_properties = GetPhysicalDeviceMemoryProperties(physical_device);
117   VkPhysicalDeviceRayTracingPipelinePropertiesKHR ray_tracing_properties =
118       GetPhysicalDeviceRayTracingPipelineProperties(physical_device);
119   VkPhysicalDeviceRayTracingPipelineFeaturesKHR ray_tracing_features =
120       GetPhysicalDeviceRayTracingPipelineFeatures(physical_device);
121   if (properties.deviceType == VK_PHYSICAL_DEVICE_TYPE_DISCRETE_GPU) {
122     score += 1000000000;
123   }
124   // Design a score system, key features: geometry and raytracing
125   if (features.geometryShader) {
126     score += 1000000;
127   }
128   if (ray_tracing_features.rayTracingPipeline) {
129     score += 100000000;
130   }
131   // Consider memory size, get from existed function
132   uint64_t device_local_memory_size = GetDeviceLocalMemorySize(physical_device);
133 
134   // Score for memory size
135   score += device_local_memory_size / 1000000;
136
