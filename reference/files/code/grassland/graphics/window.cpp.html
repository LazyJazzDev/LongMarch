职责与设计

GLFW 逻辑窗口、输入分发与 HDR 合成公共层；本文件实现接口行为。

  • 构造创建 GLFW_NO_API 窗口并连接各 EventManager；Mac 安装本地缩放手势监视器。
  • GetSize/GetCursorPosition 使用内容区左上角为原点的逻辑坐标；GetFramebufferSize 返回物理像素，最小化可为零。
  • HDR 合成使用窗口拥有的两张 RGBA16F 图像；白点缩放或线性 BT.709→BT.2020/PQ 变换都不修改调用者场景图像。
  • ImGui 兼容接口仍在底层存在；原生 demo 的 GUI 迁移没有删除这些接口。

149 Image *Window::PrepareHDRComposition(Core *core, Extent2D extent) {
150   if (!enable_hdr_)
151     return nullptr;
152   if (!hdr_presentation_) {
153     auto pending = std::make_unique<HDRPresentation>();
154     auto &p = *pending;
155     p.core = core;
156     const char *source = R"(
157 Texture2D<float4> source_image : register(t0, space0);
158 [[vk::image_format("rgba16f")]] RWTexture2D<float4> output_image : register(u0, space1);
159 cbuffer Settings : register(b0, space2) { float white_scale; float pq_output; float white_nits; float padding; };
160 float3 EncodePQ(float3 nits) {
161   float3 p = pow(saturate(nits / 10000.0), 2610.0 / 16384.0);
162   return pow((3424.0 / 4096.0 + (2413.0 / 128.0) * p) /
163              (1.0 + (2392.0 / 128.0) * p), 2523.0 / 32.0);
164 }
165 [numthreads(8,8,1)] void Main(uint3 id : SV_DispatchThreadID) {
166   uint width, height;
167   output_image.GetDimensions(width, height);
168   if (id.x >= width || id.y >= height) return;
169   float4 color = source_image.Load(int3(id.xy, 0));
170   float3 rgb = max(color.rgb, 0.0);
171   if (pq_output != 0.0) {
172     // Linear BT.709/D65 to BT.2020/D65, then absolute-luminance ST 2084.
173     rgb = mul(float3x3(0.627404, 0.329283, 0.043313,
174                        0.069097, 0.919540, 0.011362,
175                        0.016391, 0.088013, 0.895595), rgb);
176     rgb = EncodePQ(rgb * white_nits);
177   } else {
178     rgb = min(rgb * white_scale, 65504.0);
179   }
180   output_image[id.xy] = float4(rgb, color.a);
181 }
182 )";
183     if (core->CreateShader(source, "Main", "cs_6_0", &p.shader) ||
184         core->CreateComputeProgram(p.shader.get(), &p.program))
185       throw std::runtime_error("Cannot create HDR reference-white presentation shader");
186     p.program->AddResourceBinding(RESOURCE_TYPE_IMAGE, 1);
187     p.program->AddResourceBinding(RESOURCE_TYPE_WRITABLE_IMAGE, 1);
188     p.program->AddResourceBinding(RESOURCE_TYPE_UNIFORM_BUFFER, 1);
189     p.program->Finalize();
190     if (core->CreateBuffer(16, BUFFER_TYPE_STATIC, &p.settings))
191       throw std::runtime_error("Cannot create HDR presentation settings");
192     hdr_presentation_ = std::move(pending);
193   }
194   auto &p = *hdr_presentation_;
195   if (p.core != core)
196     throw std::invalid_argument("HDR presentation belongs to a different graphics core");
197   if (!p.composition || p.composition->Extent().width != extent.width ||
198       p.composition->Extent().height != extent.height) {
199     core->WaitGPU();
200     core->CreateImage(extent.width, extent.height, IMAGE_FORMAT_R16G16B16A16_SFLOAT, &p.composition);
201     core->CreateImage(extent.width, extent.height, IMAGE_FORMAT_R16G16B16A16_SFLOAT, &p.aligned);
202   }
203   return p.composition.get();
204 }

375 glm::ivec4 Window::GetMonitorWorkArea() const {
376 #if defined(__linux__) && defined(GLFW_PLATFORM)
377   if (glfwGetPlatform() == GLFW_PLATFORM_WAYLAND) {
378     // GLFW cannot locate an ordinary Wayland window in global coordinates.
379     // Use the primary output only as a sizing hint; let the compositor place it.
380     glm::ivec4 area{0, 0, GetWidth(), GetHeight()};
381     if (auto *monitor = glfwGetPrimaryMonitor())
382       glfwGetMonitorWorkarea(monitor, &area.x, &area.y, &area.z, &area.w);
383     return area;
384   }
385 #endif
386   const auto position = GetPosition();
387   const auto size = GetSize();
388   int count = 0;
389   auto **monitors = glfwGetMonitors(&count);
390   auto *monitor = glfwGetPrimaryMonitor();
391   int64_t best_overlap = 0;
392   for (int i = 0; i < count; ++i) {
393     glm::ivec4 area;
394     glfwGetMonitorWorkarea(monitors[i], &area.x, &area.y, &area.z, &area.w);
395     const auto overlap =
396         glm::max(glm::ivec2{0}, glm::min(position + size, glm::ivec2{area.x + area.z, area.y + area.w}) -
397                                     glm::max(position, glm::ivec2{area.x, area.y}));
398     const int64_t pixels = int64_t(overlap.x) * overlap.y;
399     if (pixels > best_overlap) {
400       best_overlap = pixels;
401       monitor = monitors[i];
402     }
403   }
404   if (!monitor)
405     return {position.x, position.y, size.x, size.y};
406   glm::ivec4 area;
407   glfwGetMonitorWorkarea(monitor, &area.x, &area.y, &area.z, &area.w);
408   return area;
409 }

423 void Window::NotifyResize() {
424   if (!window_)
425     return;
426   const auto size = GetSize();
427   if (size == resize_size_)
428     return;
429   // Wayland programmatic resizing may omit the logical-size callback.
430   // Deduplicate both native callbacks and Resize() by logical size; pixel-only
431   // scaling remains exclusively a FramebufferResizeEvent.
432   resize_size_ = size;
433   resize_event_.InvokeCallbacks(size.x, size.y);
434 }

436 void Window::Resize(int new_width, int new_height) {
437   glfwSetWindowSize(window_, new_width, new_height);
438   // GLFW's Wayland backend can omit the logical-size callback here.
439   // NotifyResize deduplicates against any native callback already delivered.
440   if (glfwGetPlatform() == GLFW_PLATFORM_WAYLAND)
441     NotifyResize();
442 }
