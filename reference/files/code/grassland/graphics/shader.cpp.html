职责与设计

着色器资源与 Slang 编译器接入；本文件实现接口行为。

  • CompileShader 的字符串重载写入临时 VFS，再委托统一编译路径。
  • 每线程缓存独立 Slang global session；使用自定义只读 VFS 解析 includes，保持重复 include 语义由 Slang 处理。
  • 默认目标 DXIL；后端可通过 args 指定 SPIR-V 等目标。矩阵列主序、优化 O3、register/space 零偏移属于公共绑定约定。
  • 多数编译阶段失败返回 data 为空的 CompiledShaderBlob，创建全局编译器失败或不支持的 profile 则抛异常。

137       *out = new SourceBlob(std::move(data));
138       return SLANG_OK;
139     } catch (...) {
140       return SLANG_FAIL;
141     }
142   }
143 
144  private:
145   std::atomic<uint32_t> references_{1};
146   const VirtualFileSystem &vfs_;
147 };

149 void Diagnostics(ISlangBlob *blob) {
150   if (blob && blob->getBufferSize())
151     LogInfo("{}", std::string(static_cast<const char *>(blob->getBufferPointer()), blob->getBufferSize()));
152 }

154 SlangStage ShaderStage(const std::string &profile) {
155   const auto stage = profile.substr(0, profile.find('_'));
156   if (stage == "vs")
157     return SLANG_STAGE_VERTEX;
158   if (stage == "ps")
159     return SLANG_STAGE_FRAGMENT;
160   if (stage == "cs")
161     return SLANG_STAGE_COMPUTE;
162   if (stage == "gs")
163     return SLANG_STAGE_GEOMETRY;
164   if (stage == "hs")
165     return SLANG_STAGE_HULL;
166   if (stage == "ds")
167     return SLANG_STAGE_DOMAIN;
168   if (stage == "ms")
169     return SLANG_STAGE_MESH;
170   if (stage == "as")
171     return SLANG_STAGE_AMPLIFICATION;
172   if (stage == "lib")
173     return SLANG_STAGE_NONE;  // [shader(...)] identifies ray tracing entry points.
174   throw std::invalid_argument("Unsupported shader profile: " + profile);
175 }

178 CompiledShaderBlob CompileShader(const std::string &source_code,
179                                  const std::string &entry_point,
180                                  const std::string &target,
181                                  const std::vector<std::string> &args) {
182   VirtualFileSystem vfs;
183   vfs.WriteFile("shader.slang", source_code);
184   return CompileShader(vfs, "shader.slang", entry_point, target, args);

187 CompiledShaderBlob CompileShader(const VirtualFileSystem &vfs,
188                                  const std::string &source_file,
189                                  const std::string &entry_point,
190                                  const std::string &target,
191                                  const std::vector<std::string> &args) {
192   CompiledShaderBlob result;
193   result.entry_point = entry_point;
194   // Global sessions cache Slang's standard library. Separate sessions per thread
195   // avoid concurrent access to the compiler's mutable state.
196   thread_local Slang::ComPtr<slang::IGlobalSession> global;
197   if (!global) {
198     if (SLANG_FAILED(slang::createGlobalSession(global.writeRef())))
199       throw std::runtime_error("Cannot initialize Slang compiler");
200     UseBundledSlangPlugins(global);
201   }
202   const auto stage = ShaderStage(target);
203   const auto suffix = target.find('_');
204   std::vector<std::string> options{"-profile", "sm" + target.substr(suffix), "-matrix-layout-column-major", "-O3",
205                                    "-I.",      "-fvk-use-entrypoint-name",   "-emit-spirv-directly"};
206   if (std::find(args.begin(), args.end(), "-target") == args.end())
207     options.insert(options.begin(), {"-target", "dxil"});
208   // Preserve the public register/space binding contract on SPIR-V targets.
209   for (const char *kind : {"b", "t", "u", "s"}) {
210     options.push_back(std::string("-fvk-") + kind + "-shift");
211     options.insert(options.end(), {"0", "all"});
212   }
213 #ifndef NDEBUG
214   options.insert(options.end(), {"-g", "-DDEBUG_SHADER"});
215 #endif
216   options.insert(options.end(), args.begin(), args.end());
217   std::vector<const char *> arguments;
218   for (const auto &option : options)
219     arguments.push_back(option.c_str());
220   slang::SessionDesc desc{};
221   Slang::ComPtr<ISlangUnknown> allocation;
222   if (SLANG_FAILED(global->parseCommandLineArguments(static_cast<int>(arguments.size()), arguments.data(), &desc,
223                                                      allocation.writeRef()))) {
224     LogError("Invalid Slang compiler options for {}", source_file);
225     return result;
226   }
227   Slang::ComPtr<ISlangFileSystem> files;
228   files.attach(new ShaderFileSystem(vfs));
229   desc.fileSystem = files;
230   Slang::ComPtr<slang::ISession> session;
231   if (SLANG_FAILED(global->createSession(desc, session.writeRef())))
232     return result;
233   std::vector<uint8_t> source;
234   if (vfs.ReadFile(source_file, source)) {
235     LogError("Missing shader source: {}", source_file);
236     return result;
237   }
238   source.push_back(0);
239   Slang::ComPtr<ISlangBlob> diagnostics;
240   auto module = session->loadModuleFromSourceString(
241       "longmarch_shader", source_file.c_str(), reinterpret_cast<const char *>(source.data()), diagnostics.writeRef());
242   Diagnostics(diagnostics);
243   if (!module)
244     return result;
245   Slang::ComPtr<slang::IEntryPoint> entry;
246   SlangResult status;
247   if (stage == SLANG_STAGE_NONE)
248     status = module->findEntryPointByName(entry_point.c_str(), entry.writeRef());
249   else
250     status = module->findAndCheckEntryPoint(entry_point.c_str(), stage, entry.writeRef(), diagnostics.writeRef());
251   Diagnostics(diagnostics);
252   if (SLANG_FAILED(status))
253     return result;
254   slang::IComponentType *components[] = {module, entry};
255   Slang::ComPtr<slang::IComponentType> program, linked;
256   status = session->createCompositeComponentType(components, 2, program.writeRef(), diagnostics.writeRef());
257   Diagnostics(diagnostics);
258   if (SLANG_FAILED(status))
259     return result;
260   status = program->link(linked.writeRef(), diagnostics.writeRef());
261   Diagnostics(diagnostics);
262   if (SLANG_FAILED(status))
263     return result;
264   Slang::ComPtr<ISlangBlob> code;
265   status = linked->getEntryPointCode(0, 0, code.writeRef(), diagnostics.writeRef());
266   Diagnostics(diagnostics);
267   if (SLANG_FAILED(status) || !code)
268     return result;
269   result.data.resize(code->getBufferSize());
270   std::memcpy(result.data.data(), code->getBufferPointer(), result.data.size());
271   return result;
272 }
