solver_scene_update.cpp

code/snowberg/solver/solver_scene_update.cpp · 499 行 · 查看版本源码 ↗

职责与设计

CUDA 顶点块下降、SDF 接触与塑性推进

  • 单场景更新重力 (0,−9.8,0)，40 次按颜色迭代；碰撞壳层厚度硬编码 0.018。
  • 每粒子累加惯性、拉伸、弯曲与接触的力/Hessian，然后直接 H.inverse()*f；无阻尼线搜索或数值失败恢复。
  • UpdateBatch 20 次迭代，碰撞摩擦硬编码 5，忽略 rigid_object.friction。
  • 批量求解将 tid 减去场景偏移后，用局部 tid 读取全局合并 particle_indices；第二个及后续场景可能读到前面场景的粒子 ID。此实现需要修正与 CUDA 验证后才能依赖。
  • 部分核在 scene.stream_，其余使用默认流，排序依赖默认流同步语义；不能作为无同步多流执行保证。
扩展语法核对记录

CUDA三尖括号kernel launch不属于C++语法。已逐项核对Update与UpdateBatch中的InitializeSolver、SolveVBDParticlePosition、CopyParticleIndices、速度及塑性kernel调用，所有kernel自身另有对象说明；调用原文保留。

23                                  const Matrix3<float> &X,
24                                  Vector3<float> &jacobian,
25                                  Matrix3<float> &hessian,
26                                  int self_index) {
27   Matrix<float, 3, 2> F;
28   F.col(0) = X.col(1) - X.col(0);
29   F.col(1) = X.col(2) - X.col(0);
30   auto a = 1.0 + stretching.mu / stretching.lambda;
31   Matrix2<float> Dm_inv = stretching.Dm.inverse();
32   F = F * Dm_inv;
33   Vector2<float> dXdxi;
34   if (self_index == 0) {
35     dXdxi = (-Dm_inv.row(0) - Dm_inv.row(1)).transpose();
36   } else if (self_index == 1) {
37     dXdxi = Dm_inv.row(0).transpose();
38   } else {
39     dXdxi = Dm_inv.row(1).transpose();
40   }
41 
42   Vector3<float> Fz = F.col(0).cross(F.col(1));
43   Vector3<float> n = Fz.normalized();
44   float J = Fz.norm();
45   jacobian = stretching.mu * (F.col(0) * dXdxi[0] + F.col(1) * dXdxi[1]);
46   hessian = stretching.mu * dXdxi.squaredNorm() * Matrix3<float>::Identity();
47   Vector3<float> compressed_jacobian = (F.col(1).cross(n) * dXdxi[0] + n.cross(F.col(0)) * dXdxi[1]);
48   jacobian += stretching.lambda * (J - a) * compressed_jacobian;
49   hessian += stretching.lambda * compressed_jacobian * compressed_jacobian.transpose();

154     Vector3<float> delta_x = H.inverse() * f;
155     scene_ref.x[pidx] += delta_x;
156   }

159 __global__ void SolveVBDParticlePositionBatched(const int *scene_offsets,
160                                                 int num_scenes,
161                                                 SceneRef *scene_refs,
162                                                 const int *particle_indices,
163                                                 int total_particles,
164                                                 float dt) {
165   int tid = threadIdx.x + blockIdx.x * blockDim.x;
166   int &sid = num_scenes;
167   sid--;
168   while (sid >= 0 && scene_offsets[sid] > tid)
169     sid--;
170   const SceneRef &scene_ref = scene_refs[sid];
171   if (tid < total_particles) {
172     tid -= scene_offsets[sid];
173     int pid = particle_indices[tid];
174     int pidx = scene_ref.ParticleIndex(pid);
175     Vector3<float> x = scene_ref.x[pidx];
176     Vector3<float> x_prev = scene_ref.x_prev[pidx];
177     float m = scene_ref.m[pidx];
178     Vector3<float> f = -(m / (dt * dt)) * (x - (x_prev + scene_ref.v[pidx] * dt));
179     Matrix3<float> H = (m / (dt * dt)) * Matrix3<float>::Identity();
180 
181     for (int i = 0; i < scene_ref.stretching_directory.count[pid]; i++) {
182       int stretching_id = scene_ref.stretching_directory.positions[scene_ref.stretching_directory.first[pid] + i] / 3;
183       auto stretching = scene_ref.stretchings[stretching_id];
184 
185       uint32_t u = scene_ref.stretching_indices[stretching_id * 3 + 0];
186       uint32_t v = scene_ref.stretching_indices[stretching_id * 3 + 1];
187       uint32_t w = scene_ref.stretching_indices[stretching_id * 3 + 2];
188       int self_index = pid == u ? 0 : (pid == v ? 1 : 2);
189 
190       Matrix3<float> X;
191       X << scene_ref.x[u], scene_ref.x[v], scene_ref.x[w];
192       Vector3<float> jacobian;
193       Matrix3<float> hessian;
194       // jacobian = neo_hookean.Jacobian(X).block(0, 3 * self_index, 1, 3).transpose() * stretching.area;
195       // hessian = neo_hookean.Hessian(X).m[0].block(3 * self_index, 3 * self_index, 3, 3) * stretching.area;
196       StretchingEnergy(stretching, X, jacobian, hessian, self_index);
197 
198       float k_damping = stretching.damping / dt;
199       f -= jacobian + hessian * (x - x_prev) * k_damping;
200       H += hessian * (1.0 + k_damping);
201     }
202 
203     for (int i = 0; i < scene_ref.bending_directory.count[pid]; i++) {
204       int bending_id = scene_ref.bending_directory.positions[scene_ref.bending_directory.first[pid] + i] / 4;
205       auto bending = scene_ref.bendings[bending_id];
206 
207       grassland::DihedralAngle<float> dihedral_angle;
208       uint32_t u = scene_ref.bending_indices[bending_id * 4 + 0];
209       uint32_t v = scene_ref.bending_indices[bending_id * 4 + 1];
210       uint32_t w = scene_ref.bending_indices[bending_id * 4 + 2];
211       uint32_t z = scene_ref.bending_indices[bending_id * 4 + 3];
212       int self_index = pid == u ? 0 : (pid == v ? 1 : (pid == w ? 2 : 3));
213 
214       Matrix<float, 3, 4> X;
215       X << scene_ref.x[u], scene_ref.x[v], scene_ref.x[w], scene_ref.x[z];
216       float theta = dihedral_angle(X).value() - bending.theta_rest;
217       Vector3<float> jacobian;  // = dihedral_angle.Jacobian(X).block(0, 3 * self_index, 1, 3).transpose();
218       Matrix3<float> hessian;   // = Matrix3<float>::Zero();
219       // hessian = dihedral_angle.SubHessian(X, self_index);
220       DihedralAngleSubHessianJacobian(X, jacobian, hessian, self_index);
221       hessian = (2.0 * theta * hessian + 2.0 * jacobian * jacobian.transpose()).derived();
222       jacobian *= 2.0 * theta;
223 
224       // if (fabs(theta) > 0.0f) {
225       //   printf("theta: %f jacobian: %f %f %f, current_angle: %f, theta_rest: %f\n", theta, jacobian[0], jacobian[1],
226       //   jacobian[2], dihedral_angle(X).value(), bending.theta_rest);
227       // }
228 
229       float k_damping = bending.damping / dt;
230       f -= jacobian * bending.stiffness + hessian * (x - x_prev) * k_damping * bending.stiffness;
231       H += hessian * (1.0 + k_damping) * bending.stiffness;
232     }
233 
234     const float k_friction = 5.0f;
235     const Vector3<float> rel_vel = (x - x_prev) / dt;
236 
237     constexpr float K_DAMPING = 1e-6;
238     for (int i = 0; i < scene_ref.num_rigid_object; i++) {
239       float sdf;
240       Vector3<float> jacobian;
241       Matrix3<float> hessian;
242       RigidObjectRef rigid_object = scene_ref.rigid_objects[i];
243       rigid_object.mesh_sdf.SDF(x, rigid_object.state.R, rigid_object.state.t, &sdf, &jacobian, &hessian);
244       Vector3<float> r = x - sdf * jacobian - rigid_object.state.t;
245       sdf -= 0.018f;
246       if (sdf < 0.0) {
247         float k_stiffness = rigid_object.stiffness * m;
248         float force_mag = -2.0 * k_stiffness * sdf;
249         Matrix3<float> partial_H =
250             2.0 * k_stiffness * jacobian * jacobian.transpose() + 2.0 * k_stiffness * sdf * hessian;
251         f -= 2.0 * k_stiffness * sdf * jacobian + partial_H * (x - x_prev) * K_DAMPING;
252         H += partial_H + partial_H * K_DAMPING;
253 
254         Vector3<float> velocity_component = rel_vel - rigid_object.state.v - rigid_object.state.omega.cross(r);
255         velocity_component = velocity_component - jacobian * jacobian.transpose() * velocity_component;
256         float max_friction_force = force_mag * k_friction;
257         float vel_comp_norm = velocity_component.norm();
258         if (vel_comp_norm > max_friction_force * dt / m) {
259           f -= velocity_component / vel_comp_norm * max_friction_force;
260         } else if (vel_comp_norm > 1e-9) {
261           f -= velocity_component / dt * m;
262           // f -= velocity_component / vel_comp_norm * max_friction_force;
263         }
264       }
265     }
266 
267     Vector3<float> delta_x = H.inverse() * f;
268     scene_ref.x[pidx] += delta_x;
269   }
270 }

272 __global__ void UpdateStretchingPlasticity(SceneRef scene_ref) {
273   int sid = threadIdx.x + blockIdx.x * blockDim.x;
274   if (sid < scene_ref.num_stretching) {
275     ElementStretching stretching = scene_ref.stretchings[sid];
276     if (stretching.sigma_lb > 0.0 || stretching.sigma_ub > 0.0) {
277       uint32_t u = scene_ref.stretching_indices[sid * 3 + 0];
278       uint32_t v = scene_ref.stretching_indices[sid * 3 + 1];
279       uint32_t w = scene_ref.stretching_indices[sid * 3 + 2];
280       Matrix3<float> X;
281       X << scene_ref.x[u], scene_ref.x[v], scene_ref.x[w];
282       Matrix<float, 3, 2> F;
283       F << X.col(1) - X.col(0), X.col(2) - X.col(0);
284       Matrix<float, 3, 2> Fe = F * stretching.Dm.inverse();
285       Matrix<float, 3, 2> U;
286       Matrix<float, 2, 2> S;
287       Matrix<float, 2, 2> Vt;
288       SVD(Fe, U, S, Vt);
289       if (stretching.sigma_lb > 0.0) {
290         // if (S(0, 0) < stretching.theta_lb || S(1, 1) < stretching.theta_lb) {
291         //   // printf("Stretching clamped (lower)\n");
292         // }
293         S(0, 0) = max(S(0, 0), stretching.sigma_lb);
294         S(1, 1) = max(S(1, 1), stretching.sigma_lb);
295       }
296       if (stretching.sigma_ub > 0.0) {
297         // if (S(0, 0) > stretching.theta_ub || S(1, 1) > stretching.theta_ub) {
298         //   printf("Stretching clamped (upper)\n");
299         //   printf("%f %f\n", S(0, 0), S(1, 1));
300         // }
301         S(0, 0) = min(S(0, 0), stretching.sigma_ub);
302         S(1, 1) = min(S(1, 1), stretching.sigma_ub);
303       }
304       S(0, 0) = 1.0 / S(0, 0);
305       S(1, 1) = 1.0 / S(1, 1);
306       // printf("BP1");
307       stretching.Dm = Vt.transpose() * S * U.transpose() * F;
308       // printf("BP2");
309       scene_ref.stretchings[sid] = stretching;
310     }
311   }
312 }

314 __global__ void UpdateBendingPlasticity(SceneRef scene_ref) {
315   int bid = threadIdx.x + blockIdx.x * blockDim.x;
316   if (bid < scene_ref.num_bending) {
317     ElementBending bending = scene_ref.bendings[bid];
318     constexpr float pi = 3.14159265358979323846;
319     if (bending.elastic_limit < pi) {
320       uint32_t a = scene_ref.bending_indices[bid * 4 + 0];
321       uint32_t b = scene_ref.bending_indices[bid * 4 + 1];
322       uint32_t c = scene_ref.bending_indices[bid * 4 + 2];
323       uint32_t d = scene_ref.bending_indices[bid * 4 + 3];
324       Matrix<float, 3, 4> X;
325       X << scene_ref.x[a], scene_ref.x[b], scene_ref.x[c], scene_ref.x[d];
326       DihedralAngle<float> dihedral_angle;
327       float theta = dihedral_angle(X).value();
328       float theta_rest = bending.theta_rest;
329 
330       // limit difference between theta and theta_rest to bending.bending_bound
331       float diff = theta_rest - theta;
332       if (diff > pi) {
333         diff -= 2.0 * pi;
334       }
335       if (diff < -pi) {
336         diff += 2.0 * pi;
337       }
338       if (diff < 0.0) {
339         diff = max(diff, -bending.elastic_limit);
340       }
341       if (diff > 0.0) {
342         diff = min(diff, bending.elastic_limit);
343       }
344 
345       theta_rest = theta + diff;
346       if (theta_rest < -pi) {
347         theta_rest += 2.0 * pi;
348       }
349       if (theta_rest > pi) {
350         theta_rest -= 2.0 * pi;
351       }
352       bending.theta_rest = theta_rest;
353       scene_ref.bendings[bid] = bending;
354     }
355   }
356 }

358 __global__ void UpdateVelocity(SceneRef scene_ref, float dt) {
359   int pid = threadIdx.x + blockIdx.x * blockDim.x;
360   if (pid < scene_ref.num_particle) {
361     scene_ref.v[pid] = (scene_ref.x[pid] - scene_ref.x_prev[pid]) / dt;
362   }

365 void SceneDevice::Update(SceneDevice &scene, float dt) {
366   DeviceClock clk;
367   SceneRef scene_ref = scene;
368   scene.x_prev_ = scene.x_;
369   InitializeSolver<<<DEFAULT_DISPATCH_SIZE(scene_ref.num_particle), 0, scene.stream_>>>(
370       scene_ref, Vector3<float>{0.0, -9.8, 0.0}, dt);
371   clk.Record("Initialize Solver");
372 
373   const int num_vbd_iterations_ = 40;
374   for (int iter = 0; iter < num_vbd_iterations_; iter++) {
375     for (int c = 0; c < scene.particle_directory_host_.first.size(); c++) {
376       SolveVBDParticlePosition<<<DEFAULT_DISPATCH_SIZE(scene.particle_directory_host_.count[c])>>>(
377           scene_ref, scene.particle_directory_.positions.data().get() + scene.particle_directory_host_.first[c],
378           scene.particle_directory_host_.count[c], dt);
379     }
380   }
381   clk.Record("Solve VBD");
382 
383   UpdateVelocity<<<DEFAULT_DISPATCH_SIZE(scene_ref.num_particle), 0, scene.stream_>>>(scene_ref, dt);
384   clk.Record("Update Velocity");
385 
386   UpdateStretchingPlasticity<<<DEFAULT_DISPATCH_SIZE(scene_ref.num_stretching)>>>(scene_ref);
387   clk.Record("Update stretching plasticity");
388 
389   UpdateBendingPlasticity<<<DEFAULT_DISPATCH_SIZE(scene_ref.num_bending)>>>(scene_ref);
390   clk.Record("Update bending plasticity");
391 
392   clk.Finish();
393   printf("particles: %d, stretchings: %d, bendings: %d\n", scene_ref.num_particle, scene_ref.num_stretching,
394          scene_ref.num_bending);
395 }
