职责与设计

网格到求解粒子与单元的装配；本文件实现接口行为。

  • 质量平均分配到所有顶点，不按三角形面积分配。
  • 普通网格按每三索引建立拉伸，仅对有双向邻接的内部边建立弯曲。
  • GridCloth 采用列主索引 i*n_row+j；每格四个拉伸单元和多组弯曲单元，系数分别缩放。

145 void ObjectPack::PushStretching(int u,
146                                 int v,
147                                 int w,
148                                 float mu,
149                                 float lambda,
150                                 float damping,
151                                 float sigma_lb,
152                                 float sigma_ub) {
153   ElementStretching stretching;
154   stretching.mu = mu;
155   stretching.lambda = lambda;
156 
157   const Vector3<float> x0 = x[u];
158   const Vector3<float> x1 = x[v];
159   const Vector3<float> x2 = x[w];
160 
161   Matrix2<float> s;
162   s(0, 0) = (x1 - x0).norm();
163   s(1, 0) = 0.0;
164   s(0, 1) = (x2 - x0).dot(x1 - x0) / s(0, 0);
165   s(1, 1) = std::sqrt((x2 - x0).squaredNorm() - s(0, 1) * s(0, 1));
166 
167   stretching.Dm = s;
168   stretching.area = 0.5f * stretching.Dm.determinant();
169   stretching.sigma_lb = sigma_lb;
170   stretching.sigma_ub = sigma_ub;
171   stretching.damping = damping;
172 
173   stretchings.push_back(stretching);
174   stretching_indices.push_back(u);
175   stretching_indices.push_back(v);
176   stretching_indices.push_back(w);
177 }

179 void ObjectPack::PushBending(int a, int b, int c, int d, float stiffness, float damping, float elastic_limit) {
180   ElementBending bending;
181   DihedralAngle<float> dihedral_angle;
182   Matrix<float, 3, 4> V;
183   V << x[a], x[b], x[c], x[d];
184 
185   const Vector3<float> e = V.col(1) - V.col(2);
186 
187   const float h = (e.cross(V.col(0) - V.col(1)).norm() + e.cross(V.col(2) - V.col(3)).norm()) / e.norm();
188 
189   bending.stiffness = stiffness * e.norm() * 6.0f / h;
190   bending.damping = damping;
191   bending.theta_rest = dihedral_angle(V).value();
192   bending.elastic_limit = elastic_limit;
193 
194   bendings.push_back(bending);
195   bending_indices.push_back(a);
196   bending_indices.push_back(b);
197   bending_indices.push_back(c);
198   bending_indices.push_back(d);
199 }
