solver_scene.cpp

code/snowberg/solver/solver_scene.cpp · 297 行 · 查看版本源码 ↗

职责与设计

主机对象合并、GPU 数据上传与图着色；本文件实现接口行为。

  • CPU Scene 只追加数据，不支持删除或同步设备后新增拓扑。
  • SceneDevice 深复制数组和 SDF、预建单元邻接及颜色桶；设备查询以 Thrust 临时数组回读。
  • ID 查询返回 −1 未命中，但上层 getter/setter 未验证结果；空数组的 BinarySearch 也没有安全处理。

直接依赖

  • snowberg/solver/solver_scene.h

42     object_pack_view.stretching_ids.push_back(stretching_id);
43     stretching_ids_.push_back(stretching_id);
44     stretchings_.push_back(object_pack.stretchings[i]);
45     stretching_indices_.push_back(object_pack_view.particle_ids[object_pack.stretching_indices[i * 3]]);
46     stretching_indices_.push_back(object_pack_view.particle_ids[object_pack.stretching_indices[i * 3 + 1]]);
47     stretching_indices_.push_back(object_pack_view.particle_ids[object_pack.stretching_indices[i * 3 + 2]]);
48   }
49 
50   int pack_num_bending = object_pack.bendings.size();
51   int first_bending_id = next_bending_id_;
52   next_bending_id_ += pack_num_bending;
53   object_pack_view.bending_ids.reserve(pack_num_bending);
54   for (int i = 0; i < pack_num_bending; i++) {
55     int bending_id = first_bending_id + i;
56     object_pack_view.bending_ids.push_back(bending_id);
57     bending_ids_.push_back(bending_id);
58     bendings_.push_back(object_pack.bendings[i]);
59     bending_indices_.push_back(object_pack_view.particle_ids[object_pack.bending_indices[i * 4]]);
60     bending_indices_.push_back(object_pack_view.particle_ids[object_pack.bending_indices[i * 4 + 1]]);
61     bending_indices_.push_back(object_pack_view.particle_ids[object_pack.bending_indices[i * 4 + 2]]);
62     bending_indices_.push_back(object_pack_view.particle_ids[object_pack.bending_indices[i * 4 + 3]]);
63   }
64 
65   return object_pack_view;
66 }

69   int rigid_object_id = next_rigid_object_id_++;
70   rigid_object_meshes_.push_back(rigid_object.mesh_sdf);
71   rigid_object_ids_.push_back(rigid_object_id);

73   rigid_object_ref.mesh_sdf = rigid_object_meshes_.back();
74   rigid_objects_.push_back(rigid_object_ref);
75   return rigid_object_id;

218 std::vector<Vector3<float>> SceneDevice::GetPositions(const std::vector<int> &particle_ids) const {
219   thrust::device_vector<int> query_particle_ids(particle_ids);
220   thrust::device_vector<Vector3<float>> result_positions(particle_ids.size());
221   thrust::transform(query_particle_ids.begin(), query_particle_ids.end(), result_positions.begin(),
222                     SearchAndCopyOp{x_.data().get(), particle_ids_.data().get(), static_cast<int>(x_.size())});

224   thrust::copy(result_positions.begin(), result_positions.end(), result_positions_host.begin());
