code/snowberg/gui/surface/application.cpp · 384 行 · 查看版本源码 ↗

职责与设计

surface 窗口生命周期与双帧实例化渲染；本文件实现接口行为。

  • Run 初始化→事件/更新/渲染循环→可选截图→GPU 等待与清理。
  • 帧缓冲最大边≤2048 时 3×超采样，≤4096 时 2×，更大时 1×；绘制坐标仍是原始帧缓冲像素。
  • 实例记录后拼接 uint32 数据，extra.y 在上传时增加实例记录区字节数。
  • RenderFrameTarget 稳定排序连续模型批次；深度负责图层，同深度跨模型提交顺序不保证。

77 void Application::Run(int max_frames) {
78   OnInit();
79   int frames = 0;
80   while (!window_->ShouldClose() && (max_frames <= 0 || frames < max_frames)) {
81     grassland::graphics::Window::PollEvents();
82     OnUpdate();
83     OnRender();
84     frames++;
85   }
86   if (!screenshot_path_.empty()) {
87     SaveScreenshot();
88   }
89   OnClose();
90 }

137 void Application::OnInit() {
138   const auto &shader_code = surface_shader_;
139   core_->CreateShader(shader_code, "VSMain", "vs_6_0", &vertex_shader_);
140   core_->CreateShader(shader_code, "PSMain", "ps_6_0", &pixel_shader_);
141   core_->CreateProgram({graphics::IMAGE_FORMAT_R8G8B8A8_UNORM}, graphics::IMAGE_FORMAT_D32_SFLOAT, &program_);
142   program_->BindShader(vertex_shader_.get(), graphics::SHADER_TYPE_VERTEX);
143   program_->BindShader(pixel_shader_.get(), graphics::SHADER_TYPE_PIXEL);
144   program_->AddInputBinding(sizeof(Vertex));
145   program_->AddInputBinding(sizeof(uint32_t), true);
146   program_->AddInputAttribute(0, graphics::INPUT_TYPE_FLOAT2, offsetof(Vertex, position));
147   program_->AddInputAttribute(0, graphics::INPUT_TYPE_FLOAT4, offsetof(Vertex, color));
148   program_->AddInputAttribute(1, graphics::INPUT_TYPE_UINT, 0);
149   program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
150   program_->AddResourceBinding(graphics::RESOURCE_TYPE_UNIFORM_BUFFER, 1);
151   program_->SetCullMode(graphics::CULL_MODE_NONE);
152   program_->Finalize();
153 
154   const auto resolve_code = GetShaderCode("shaders/surface_resolve.slang");
155   core_->CreateShader(resolve_code, "VSMain", "vs_6_0", &resolve_vertex_shader_);
156   core_->CreateShader(resolve_code, "PSMain", "ps_6_0", &resolve_pixel_shader_);
157   core_->CreateProgram({graphics::IMAGE_FORMAT_R8G8B8A8_UNORM}, graphics::IMAGE_FORMAT_UNDEFINED, &resolve_program_);
158   resolve_program_->BindShader(resolve_vertex_shader_.get(), graphics::SHADER_TYPE_VERTEX);
159   resolve_program_->BindShader(resolve_pixel_shader_.get(), graphics::SHADER_TYPE_PIXEL);
160   resolve_program_->AddResourceBinding(graphics::RESOURCE_TYPE_UNIFORM_BUFFER, 1);
161   resolve_program_->AddResourceBinding(graphics::RESOURCE_TYPE_IMAGE, 1);
162   resolve_program_->AddResourceBinding(graphics::RESOURCE_TYPE_IMAGE, 1);
163   resolve_program_->SetCullMode(graphics::CULL_MODE_NONE);
164   resolve_program_->Finalize();
165 
166   core_->CreateBuffer(sizeof(glm::mat4), graphics::BUFFER_TYPE_DYNAMIC, &global_uniform_buffer_);
167   core_->CreateBuffer(sizeof(ResolveParams), graphics::BUFFER_TYPE_DYNAMIC, &resolve_uniform_buffer_);
168   const uint32_t first_instance_index = 0;
169   core_->CreateBuffer(sizeof(uint32_t), graphics::BUFFER_TYPE_DYNAMIC, &instance_index_buffer_);
170   instance_index_buffer_->UploadData(&first_instance_index, sizeof(uint32_t));
171   core_->CreateBuffer(sizeof(InstanceInfo), graphics::BUFFER_TYPE_DYNAMIC, &main_frame_.instance_buffer);
172   core_->CreateBuffer(sizeof(InstanceInfo), graphics::BUFFER_TYPE_DYNAMIC, &second_frame_.instance_buffer);
173 
174   BuildScreenFrameObjects();
175 
176   title_time_ = grassland::GetTimeSeconds();
177   CustomOnInit();
178 }

180 void Application::OnUpdate() {
181   const auto size = window_->GetFramebufferSize();
182   const int width = size.x, height = size.y;
183   if (width > 0 && height > 0 && glm::ivec2{width, height} != framebuffer_size_) {
184     BuildScreenFrameObjects();
185     OnFramebufferResize();
186   }
187 
188   for (auto *target : {&main_frame_, &second_frame_}) {
189     target->instances.clear();
190     target->data.clear();
191     target->data_instances.clear();
192   }
193   second_frame_alpha_ = 0.0f;
194 
195   CustomOnUpdate();
196   UpdateTitle();
197 }

199 void Application::OnRender() {
200   const auto size = window_->GetFramebufferSize();
201   const int width = size.x, height = size.y;
202   if (width <= 0 || height <= 0) {
203     // Minimized windows have no framebuffer to present.
204     return;
205   }
206 
207   // Instance indices are fed through a per-instance vertex stream so that the
208   // first instance offset works identically on every backend.
209   const size_t max_instances = std::max(main_frame_.instances.size(), second_frame_.instances.size());
210   if (max_instances * sizeof(uint32_t) > instance_index_buffer_->Size()) {
211     std::vector<uint32_t> indices(std::max<size_t>(max_instances, instance_index_buffer_->Size() / 2));
212     for (size_t i = 0; i < indices.size(); i++) {
213       indices[i] = static_cast<uint32_t>(i);
214     }
215     instance_index_buffer_->Resize(indices.size() * sizeof(uint32_t));
216     instance_index_buffer_->UploadData(indices.data(), indices.size() * sizeof(uint32_t));
217   }
218 
219   std::unique_ptr<graphics::CommandContext> context;
220   core_->CreateCommandContext(&context);
221 
222   const bool use_second_frame = second_frame_alpha_ > 0.0f;
223   if (use_second_frame) {
224     RenderFrameTarget(context.get(), second_frame_);
225   }
226   RenderFrameTarget(context.get(), main_frame_);
227 
228   ResolveParams params{use_second_frame ? second_frame_alpha_ : 0.0f, static_cast<uint32_t>(supersample_scale_)};
229   resolve_uniform_buffer_->UploadData(&params, sizeof(params));
230 
231   auto *second_image = use_second_frame ? second_frame_.color_image.get() : main_frame_.color_image.get();
232   context->CmdBeginRendering({present_image_.get()}, nullptr);
233   context->CmdBindProgram(resolve_program_.get());
234   context->CmdBindResources(0, {resolve_uniform_buffer_.get()});
235   context->CmdBindResources(1, {main_frame_.color_image.get()});
236   context->CmdBindResources(2, {second_image});
237   context->CmdSetViewport({0.0f, 0.0f, float(framebuffer_size_.x), float(framebuffer_size_.y), 0.0f, 1.0f});
238   context->CmdSetScissor({{0, 0}, {uint32_t(framebuffer_size_.x), uint32_t(framebuffer_size_.y)}});
239   context->CmdSetPrimitiveTopology(graphics::PRIMITIVE_TOPOLOGY_TRIANGLE_LIST);
240   context->CmdDraw(3, 1, 0, 0);
241   context->CmdEndRendering();
242 
243   last_presented_image_ = ComposeUI(context.get(), present_image_.get());
244   context->CmdPresent(window_.get(), last_presented_image_);
245   core_->SubmitCommandContext(context.get());
246 }

248 void Application::RenderFrameTarget(graphics::CommandContext *context, FrameTarget &target) {
249   auto &instances = target.instances;
250 
251   // Group instances of the same model into a single draw call without sorting
252   // or moving the large InstanceInfo records: sort the short list of runs of
253   // consecutive instances instead. The stable sort keeps submission order within
254   // a model; the depth buffer resolves layering.
255   struct Batch {
256     DeviceModel *model;
257     size_t head, count;
258   };
259 
260   std::vector<Batch> batches;
261   for (size_t head = 0, tail = 0; head < instances.size(); head = tail) {
262     while (tail < instances.size() && instances[head].first == instances[tail].first)
263       ++tail;
264     batches.push_back({instances[head].first, head, tail - head});
265   }
266   std::stable_sort(batches.begin(), batches.end(),
267                    [](const auto &a, const auto &b) { return std::less<DeviceModel *>{}(a.model, b.model); });
268 
269   if (!instances.empty()) {
270     // One upload: instance records, then the data words they reference.
271     const size_t records_size = instances.size() * sizeof(InstanceInfo);
272     const size_t data_size = target.data.size() * sizeof(uint32_t);
273     auto &upload = target.upload;
274     upload.resize(records_size + data_size);
275     auto *records = reinterpret_cast<InstanceInfo *>(upload.data());
276     for (size_t i = 0; i < instances.size(); i++) {
277       records[i] = instances[i].second;
278     }
279     for (size_t index : target.data_instances) {
280       records[index].extra.y += uint32_t(records_size);
281     }
282     if (data_size)
283       std::memcpy(upload.data() + records_size, target.data.data(), data_size);
284     if (upload.size() > target.instance_buffer->Size()) {
285       target.instance_buffer->Resize(upload.size());
286     }
287     target.instance_buffer->UploadData(upload.data(), upload.size());
288   }
289 
290   const auto extent = target.color_image->Extent();
291   context->CmdClearImage(target.color_image.get(), {{clear_color_.r, clear_color_.g, clear_color_.b, clear_color_.a}});
292   context->CmdClearImage(depth_image_.get(), {{1.0f}});
293   context->CmdBeginRendering({target.color_image.get()}, depth_image_.get());
294   context->CmdBindProgram(program_.get());
295   context->CmdBindResources(0, {target.instance_buffer.get()});
296   context->CmdBindResources(1, {global_uniform_buffer_.get()});
297   context->CmdBindVertexBuffers(1, {instance_index_buffer_.get()}, {0});
298   context->CmdSetViewport({0.0f, 0.0f, float(extent.width), float(extent.height), 0.0f, 1.0f});
299   context->CmdSetScissor({{0, 0}, extent});
300   context->CmdSetPrimitiveTopology(graphics::PRIMITIVE_TOPOLOGY_TRIANGLE_LIST);
301 
302   for (const auto &batch : batches) {
303     auto *device_model = batch.model;
304     if (!device_model->IndexCount())
305       continue;
306     context->CmdBindVertexBuffers(0, {device_model->VertexBuffer()}, {0});
307     context->CmdBindIndexBuffer(device_model->IndexBuffer(), 0);
308     context->CmdDrawIndexed(device_model->IndexCount(), uint32_t(batch.count), 0, 0, uint32_t(batch.head));
309   }
310   context->CmdEndRendering();
311 }

314   core_->WaitGPU();

333 void Application::BuildScreenFrameObjects() {
334   const auto size = window_->GetFramebufferSize();
335   const int width = size.x, height = size.y;
336   framebuffer_size_ = {std::max(width, 1), std::max(height, 1)};
337   supersample_scale_ = ChooseSupersampleScale(framebuffer_size_);
338   const auto sample_size = framebuffer_size_ * supersample_scale_;
339 
340   core_->WaitGPU();
341   main_frame_.color_image.reset();
342   second_frame_.color_image.reset();
343   depth_image_.reset();
344   present_image_.reset();
345   core_->CreateImage(sample_size.x, sample_size.y, graphics::IMAGE_FORMAT_R8G8B8A8_UNORM, &main_frame_.color_image);
346   core_->CreateImage(sample_size.x, sample_size.y, graphics::IMAGE_FORMAT_R8G8B8A8_UNORM, &second_frame_.color_image);
347   core_->CreateImage(sample_size.x, sample_size.y, graphics::IMAGE_FORMAT_D32_SFLOAT, &depth_image_);
348   core_->CreateImage(framebuffer_size_.x, framebuffer_size_.y, graphics::IMAGE_FORMAT_R8G8B8A8_UNORM, &present_image_);
349 
350   // Maps framebuffer pixel coordinates (y pointing down) to normalized device coordinates.
351   glm::mat4 view{1.0f};
352   view[0][0] = 2.0f / float(framebuffer_size_.x);
353   view[1][1] = -2.0f / float(framebuffer_size_.y);
354   view[3][0] = -1.0f;
355   view[3][1] = 1.0f;
356   global_uniform_buffer_->UploadData(&view, sizeof(view));
357 }

359 void Application::UpdateTitle() {
360   // FPSCounter averages over the frames of the last second; the title shows it
361   // four times a second.
362   fps_counter_.TickFrame();
363   const double now = grassland::GetTimeSeconds();
364   if (now - title_time_ >= 0.25) {
365     window_->SetTitle(
366         fmt::format("[{}] {} FPS: {:.1f}", graphics::BackendAPIString(core_->API()), name_, fps_counter_.GetFPS()));
367     title_time_ = now;
368   }
369 }

371 void Application::SaveScreenshot() {
372   core_->WaitGPU();
373   const auto extent = last_presented_image_->Extent();
374   std::vector<uint8_t> pixels(size_t(extent.width) * extent.height * 4);
375   last_presented_image_->DownloadData(pixels.data());
376   if (stbi_write_png(screenshot_path_.c_str(), int(extent.width), int(extent.height), 4, pixels.data(),
377                      int(extent.width) * 4)) {
378     LogInfo("Screenshot saved to {}", screenshot_path_);
379   } else {
380     LogError("Failed to save screenshot to {}", screenshot_path_);
381   }
382 }
