code/snowberg/gui/gui.cpp · 471 行 · 查看版本源码 ↗

职责与设计

声明式控件收集、指针状态机和玻璃绘制；本文件实现接口行为。

  • 输入在 BeginFrame 读取，Activate 在按下时捕获 ID、同控件内释放时触发点击；捕获只保留字符串。
  • 面板宽度由调用者指定，高度根据子控件累计；普通边距 18、纵向间隔 7、行间横向间距 8。
  • 圆角网格按物理宽/高/半径缓存；无淘汰策略，动态尺寸可能持续增大缓存。
  • EndFrame 借用返回的 scene 或自身 output_，应用负责提交命令及资源存活；尺寸/格式改变会 WaitGPU 后重建。

284   return {{cursor_x_ - frame.x, cursor_y_ - frame.y}, hovered, pointer_pressed_ && hovered, clicked};
285 }

287 void Context::DrawRect(float x, float y, float width, float height, Color color, float radius) {
288   if (width <= 0 || height <= 0)
289     return;
290   const int pixel_width = std::max(1, int(std::round(width * scale_x_)));
291   const int pixel_height = std::max(1, int(std::round(height * scale_y_)));
292   const int pixel_radius = std::max(0, int(std::round(radius * std::min(scale_x_, scale_y_))));
293   const auto key = std::make_tuple(pixel_width, pixel_height, pixel_radius);
294   auto found = rect_models_.find(key);
295   if (found == rect_models_.end()) {
296     std::unique_ptr<draw::Model> model;
297     painter_->CreateModel(&model);
298     std::vector<draw::Vertex> vertices;
299     std::vector<uint32_t> indices;
300     constexpr int segments = 8;
301     const float corner_radius = std::min({float(pixel_radius), pixel_width * 0.5f, pixel_height * 0.5f});
302     const float rx = corner_radius / pixel_width, ry = corner_radius / pixel_height;
303     vertices.push_back({{0.5f, 0.5f}, {0.5f, 0.5f}, {1, 1, 1, 1}});
304     for (int corner = 0; corner < 4; ++corner) {
305       const float cx = corner == 0 || corner == 3 ? rx : 1.0f - rx;
306       const float cy = corner < 2 ? ry : 1.0f - ry;
307       const float start = 3.14159265f + corner * 1.570796325f;
308       for (int i = 0; i <= segments; ++i) {
309         const float angle = start + i * 1.570796325f / segments;
310         const glm::vec2 p{cx + std::cos(angle) * rx, cy + std::sin(angle) * ry};
311         vertices.push_back({p, p, {1, 1, 1, 1}});
312         if (vertices.size() > 2)
313           indices.insert(indices.end(),
314                          {0, static_cast<uint32_t>(vertices.size() - 2), static_cast<uint32_t>(vertices.size() - 1)});
315       }
316     }
317     indices.insert(indices.end(), {0, static_cast<uint32_t>(vertices.size() - 1), 1});
318     model->SetModelData(vertices, indices);
319     found = rect_models_.emplace(key, std::move(model)).first;
320   }
321   glm::mat4 transform = glm::translate(glm::mat4(1), glm::vec3(x * scale_x_, y * scale_y_, 0));
322   transform = glm::scale(transform, glm::vec3(width * scale_x_, height * scale_y_, 1));
323   painter_->CmdDrawInstance(found->second.get(), draw::PixelCoordToNDC(target_width_, target_height_) * transform,
324                             Vec(color));
325 }

332 void Context::PaintItem(const Item &item) {
333   switch (item.kind) {
334     case Item::Kind::kText:
335       DrawText(item.x, item.y + 16, item.label, theme_.muted, 15);
336       break;
337     case Item::Kind::kHeading:
338       DrawText(item.x, item.y + 21, item.label, theme_.text, 18);
339       break;
340     case Item::Kind::kSeparator:
341       DrawRect(item.x, item.y, item.width, 1, {1, 1, 1, 0.20f});
342       break;
343     case Item::Kind::kButton:
344     case Item::Kind::kChoice:
345       DrawRect(item.x, item.y, item.width, item.height,
346                item.active    ? Alpha(theme_.accent, 0.9f)
347                : item.hovered ? Alpha(theme_.control, 1.0f)
348                               : theme_.control);
349       DrawText(item.x + 12, item.y + 24, item.label, item.active ? Color{1, 1, 1, 1} : theme_.text, 15);
350       break;
351     case Item::Kind::kCheckbox:
352       DrawRect(item.x, item.y + 3, 24, 24, item.checked ? theme_.accent : theme_.control);
353       if (item.checked)
354         DrawText(item.x + 5, item.y + 22, "✓", {1, 1, 1, 1}, 16);
355       DrawText(item.x + 34, item.y + 22, item.label, theme_.text, 15);
356       break;
357     case Item::Kind::kSlider:
358       DrawText(item.x, item.y + 17, item.label, theme_.text, 14);
359       DrawRect(item.x, item.y + 35, item.width, 7, theme_.control, 4);
360       DrawRect(item.x, item.y + 35, std::max(7.0f, item.width * item.value), 7, theme_.accent, 4);
361       DrawRect(item.x + item.width * item.value - 6, item.y + 30, 15, 16,
362                theme_.dark ? Color{0.91f, 0.94f, 1.0f, 1} : Color{1, 1, 1, 1}, 8);
363       break;
364     case Item::Kind::kPlot:
365       DrawText(item.x, item.y + 17, item.label, theme_.text, 14);
366       DrawRect(item.x, item.y + 23, item.width, 58, Alpha(theme_.control, 0.45f));
367       for (size_t i = 0; i < item.samples.size(); ++i) {
368         const float w = item.width / std::max<size_t>(1, item.samples.size());
369         DrawRect(item.x + i * w, item.y + 80 - item.samples[i] * 54, std::max(1.0f, w - 1),
370                  std::max(2.0f, item.samples[i] * 54), theme_.accent);
371       }
372       break;
373     case Item::Kind::kCanvas: {
374       Canvas canvas(this, item.x, item.y, item.width, item.height);
375       item.paint(canvas);
376       break;
377     }
378   }
379 }

393 void Canvas::ParticleField(int count, float time, Color color) {
394   for (int i = 0; i < count; ++i) {
395     const float phase = i * 2.399963f;
396     const float speed = 0.13f + float(i % 7) * 0.018f;
397     const float x = std::fmod((i * 0.618034f + time * speed), 1.0f) * width_;
398     const float y = std::fmod((i * 0.414214f + time * speed * 0.7f), 1.0f) * height_;
399     const float radius = 1.5f + 1.5f * (0.5f + 0.5f * std::sin(phase + time));
400     Circle(x, y, radius, color);
401   }

404 grassland::graphics::Image *Context::EndFrame(grassland::graphics::CommandContext *commands,
405                                               grassland::graphics::Image *scene) {
406   if (current_)
407     throw std::logic_error("GUI panel not closed");
408   const auto logical = logical_override_.x > 0 && logical_override_.y > 0 ? logical_override_ : window_->GetSize();
409   const auto physical = scene->Extent();
410   if (logical.x <= 0 || logical.y <= 0 || physical.width == 0 || physical.height == 0)
411     return scene;
412   target_width_ = static_cast<int>(physical.width);
413   target_height_ = static_cast<int>(physical.height);
414   scale_x_ = static_cast<float>(physical.width) / logical.x;
415   scale_y_ = static_cast<float>(physical.height) / logical.y;
416   auto *target = scene;
417   if (theme_.backdrop_blur && !panels_.empty()) {
418     if (!output_ || output_->Extent().width != physical.width || output_->Extent().height != physical.height ||
419         output_->Format() != scene->Format()) {
420       core_->WaitGPU();
421       horizontal_.reset();
422       vertical_.reset();
423       output_.reset();
424       core_->CreateImage(physical.width, physical.height, scene->Format(), &horizontal_);
425       core_->CreateImage(physical.width, physical.height, scene->Format(), &vertical_);
426       core_->CreateImage(physical.width, physical.height, scene->Format(), &output_);
427     }
428     const BlurSettings horizontal{{2, 0}, {target_width_, target_height_}};
429     const BlurSettings vertical{{0, 2}, {target_width_, target_height_}};
430     blur_settings_horizontal_->UploadData(&horizontal, sizeof(horizontal));
431     blur_settings_vertical_->UploadData(&vertical, sizeof(vertical));
432     commands->CmdBindComputeProgram(blur_program_.get());
433     commands->CmdBindResources(0, {scene}, grassland::graphics::BIND_POINT_COMPUTE);
434     commands->CmdBindResources(1, {horizontal_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
435     commands->CmdBindResources(2, {blur_settings_horizontal_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
436     commands->CmdDispatch((physical.width + 7) / 8, (physical.height + 7) / 8, 1);
437     commands->CmdBindResources(0, {horizontal_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
438     commands->CmdBindResources(1, {vertical_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
439     commands->CmdBindResources(2, {blur_settings_vertical_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
440     commands->CmdDispatch((physical.width + 7) / 8, (physical.height + 7) / 8, 1);
441     GlassSettings glass{};
442     glass.extent = {target_width_, target_height_};
443     glass.count = std::min<int>(panels_.size(), 16);
444     for (int i = 0; i < glass.count; ++i) {
445       const auto &panel = panels_[i];
446       glass.rects[i] = {panel.x * scale_x_, panel.y * scale_y_, (panel.x + panel.width) * scale_x_,
447                         (panel.y + panel.height) * scale_y_};
448     }
449     glass_settings_->UploadData(&glass, sizeof(glass));
450     commands->CmdBindComputeProgram(composite_program_.get());
451     commands->CmdBindResources(0, {scene}, grassland::graphics::BIND_POINT_COMPUTE);
452     commands->CmdBindResources(1, {vertical_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
453     commands->CmdBindResources(2, {output_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
454     commands->CmdBindResources(3, {glass_settings_.get()}, grassland::graphics::BIND_POINT_COMPUTE);
455     commands->CmdDispatch((physical.width + 7) / 8, (physical.height + 7) / 8, 1);
456     target = output_.get();
457   }
458   painter_->CmdSetDrawRegion(0, 0, target_width_, target_height_);
459   for (const auto &panel : panels_) {
460     DrawRect(panel.x, panel.y + 5, panel.width, panel.height, {0, 0, 0, theme_.dark ? 0.32f : 0.16f}, 19);
461     DrawRect(panel.x, panel.y, panel.width, panel.height, {1, 1, 1, theme_.dark ? 0.18f : 0.38f}, 19);
462     DrawRect(panel.x + 1, panel.y + 1, panel.width - 2, panel.height - 2, theme_.panel, 18);
463     if (!panel.title.empty())
464       DrawText(panel.x + 18, panel.y + 35, panel.title, theme_.text, 21);
465     for (const auto &item : panel.items)
466       PaintItem(item);
467   }
468   painter_->Render(commands, target);
469   return target;
470 }
