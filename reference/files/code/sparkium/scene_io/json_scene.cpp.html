职责与设计

JSON v1 场景装配、二进制网格/毛发读取与材质图源码生成。；本文件实现接口行为。

  • 根必须包含 format="sparkium-scene"、version=1，以及 renderer、film、camera、materials、geometries、entities。renderer 支持 auto/rasterization/ray_tracing/ray_query/rt_fallback；samples 默认 32、max_bounces 默认 32，均须大于零。
  • film 宽高为正整数；persistence 默认 1、clamping 默认 100，view_transform 为 normalized/standard/filmic。相机 fov_degrees 默认 60 且在 (0,180)；eye/target/up 必须形成有效视图。薄透镜在 aperture_radius>0 时可自动选择；focus_distance 默认视点距离，叶片旋转按原值传入。
  • 材质支持 lambertian、specular、light、principled、shader_graph；Principled 加载 normal/base_color/metallic/specular/roughness/anisotropic/anisotropic_rotation/emission 八类贴图并支持 normal_reverse_y。图材质预加载所有 image_texture 节点，包括未连接节点。
  • 几何支持 sphere、OBJ mesh、binary_mesh、inline_mesh、hair；内嵌网格校验三角索引范围，二进制网格只校验长度与标志，不校验索引范围。可显式重算法线/切线；毛发经 GeometryHair 转为三角网格。
  • 实体支持 mesh 和 point_light。mesh 通过几何/材质 ID 查找；transform 的 matrix 为列主序且优先于 TRS，TRS 使用 T*Rx*Ry*Rz*S。look_at 会覆盖 transform；实体 look_at 未重复执行相机的退化校验。
  • 材质图输出包括 base_color、metallic、specular、roughness、anisotropic、anisotropic_rotation、sheen、clearcoat、clearcoat_roughness、ior、transmission、transmission_roughness、emission、normal、opacity、shadow_opacity、thin_walled、subsurface、subsurface_scale、subsurface_radius、subsurface_method。未列出的输出被忽略。
  • 图支持常量、纹理坐标、图像、mapping、noise/voronoi/gradient/wave/sky/brick、顶点与对象信息、geometry_info/light_path、mix/math、color_ramp/rgb_curves、HSV/gamma/bright_contrast、layer_weight、normal_map、bump、combine/separate/passthrough/invert_y。noise 为值噪声，sky 为近似梯度，bump 不使用高度；light_path 的 glossy/diffuse/singular 都按反射标记判断。
  • color_ramp 按输入顺序线性混合，不排序；rgb_curves 是每通道均匀采样的分段线性插值。math 的 DIVIDE 使用 max(abs(b),1e-8)，不保留负除数符号；未知 math 操作按 ADD，未知节点类型抛异常。

199 BinaryHairData BinaryHair(const std::filesystem::path &path) {
200   std::ifstream stream(path, std::ios::binary | std::ios::ate);
201   if (!stream)
202     throw std::runtime_error("cannot open binary hair: " + path.string());
203   const auto size = stream.tellg();
204   stream.seekg(0);
205   std::array<char, 8> magic{};
206   uint32_t strand_count = 0, point_count = 0;
207   stream.read(magic.data(), magic.size());
208   stream.read(reinterpret_cast<char *>(&strand_count), sizeof(strand_count));
209   stream.read(reinterpret_cast<char *>(&point_count), sizeof(point_count));
210   const uint64_t expected = 16ull + uint64_t(strand_count + 1) * sizeof(uint32_t) +
211                             uint64_t(point_count) * sizeof(Vector3<float>) + uint64_t(point_count) * sizeof(float);
212   if (!stream || std::memcmp(magic.data(), "SPKHAIR1", 8) != 0 || uint64_t(size) != expected)
213     throw std::runtime_error("invalid Sparkium binary hair: " + path.string());
214   BinaryHairData result;
215   result.offsets.resize(strand_count + 1);
216   result.points.resize(point_count);
217   result.radii.resize(point_count);
218   stream.read(reinterpret_cast<char *>(result.offsets.data()), result.offsets.size() * sizeof(uint32_t));
219   stream.read(reinterpret_cast<char *>(result.points.data()), result.points.size() * sizeof(Vector3<float>));
220   stream.read(reinterpret_cast<char *>(result.radii.data()), result.radii.size() * sizeof(float));
221   if (!stream || result.offsets.empty() || result.offsets.front() != 0 || result.offsets.back() != point_count)
222     throw std::runtime_error("invalid Sparkium hair offsets: " + path.string());
223   return result;
224 }

238 float GraphHash(float3 p) {
239   p = frac(p * 0.3183099f + 0.1f); p *= 17.0f;
240   return frac(p.x * p.y * p.z * (p.x + p.y + p.z));
241 }
242 float GraphNoise(float3 p) {
243   float3 i = floor(p), f = frac(p); f = f * f * (3.0f - 2.0f * f);
244   float n000=GraphHash(i), n100=GraphHash(i+float3(1,0,0));
245   float n010=GraphHash(i+float3(0,1,0)), n110=GraphHash(i+float3(1,1,0));
246   float n001=GraphHash(i+float3(0,0,1)), n101=GraphHash(i+float3(1,0,1));
247   float n011=GraphHash(i+float3(0,1,1)), n111=GraphHash(i+float3(1,1,1));
248   return lerp(lerp(lerp(n000,n100,f.x),lerp(n010,n110,f.x),f.y),
249               lerp(lerp(n001,n101,f.x),lerp(n011,n111,f.x),f.y),f.z);
250 }
251 float3 GraphSrgbToLinear(float3 color) {
252   float3 lower=color/12.92f;
253   float3 upper=pow((color+0.055f)/1.055f,2.4f);
254   return lerp(lower,upper,step(0.04045f,color));
255 }
256 float GraphVoronoi(float3 p) {
257   float3 cell=floor(p), local=frac(p); float distance=1e9f;
258   for(int z=-1;z<=1;++z) for(int y=-1;y<=1;++y) for(int x=-1;x<=1;++x) {
259     float3 o=float3(x,y,z), sample_point=o+GraphHash(cell+o+float3(0,0,0))*float3(0.73f,0.91f,0.57f);
260     distance=min(distance,length(sample_point-local));
261   }
262   return distance;
263 }
264 float3 GraphRotateXYZ(float3 p, float3 r) {
265   float3 s=sin(r), c=cos(r);
266   p=float3(p.x, c.x*p.y-s.x*p.z, s.x*p.y+c.x*p.z);
267   p=float3(c.y*p.x+s.y*p.z, p.y, -s.y*p.x+c.y*p.z);
268   return float3(c.z*p.x-s.z*p.y, s.z*p.x+c.z*p.y, p.z);
269 }
270 float3 GraphRgbToHsv(float3 c) {
271   float4 K=float4(0.0f,-1.0f/3.0f,2.0f/3.0f,-1.0f);
272   float4 p=lerp(float4(c.bg,K.wz),float4(c.gb,K.xy),step(c.b,c.g));
273   float4 q=lerp(float4(p.xyw,c.r),float4(c.r,p.yzx),step(p.x,c.r));
274   float d=q.x-min(q.w,q.y), e=1.0e-10f;
275   return float3(abs(q.z+(q.w-q.y)/(6.0f*d+e)),d/(q.x+e),q.x);
276 }
277 float3 GraphHsvToRgb(float3 c) {
278   float3 p=abs(frac(c.xxx+float3(0.0f,2.0f/3.0f,1.0f/3.0f))*6.0f-3.0f);
279   return c.z*lerp(float3(1,1,1),saturate(p-1.0f),c.y);
280 }

597       std::string hsv = "GraphRgbToHsv((" + color + ").xyz)";
598       std::string adjusted = "GraphHsvToRgb(float3(frac((" + hsv + ").x+(" + hue + ").x-0.5f),max(0.0f,(" + hsv +
599                              ").y*(" + saturation + ").x),(" + hsv + ").z*(" + value + ").x))";
600       expression = "float4(lerp((" + color + ").xyz," + adjusted + ",saturate((" + factor + ").x)),(" + color + ").w)";
601     } else if (type == "gamma") {
602       auto color = Input(node, "color", one), gamma = Input(node, "gamma", one);
603       expression = "float4(pow(max((" + color + ").xyz,0.0f),max((" + gamma + ").xxx,1e-6f)),(" + color + ").w)";
604     } else if (type == "bright_contrast") {
605       auto color = Input(node, "color", one), brightness = Input(node, "brightness", zero),
606            contrast = Input(node, "contrast", zero);
607       expression = "float4((" + color + ").xyz*(1.0f+(" + contrast + ").x)+(" + brightness + ").xxx,(" + color + ").w)";
608     } else if (type == "layer_weight") {
609       auto blend = Input(node, "blend", "float4(0.5,0.5,0.5,0.5)");
610       if (output == "facing")
611         expression =
612             "(1.0f-abs(dot(normalize(hit_record.normal),normalize("
613             "view_direction)))).xxxx";
614       else
615         expression =
616             "pow(1.0f-saturate(abs(dot(normalize(hit_record.normal),"
617             "normalize(view_direction)))),max(0.01f,(" +
618             blend + ").x*5.0f)).xxxx";
619     } else if (type == "normal_map") {
620       auto c = Input(node, "color", "float4(0.5,0.5,1,1)");
621       auto strength = Input(node, "strength", one);
622       std::string tangent_normal =
623           "normalize(lerp(float3(0,0,1),(" + c + ").xyz*2.0f-1.0f,saturate((" + strength + ").x)))";
624       std::string mapped_normal = "normalize(mul(" + tangent_normal +
625                                   ",float3x3(hit_record.tangent,cross(hit_record.normal,hit_record."
626                                   "tangent)*hit_record.signal,hit_record.normal)))";
627       expression = "float4(abs(hit_record.signal)>0.5f?" + mapped_normal + ":hit_record.normal,0)";
628     } else if (type == "bump") {
629       expression = Input(node, "normal", "float4(hit_record.normal,0)");
630     } else if (type == "combine") {
631       expression = "float4((" + Input(node, "x", zero) + ").x,(" + Input(node, "y", zero) + ").x,(" +
632                    Input(node, "z", zero) + ").x,1.0f)";
633     } else if (type == "separate") {
634       auto c = Input(node, "color", zero);
635       int component = (output == "green" || output == "y") ? 1 : (output == "blue" || output == "z") ? 2 : 0;
636       const char *swizzle = component == 1 ? ".yyyy" : component == 2 ? ".zzzz" : ".xxxx";
637       expression = "(" + c + ")" + swizzle;
638     } else if (type == "passthrough") {
639       expression = Input(node, "value", zero);
640     } else if (type == "invert_y") {
641       auto vector = Input(node, "vector", zero);
642       expression = "float4((" + vector + ").x,1.0f-(" + vector + ").y,(" + vector + ").z,(" + vector + ").w)";
643     } else if (type == "brick_texture") {
644       auto p = Input(node, "vector", "float4(hit_record.object_position,0)");
645       auto scale = Input(node, "scale", "float4(5,5,5,5)");
646       auto mortar_size = Input(node, "mortar_size", "float4(0.02,0.02,0.02,0.02)");
647       auto c1 = Input(node, "color1", zero), c2 = Input(node, "color2", one), mortar = Input(node, "mortar", zero);
648       std::string q = "frac((" + p + ").xy*(" + scale + ").x)";
649       std::string edge =
650           "(min(min(" + q + ".x," + q + ".y),min(1.0f-" + q + ".x,1.0f-" + q + ".y))<(" + mortar_size + ").x)";
651       expression = "(" + edge + "?" + mortar + ":lerp(" + c1 + "," + c2 + ",fmod(floor((" + p + ").x*(" + scale +
652                    ").x)+floor((" + p + ").y*(" + scale + ").x),2.0f)))";
653     } else {
654       throw std::runtime_error("unsupported shader node type: " + type);
655     }
656     lines_ << "  float4 " << var << "=" << expression << ";\n";
657     active_.erase(key);
658     cache_[key] = var;
659     return var;
660   }
661 
662   const Value &graph_;
663   const Value &nodes_;
664   const std::map<std::string, int> &texture_slots_;
665   std::ostringstream lines_;
666   std::map<std::string, std::string> cache_;
667   std::set<std::string> active_;
668   int next_variable_ = 0;
669 };

672 std::unique_ptr<JsonScene> JsonScene::Load(Core *core, const std::filesystem::path &input_path, std::string *error) {
673   try {
674     auto path = std::filesystem::absolute(input_path).lexically_normal();
675     std::ifstream stream(path);
676     if (!stream)
677       throw std::runtime_error("cannot open scene file: " + path.string());
678     rapidjson::IStreamWrapper wrapper(stream);
679     rapidjson::Document document;
680     document.ParseStream<rapidjson::kParseCommentsFlag | rapidjson::kParseTrailingCommasFlag>(wrapper);
681     if (document.HasParseError()) {
682       std::ostringstream message;
683       message << "JSON parse error at byte " << document.GetErrorOffset() << ": "
684               << rapidjson::GetParseError_En(document.GetParseError());
685       throw std::runtime_error(message.str());
686     }
687     if (!document.IsObject())
688       throw std::runtime_error("scene root must be an object");
689     if (!document.HasMember("format") || ReadString(document["format"]) != "sparkium-scene")
690       throw std::runtime_error("unsupported or missing format (expected 'sparkium-scene')");
691     if (!document.HasMember("version") || ReadInt(document["version"]) != 1)
692       throw std::runtime_error("unsupported scene version (expected 1)");
693 
694     auto result = std::unique_ptr<JsonScene>(new JsonScene);
695     result->core_ = core;
696     result->path_ = path;
697     result->name_ = document.HasMember("name") ? ReadString(document["name"]) : path.stem().string();
698     result->scene_ = std::make_unique<Scene>(core);
699 
700     const auto &renderer = RequireObject(Member(document, "renderer"));
701     result->scene_->settings.raytracing.samples_per_dispatch =
702         renderer.HasMember("samples_per_dispatch") ? ReadInt(renderer["samples_per_dispatch"]) : 32;
703     result->scene_->settings.raytracing.max_bounces =
704         renderer.HasMember("max_bounces") ? ReadInt(renderer["max_bounces"]) : 32;
705     if (result->scene_->settings.raytracing.samples_per_dispatch <= 0 ||
706         result->scene_->settings.raytracing.max_bounces <= 0)
707       throw std::runtime_error("samples_per_dispatch and max_bounces must be positive");
708     result->scene_->settings.raytracing.alpha_shadow = BoolMember(renderer, "alpha_shadow", false);
709     result->scene_->settings.raster.ambient_light = Vec3Member(renderer, "ambient_light", {0.1f, 0.1f, 0.1f});
710     result->scene_->settings.raytracing.background_color =
711         Vec3Member(renderer, "background_color", result->scene_->settings.raster.ambient_light);
712     std::string pipeline = renderer.HasMember("pipeline") ? ReadString(renderer["pipeline"]) : "auto";
713     if (pipeline == "rasterization")
714       result->render_pipeline_ = RENDER_PIPELINE_RASTERIZATION;
715     else if (pipeline == "ray_tracing")
716       result->render_pipeline_ = RENDER_PIPELINE_RAY_TRACING;
717     else if (pipeline == "ray_query")
718       result->render_pipeline_ = RENDER_PIPELINE_RAY_QUERY;
719     else if (pipeline == "rt_fallback")
720       result->render_pipeline_ = RENDER_PIPELINE_RT_FALLBACK;
721     else if (pipeline == "auto")
722       result->render_pipeline_ = RENDER_PIPELINE_AUTO;
723     else
724       throw std::runtime_error("renderer.pipeline must be auto, rasterization, ray_tracing, rt_fallback, or ray_query");
725 
726     const auto &film = RequireObject(Member(document, "film"));
727     int width = ReadInt(Member(film, "width"));
728     int height = ReadInt(Member(film, "height"));
729     if (width <= 0 || height <= 0)
730       throw std::runtime_error("film dimensions must be positive");
731     result->film_ = std::make_unique<Film>(core, width, height);
732     result->film_->info.persistence = FloatMember(film, "persistence", 1.0f);
733     result->film_->info.clamping = FloatMember(film, "clamping", 100.0f);
734     result->film_->info.max_exposure = FloatMember(film, "max_exposure", result->film_->info.max_exposure);
735     std::string view_transform = film.HasMember("view_transform") ? ReadString(film["view_transform"]) : "normalized";
736     if (view_transform == "normalized")
737       result->film_->info.view_transform = 0;
738     else if (view_transform == "standard")
739       result->film_->info.view_transform = 1;
740     else if (view_transform == "filmic")
741       result->film_->info.view_transform = 2;
742     else
743       throw std::runtime_error("film.view_transform must be normalized, standard, or filmic");
744     result->film_->info.exposure = FloatMember(film, "exposure", 0.0f);
745     result->film_->info.gamma = FloatMember(film, "gamma", 1.0f);
746     result->film_->info.contrast = FloatMember(film, "contrast", 1.0f);
747 
748     const auto &camera = RequireObject(Member(document, "camera"));
749     auto eye = Vec3Member(camera, "eye");
750     auto target = Vec3Member(camera, "target");
751     auto up = Vec3Member(camera, "up", {0.0f, 1.0f, 0.0f});
752     float fov = FloatMember(camera, "fov_degrees", 60.0f);
753     if (!(fov > 0.0f && fov < 180.0f) || glm::length(target - eye) < 1e-6f ||
754         glm::length(glm::cross(target - eye, up)) < 1e-6f)
755       throw std::runtime_error("camera requires a valid field of view and nondegenerate look-at vectors");
756     const float aperture_radius = FloatMember(camera, "aperture_radius", 0.0f);
757     const std::string camera_type =
758         camera.HasMember("type") ? ReadString(camera["type"]) : (aperture_radius > 0.0f ? "thin_lens" : "pinhole");
759     const auto view = glm::lookAt(eye, target, up);
760     const float aspect = float(width) / float(height);
761     if (camera_type == "pinhole") {
762       result->camera_ = std::make_unique<CameraPinhole>(core, view, glm::radians(fov), aspect);
763     } else if (camera_type == "thin_lens") {
764       auto lens = std::make_unique<CameraThinLens>(core, view, glm::radians(fov), aspect);
765       lens->aperture_radius = aperture_radius;
766       lens->focus_distance = FloatMember(camera, "focus_distance", glm::length(target - eye));
767       lens->aperture_blades = camera.HasMember("aperture_blades") ? ReadInt(camera["aperture_blades"]) : 0;
768       lens->aperture_rotation = FloatMember(camera, "aperture_rotation", 0.0f);
769       lens->aperture_ratio = FloatMember(camera, "aperture_ratio", 1.0f);
770       result->camera_ = std::move(lens);
771     } else {
772       throw std::runtime_error("unknown camera type: " + camera_type);
773     }
774 
775     const auto &materials = RequireObject(Member(document, "materials"));
776     for (auto it = materials.MemberBegin(); it != materials.MemberEnd(); ++it) {
777       std::string id = ReadString(it->name);
778       const auto &spec = RequireObject(it->value);
779       std::string type = ReadString(Member(spec, "type"));
780       std::unique_ptr<Material> material;
781       if (type == "lambertian") {
782         material = std::make_unique<MaterialLambertian>(core, Vec3Member(spec, "base_color", {0.8f, 0.8f, 0.8f}),
783                                                         Vec3Member(spec, "emission"));
784       } else if (type == "specular") {
785         material = std::make_unique<MaterialSpecular>(core, Vec3Member(spec, "base_color", {0.8f, 0.8f, 0.8f}));
786       } else if (type == "light") {
787         material = std::make_unique<MaterialLight>(
788             core, Vec3Member(spec, "emission"), BoolMember(spec, "two_sided", false),
789             BoolMember(spec, "block_ray", false), BoolMember(spec, "camera_visible", true),
790             FloatMember(spec, "falloff_distance", 0.0f));
791       } else if (type == "principled") {
792         auto principled = std::make_unique<MaterialPrincipled>(core, Vec3Member(spec, "base_color", glm::vec3{0.8f}));
793         principled->subsurface_color = Vec3Member(spec, "subsurface_color", principled->subsurface_color);
794         principled->subsurface_radius = Vec3Member(spec, "subsurface_radius", principled->subsurface_radius);
795 #define READ_FLOAT(field) principled->field = FloatMember(spec, #field, principled->field)
796         READ_FLOAT(subsurface);
797         READ_FLOAT(metallic);
798         READ_FLOAT(specular);
799         READ_FLOAT(specular_tint);
800         READ_FLOAT(roughness);
801         READ_FLOAT(anisotropic);
802         READ_FLOAT(anisotropic_rotation);
803         READ_FLOAT(sheen);
804         READ_FLOAT(sheen_tint);
805         READ_FLOAT(clearcoat);
806         READ_FLOAT(clearcoat_roughness);
807         READ_FLOAT(ior);
808         READ_FLOAT(transmission);
809         READ_FLOAT(transmission_roughness);
810         READ_FLOAT(emission_strength);
811 #undef READ_FLOAT
812         principled->emission_color = Vec3Member(spec, "emission_color", principled->emission_color);
813         if (spec.HasMember("textures")) {
814           const auto &textures = RequireObject(spec["textures"]);
815           const std::pair<const char *, graphics::Image **> slots[] = {
816               {"normal", &principled->textures.normal},
817               {"base_color", &principled->textures.base_color},
818               {"metallic", &principled->textures.metallic},
819               {"specular", &principled->textures.specular},
820               {"roughness", &principled->textures.roughness},
821               {"anisotropic", &principled->textures.anisotropic},
822               {"anisotropic_rotation", &principled->textures.anisotropic_rotation},
823               {"emission", &principled->textures.emission}};
824           for (auto [slot, destination] : slots) {
825             if (!textures.HasMember(slot))
826               continue;
827             auto image = std::unique_ptr<graphics::Image>{};
828             auto asset = Resolve(path, ReadString(textures[slot]));
829             if (graphics::LoadImageFromFile(core->GraphicsCore(), asset.string(), &image) != 0)
830               throw std::runtime_error("cannot load texture: " + asset.string());
831             *destination = image.get();
832             result->images_.push_back(std::move(image));
833           }
834           principled->textures.normal_reverse_y = BoolMember(textures, "normal_reverse_y", false);
835         }
836         material = std::move(principled);
837       } else if (type == "shader_graph") {
838         const auto &graph = RequireObject(Member(spec, "graph"));
839         const auto &nodes = RequireObject(Member(graph, "nodes"));
840         std::map<std::string, int> texture_slots;
841         std::vector<graphics::Image *> textures;
842         for (auto node = nodes.MemberBegin(); node != nodes.MemberEnd(); ++node) {
843           if (!RequireObject(node->value).HasMember("type") ||
844               std::string(ReadString(node->value["type"])) != "image_texture")
845             continue;
846           auto asset = Resolve(path, ReadString(Member(node->value, "path")));
847           auto image = std::unique_ptr<graphics::Image>{};
848           if (graphics::LoadImageFromFile(core->GraphicsCore(), asset.string(), &image) != 0)
849             throw std::runtime_error("cannot load shader graph texture: " + asset.string());
850           texture_slots[ReadString(node->name)] = static_cast<int>(textures.size());
851           textures.push_back(image.get());
852           result->images_.push_back(std::move(image));
853         }
854         auto code = ShaderGraphCompiler(graph, texture_slots).Compile();
855         material = std::make_unique<MaterialShaderGraph>(core, code, textures, Vec3Member(spec, "emission_hint"));
856       } else {
857         throw std::runtime_error("unknown material type: " + type);
858       }
859       result->materials_.emplace(id, std::move(material));
860     }
861 
862     const auto &geometries = RequireObject(Member(document, "geometries"));
863     for (auto it = geometries.MemberBegin(); it != geometries.MemberEnd(); ++it) {
864       std::string id = ReadString(it->name);
865       const auto &spec = RequireObject(it->value);
866       std::string type = ReadString(Member(spec, "type"));
867       if (type == "hair") {
868         auto hair = BinaryHair(Resolve(path, ReadString(Member(spec, "path"))));
869         int radial_segments = spec.HasMember("radial_segments") ? ReadInt(spec["radial_segments"]) : 3;
870         result->geometries_.emplace(
871             id, std::make_unique<GeometryHair>(core, hair.points, hair.radii, hair.offsets, radial_segments));
872         continue;
873       }
874       Mesh<> mesh;
875       if (type == "sphere") {
876         int longitude = spec.HasMember("longitude_segments") ? ReadInt(spec["longitude_segments"]) : 30;
877         int latitude = spec.HasMember("latitude_segments") ? ReadInt(spec["latitude_segments"]) : -1;
878         if (longitude < 3 || (latitude != -1 && latitude < 2))
879           throw std::runtime_error("sphere requires at least 3 longitude and 2 latitude segments");
880         mesh = Mesh<>::Sphere(longitude, latitude);
881       } else if (type == "mesh") {
882         auto asset = Resolve(path, ReadString(Member(spec, "path")));
883         if (mesh.LoadObjFile(asset.string()) != 0)
884           throw std::runtime_error("cannot load mesh: " + asset.string());
885       } else if (type == "binary_mesh") {
886         mesh = BinaryMesh(Resolve(path, ReadString(Member(spec, "path"))));
887       } else if (type == "inline_mesh") {
888         mesh = InlineMesh(spec);
889       } else {
890         throw std::runtime_error("unknown geometry type: " + type);
891       }
892       if (BoolMember(spec, "generate_normals", false))
893         mesh.GenerateNormals();
894       if (BoolMember(spec, "generate_tangents", false))
895         mesh.GenerateTangents();
896       result->geometries_.emplace(id, std::make_unique<GeometryMesh>(core, mesh));
897     }
898 
899     const auto &entities = RequireArray(Member(document, "entities"));
900     for (const auto &spec : entities.GetArray()) {
901       std::string type = ReadString(Member(spec, "type"));
902       Entity *entity_ptr = nullptr;
903       if (type == "mesh") {
904         std::string geometry_id = ReadString(Member(spec, "geometry"));
905         std::string material_id = ReadString(Member(spec, "material"));
906         auto geometry = result->geometries_.find(geometry_id);
907         auto material = result->materials_.find(material_id);
908         if (geometry == result->geometries_.end())
909           throw std::runtime_error("unknown geometry: " + geometry_id);
910         if (material == result->materials_.end())
911           throw std::runtime_error("unknown material: " + material_id);
912         glm::mat4 transform{1.0f};
913         if (spec.HasMember("transform"))
914           transform = Transform(spec["transform"]);
915         if (spec.HasMember("look_at")) {
916           const auto &look = RequireObject(spec["look_at"]);
917           auto position = Vec3Member(look, "position");
918           auto target = Vec3Member(look, "target");
919           auto up = Vec3Member(look, "up", {0.0f, 1.0f, 0.0f});
920           auto scale = Vec3Member(look, "scale", glm::vec3{1.0f});
921           transform = glm::inverse(glm::lookAt(position, target, up)) * glm::scale(glm::mat4{1.0f}, scale);
922         }
923         auto entity =
924             std::make_unique<EntityGeometryMaterial>(core, geometry->second.get(), material->second.get(), transform);
925         entity->raster_light = BoolMember(spec, "raster_light", true);
926         entity_ptr = entity.get();
927         result->entities_.push_back(std::move(entity));
928       } else if (type == "point_light") {
929         auto entity = std::make_unique<EntityPointLight>(
930             core, Vec3Member(spec, "position"), Vec3Member(spec, "color", glm::vec3{1.0f}),
931             FloatMember(spec, "strength", 0.0f), FloatMember(spec, "radius", 0.0f),
932             BoolMember(spec, "soft_falloff", false), FloatMember(spec, "sampling_weight", -1.0f));
933         entity_ptr = entity.get();
934         result->entities_.push_back(std::move(entity));
935       } else {
936         throw std::runtime_error("unknown entity type: " + type);
937       }
938       result->scene_->AddEntity(entity_ptr);
939     }
940     return result;
941   } catch (const std::exception &exception) {
942     if (error)
943       *error = exception.what();
944     return nullptr;
945   }
946 }

948 std::vector<std::filesystem::path> FindJsonScenes(const std::filesystem::path &directory) {
949   std::vector<std::filesystem::path> result;
950   if (!std::filesystem::exists(directory))
951     return result;
952   for (const auto &entry : std::filesystem::recursive_directory_iterator(directory)) {
953     if (entry.is_regular_file() && entry.path().filename() == "scene.json")
954       result.push_back(entry.path());
955   }
956 
957   std::sort(result.begin(), result.end());
958   return result;
