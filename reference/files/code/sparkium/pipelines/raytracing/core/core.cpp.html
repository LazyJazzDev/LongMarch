职责与设计

公共 shader、几何光预处理程序缓存与 Render 转发。；本文件实现接口行为。

  • 几何光程序以 geometry/material 源码对为键，VFS/后端/入口/资源布局由 Core 固定。
  • film2img 与 Blelloch 上下扫构造时加载；内建原生命中 shader 仅在有 RT 支持时惰性加载。

29 graphics::ComputeProgram *Core::GetGeometryLightPowerProgram(const CodeLines &geometry, const CodeLines &material) {
30   // The VFS, backend, entry point, profile and binding layout are fixed for this Core.
31   auto key = std::make_pair(std::string(geometry), std::string(material));
32   auto found = geometry_light_programs_.find(key);
33   if (found != geometry_light_programs_.end())
34     return found->second.program.get();
35   GeometryLightResources resources;
36   auto vfs = GetShadersVFS();
37   vfs.WriteFile("geometry_sampler.slang", geometry);
38   vfs.WriteFile("material_evaluator.slang", material);
39   if (GraphicsCore()->CreateShader(vfs, "light/geometry_material/gather_primitive_power.slang",
40                                    "GatherPrimitivePowerKernel", "cs_6_3", {"-I."}, &resources.shader) != 0)
41     throw std::runtime_error("failed to compile geometry light power shader");
42   GraphicsCore()->CreateComputeProgram(resources.shader.get(), &resources.program);
43   resources.program->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
44   resources.program->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
45   resources.program->AddResourceBinding(graphics::RESOURCE_TYPE_WRITABLE_STORAGE_BUFFER, 1);
46   resources.program->Finalize();
47   auto *program = resources.program.get();
48   geometry_light_programs_.emplace(std::move(key), std::move(resources));
49   return program;
50 }

64   core_.GraphicsCore()->CreateShader(shaders_vfs, "film2img.slang", "Main", "cs_6_0", &shader);
65   core_.SetPublicResource("film2img", std::move(shader));
66 
67   core_.GraphicsCore()->CreateComputeProgram(core_.GetShader("film2img"), &compute_program);
68   compute_program->AddResourceBinding(graphics::RESOURCE_TYPE_IMAGE, 1);
69   compute_program->AddResourceBinding(graphics::RESOURCE_TYPE_IMAGE, 1);
70   compute_program->AddResourceBinding(graphics::RESOURCE_TYPE_WRITABLE_IMAGE, 1);
71   compute_program->Finalize();
72   core_.SetPublicResource("film2img", std::move(compute_program));
73 
74   core_.GraphicsCore()->CreateShader(shaders_vfs, "blelloch_scan.slang", "BlellochUpSweep", "cs_6_3", {"-I."}, &shader);
75   core_.SetPublicResource("blelloch_scan_up", std::move(shader));
76   core_.GraphicsCore()->CreateShader(shaders_vfs, "blelloch_scan.slang", "BlellochDownSweep", "cs_6_3", {"-I."},
