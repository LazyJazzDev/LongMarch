software_pipeline.cpp

code/sparkium/pipelines/raytracing/core/software_pipeline.cpp · 375 行 · 查看版本源码 ↗

职责与设计

GPU 软件 BVH 和原生 Ray Query 共用的计算路径追踪程序。；本文件实现接口行为。

  • 软件树叶数向上补到 2 的幂，每树最多 2^21 个实际叶；节点 32 字节，实例记录 112 字节，实例表头 16 字节，构建参数 256 字节。
  • 软件 BLAS 仅在树布局/几何指针/三角数变化时重建；TLAS 每帧重建。构建次序为叶初始化、归约边界、Morton key、Bitonic sort、叶重排、再归约。
  • Ray Query 使用硬件 BLAS/TLAS，不执行软件树构建；计算 renderer 仍使用相同材质分派与数据注册。
  • 材质源码去掉 include/pragma，按 shader_graph 标记与完整文本去重，分别放命名空间；图材质分离 GraphSurface 求值与共享 BSDF。编译键还包括相机文件/入口和各资源数量。
  • 矩阵行列式绝对值小于 1e-20 拒绝；实例/节点字节地址越过 uint32 范围拒绝。AddInstance 只保存，Update 才执行校验和构建。

17 uint32_t LeafCount(size_t count) {
18   // 64 threads per group, at most 65535 groups on baseline Vulkan/D3D12.
19   // Rounding to a power of two limits each tree to 2^21 actual leaves.
20   if (count > (uint32_t(1) << 21))
21     throw std::runtime_error("software BVH exceeds 2097152 leaves per tree");
22   uint32_t result = 1;
23   while (result < count)
24     result *= 2;
25   return result;
26 }

68   graphics::CpuProfileScope compile_profile("compile_builders");

70   builders_.clear();
71   if (builder_shaders_.empty()) {
72     for (const char *entry : {"InitLeaves", "ReduceNodes", "MortonKeys", "BitonicSort", "SortLeaves"}) {
73       std::unique_ptr<graphics::Shader> shader;

75         throw std::runtime_error(std::string("failed to compile software BVH kernel: ") + entry);

83     program->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, buffer_count);
84     program->AddResourceBinding(graphics::RESOURCE_TYPE_UNIFORM_BUFFER, 1);
85     program->AddResourceBinding(graphics::RESOURCE_TYPE_WRITABLE_STORAGE_BUFFER, 1);
86     program->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);

186   vfs.WriteFile("software_camera.slang", "#include \"" + camera_file + "\"\n");
187   render_program_.reset();
188   std::vector<std::string> args{"-I.", "-DSOFTWARE_DATA_BUFFER_COUNT=" + std::to_string(buffers)};
189   args.push_back("-DSPARKIUM_CAMERA_ENTRY=" + camera_entry);
190   if (ray_query_)
191     args.push_back("-DSPARKIUM_RAY_QUERY");
192   if (has_graph)
193     args.push_back("-DSPARKIUM_SHADER_GRAPHS");
194   if (core_->GraphicsCore()->CreateShader(vfs, "software/render.slang", "Main", ray_query_ ? "cs_6_5" : "cs_6_0", args,
195                                           &render_shader_))
196     throw std::runtime_error("failed to compile compute ray tracing shader");
197   core_->GraphicsCore()->CreateComputeProgram(render_shader_.get(), &render_program_);
198   for (auto binding : std::vector<std::pair<graphics::ResourceType, uint32_t>>{
199            {graphics::RESOURCE_TYPE_WRITABLE_IMAGE, 1},
200            {graphics::RESOURCE_TYPE_WRITABLE_IMAGE, 1},
201            {ray_query_ ? graphics::RESOURCE_TYPE_ACCELERATION_STRUCTURE : graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1},
202            {graphics::RESOURCE_TYPE_UNIFORM_BUFFER, 1},
203            {graphics::RESOURCE_TYPE_STORAGE_BUFFER, buffers + 6},
204            {graphics::RESOURCE_TYPE_IMAGE, sdr_count},
205            {graphics::RESOURCE_TYPE_IMAGE, hdr_count},
206            {graphics::RESOURCE_TYPE_SAMPLER, 2}})
207     render_program_->AddResourceBinding(binding.first, binding.second);
208   render_program_->Finalize();
209   camera_file_ = camera_file;
210   camera_entry_ = camera_entry;
211   material_sources_ = materials;
212   buffer_count_ = buffers;
213   sdr_count_ = sdr_count;
214   hdr_count_ = hdr_count;
215 }

217 void SoftwarePipeline::AppendBuild(std::vector<BuildPass> &passes, BuildParameters parameters) {
218   passes.push_back({0, parameters, parameters.leaves});
219   auto reduce = [&]() {
220     for (uint32_t count = parameters.leaves / 2; count; count /= 2) {
221       parameters.first = count - 1;
222       parameters.level_count = count;
223       passes.push_back({1, parameters, count});
224     }
225   };
226 
227   reduce();
228   if (parameters.leaves > 1) {
229     passes.push_back({2, parameters, parameters.leaves});
230     for (uint32_t stage = 2; stage <= parameters.leaves; stage *= 2) {
231       for (uint32_t stride = stage / 2; stride; stride /= 2) {
232         parameters.stage = stage;
233         parameters.stride = stride;
234         passes.push_back({3, parameters, parameters.leaves});
235       }
236     }
237     passes.push_back({4, parameters, parameters.leaves});
238     reduce();
239   }
240 }

242 void SoftwarePipeline::Update(graphics::CommandContext *commands,
243                               const std::vector<graphics::Buffer *> &buffers,
244                               uint32_t sdr_count,
245                               uint32_t hdr_count,
246                               Camera *camera) {
247   auto graphics = core_->GraphicsCore();
248   graphics::CpuProfileScope prepare_profile("software_prepare");
249   const std::string camera_file = camera->ShaderFile();
250   const std::string camera_entry = camera->EntryPoint();
251   std::vector<GeometryLayout> geometries;
252   std::vector<GPUInstance> gpu_instances;
253   std::vector<graphics::RayTracingInstance> native_instances;
254   std::vector<MaterialCode> materials;
255   if (16ull + instances_.size() * sizeof(GPUInstance) > std::numeric_limits<uint32_t>::max())
256     throw std::runtime_error("software instance byte address overflow");
257   const uint32_t tlas_leaves = ray_query_ ? 1 : LeafCount(instances_.size());
258   uint64_t node_count = uint64_t(tlas_leaves) * 2 - 1;
259   uint32_t max_leaves = tlas_leaves;
260   for (const auto &instance : instances_) {
261     auto geometry = std::find_if(geometries.begin(), geometries.end(),
262                                  [&](const GeometryLayout &g) { return g.geometry == instance.geometry; });
263     if (geometry == geometries.end()) {
264       const uint32_t count = instance.geometry->PrimitiveCount(), leaves = ray_query_ ? 1 : LeafCount(count);
265       if (node_count + uint64_t(leaves) * 2 - 1 > std::numeric_limits<uint32_t>::max() / 32u)
266         throw std::runtime_error("software BVH node address overflow");
267       geometries.push_back(
268           {instance.geometry, static_cast<uint32_t>(node_count), leaves, count, instance.geometry_index});
269       geometry = geometries.end() - 1;
270       node_count += uint64_t(leaves) * 2 - 1;
271       max_leaves = std::max(max_leaves, leaves);
272     }
273     const auto *graph = instance.material->GraphImpl();
274     MaterialCode source{graph != nullptr, MaterialSource(graph ? *graph : instance.material->SamplerImpl())};
275     auto material = std::find(materials.begin(), materials.end(), source);
276     uint32_t material_index = std::distance(materials.begin(), material);
277     if (material == materials.end())
278       materials.push_back(std::move(source));
279     glm::mat4 object_to_world(instance.transform);
280     if (std::abs(glm::determinant(object_to_world)) < 1.0e-20f)
281       throw std::runtime_error("software ray tracing requires invertible instance transforms");
282     gpu_instances.push_back({instance.transform, glm::mat4x3(glm::inverse(object_to_world)), geometry->root,
283                              instance.geometry_index, material_index, geometry->count});
284     if (ray_query_) {
285       auto blas = instance.geometry->BLAS();
286       if (!blas)
287         throw std::runtime_error("failed to build native triangle BLAS");
288       native_instances.push_back(
289           blas->MakeInstance(instance.transform, static_cast<uint32_t>(native_instances.size())));
290     }
291   }
292 
293   bool rebuild = !nodes_ || tlas_leaves != tlas_leaves_ || geometries.size() != geometries_.size();
294   for (size_t i = 0; !rebuild && i < geometries.size(); ++i)
295     rebuild = geometries[i].geometry != geometries_[i].geometry || geometries[i].count != geometries_[i].count;
296   if (rebuild && !ray_query_) {
297     graphics->CreateBuffer(node_count * 32, graphics::BUFFER_TYPE_STATIC, &nodes_);
298     graphics->CreateBuffer(uint64_t(max_leaves) * 8, graphics::BUFFER_TYPE_STATIC, &keys_);
299   }
300 
301   size_t instance_bytes = 16 + gpu_instances.size() * sizeof(GPUInstance);
302   if (!instances_buffer_ || instances_buffer_->Size() < instance_bytes)
303     graphics->CreateBuffer(instance_bytes, graphics::BUFFER_TYPE_STATIC, &instances_buffer_);
304   std::array<uint32_t, 4> header{static_cast<uint32_t>(gpu_instances.size()), 0, 0, 0};
305   instances_buffer_->UploadData(header.data(), sizeof(header));
306   if (!gpu_instances.empty())
307     instances_buffer_->UploadData(gpu_instances.data(), gpu_instances.size() * sizeof(GPUInstance), 16);
308   if (ray_query_) {
309     if (!native_tlas_) {
310       if (graphics->CreateTopLevelAccelerationStructure(native_instances, &native_tlas_))
311         throw std::runtime_error("failed to create native TLAS");
312     } else if (native_tlas_->UpdateInstances(native_instances)) {
313       throw std::runtime_error("failed to update native TLAS");
314     }
315     if (!render_program_ || camera_file_ != camera_file || camera_entry_ != camera_entry ||
316         materials != material_sources_ || buffer_count_ != buffers.size() || sdr_count_ != sdr_count ||
317         hdr_count_ != hdr_count)
318       CompileRenderer(materials, buffers.size(), sdr_count, hdr_count, camera_file, camera_entry);
319     if (graphics::FrameProfile::active) {
320       graphics::FrameProfile::active->counters["native_ray_query"] = 1;
321       graphics::FrameProfile::active->counters["instances"] = instances_.size();
322     }
323     return;
324   }
325   if (builders_.empty() || builder_buffer_count_ != buffers.size())
326     CompileBuilders(buffers.size());
327   if (!render_program_ || camera_file_ != camera_file || camera_entry_ != camera_entry ||
328       materials != material_sources_ || buffer_count_ != buffers.size() || sdr_count_ != sdr_count ||
329       hdr_count_ != hdr_count)
330     CompileRenderer(materials, buffers.size(), sdr_count, hdr_count, camera_file, camera_entry);
331   std::vector<BuildPass> passes;
332   if (rebuild)
333     for (const auto &geometry : geometries) {
334       BuildParameters parameters{};
335       parameters.root = geometry.root;
336       parameters.leaves = geometry.leaves;
337       parameters.count = geometry.count;
338       parameters.geometry = geometry.buffer_index;
339       AppendBuild(passes, parameters);
340     }
341   BuildParameters tlas{};
342   tlas.leaves = tlas_leaves;
343   tlas.count = gpu_instances.size();
344   tlas.instance_tree = 1;
345   AppendBuild(passes, tlas);
346   std::vector<BuildParameters> parameters;
347   for (const auto &pass : passes)
348     parameters.push_back(pass.parameters);
349   size_t parameter_bytes = parameters.size() * sizeof(BuildParameters);
350   if (!parameters_buffer_ || parameters_buffer_->Size() < parameter_bytes)
351     graphics->CreateBuffer(parameter_bytes, graphics::BUFFER_TYPE_STATIC, &parameters_buffer_);
352   parameters_buffer_->UploadData(parameters.data(), parameter_bytes);
353   prepare_profile.End();
354   if (graphics::FrameProfile::active) {
355     graphics::FrameProfile::active->counters["bvh_dispatches"] = passes.size();
356     graphics::FrameProfile::active->counters["blas_rebuilt"] = rebuild;
357     graphics::FrameProfile::active->counters["instances"] = instances_.size();
358     graphics::FrameProfile::active->counters["bvh_nodes"] = node_count;
359   }
360   graphics::CpuProfileScope record_profile("bvh_record");
361   graphics::GpuProfileScope build_profile(commands, "bvh_build");
362   for (size_t i = 0; i < passes.size(); ++i) {
363     commands->CmdBindComputeProgram(builders_[passes[i].kernel].get());
364     commands->CmdBindResources(0, {nodes_.get()}, graphics::BIND_POINT_COMPUTE);
365     commands->CmdBindResources(1, buffers, graphics::BIND_POINT_COMPUTE);
366     commands->CmdBindResources(2, {parameters_buffer_->Range(i * sizeof(BuildParameters), sizeof(BuildParameters))},
367                                graphics::BIND_POINT_COMPUTE);
368     commands->CmdBindResources(3, {keys_.get()}, graphics::BIND_POINT_COMPUTE);
369     commands->CmdBindResources(4, {instances_buffer_.get()}, graphics::BIND_POINT_COMPUTE);
370     commands->CmdDispatch((passes[i].count + 63) / 64, 1, 1);
371   }
372   geometries_ = std::move(geometries);
373   tlas_leaves_ = tlas_leaves;
374 }
