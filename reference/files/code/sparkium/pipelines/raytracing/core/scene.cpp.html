职责与设计

三种追踪模式共用的注册、光源预处理与 dispatch。；本文件实现接口行为。

  • ray_query 会强制 software=true；模式改变时若曾渲染，清累积。一般实体参数变化没有自动清累积。
  • 原生资源槽：0颜色、1样本、2TLAS、3设置、4Sobol、5Camera、6数据数组、7实例元数据、8光CDF、9光元数据、10SDR、11HDR、12两个sampler。计算槽：0/1相同、2节点或TLAS、3设置、4数据数组后加六个特殊Buffer、5SDR、6HDR、7sampler。
  • 资源表按指针去重，每帧清除 Buffer/Image 表；shader表按脏标记重建。容量不足重编译布局，未填满容量用相机或白图补齐。
  • 每帧更新实例与光元数据；前处理提交后再提交 trace/resolve，并 WaitGPU。前端样本计数在提交前增加，异常不会自动回滚该数字。

92     cmd_context->CmdBindResources(5, {camera->Buffer()}, bind_point);
93     cmd_context->CmdBindResources(6, buffers_, bind_point);
94     cmd_context->CmdBindResources(7, {instance_metadata_buffer_.get()}, bind_point);
95     cmd_context->CmdBindResources(8, {light_selector_buffer_.get()}, bind_point);
96     cmd_context->CmdBindResources(9, {light_metadatas_buffer_.get()}, bind_point);
97     cmd_context->CmdBindResources(10, sdr_images_, bind_point);
98     cmd_context->CmdBindResources(11, hdr_images_, bind_point);
99     cmd_context->CmdBindResources(12, std::vector{linear_sampler_.get(), nearest_sampler_.get()}, bind_point);
100   }
101   if (software)
102     cmd_context->CmdDispatch((film->GetWidth() + 7) / 8, (film->GetHeight() + 7) / 8, 1);
103   else
104     cmd_context->CmdDispatchRays(film->accumulated_color_->Extent().width, film->accumulated_samples_->Extent().height,
105                                  1);
106 
107   trace_profile.End();
108   graphics::GpuProfileScope resolve_profile(cmd_context.get(), "film_resolve");
109   cmd_context->CmdBindComputeProgram(core_->GetComputeProgram("film2img"));
110   cmd_context->CmdBindResources(0, {film->accumulated_color_.get()}, graphics::BIND_POINT_COMPUTE);
111   cmd_context->CmdBindResources(1, {film->accumulated_samples_.get()}, graphics::BIND_POINT_COMPUTE);
112   cmd_context->CmdBindResources(2, {film->film_.GetRawImage()}, graphics::BIND_POINT_COMPUTE);
113   cmd_context->CmdDispatch((film->film_.GetRawImage()->Extent().width + 7) / 8,
114                            (film->film_.GetRawImage()->Extent().height + 7) / 8, 1);
115 
116   resolve_profile.End();
117   setup_profile.End();
118   graphics::CpuProfileScope submit_profile("render_submit");
119   core_->GraphicsCore()->SubmitCommandContext(cmd_context.get());
120   submit_profile.End();
121   graphics::CpuProfileScope wait_profile("render_wait");
122   core_->GraphicsCore()->WaitGPU();
123 }

171 int32_t Scene::RegisterCallableShader(graphics::Shader *callable_shader) {
172   if (!callable_shader)
173     return -1;
174   if (!callable_shader_map_.count(callable_shader)) {
175     callable_shader_map_[callable_shader] = static_cast<int32_t>(callable_shaders_.size());
176     callable_shaders_.emplace_back(callable_shader);
177     pipeline_dirty_ = true;
178   }
179   return callable_shader_map_[callable_shader];
180 }

182 int32_t Scene::RegisterBuffer(graphics::Buffer *buffer) {
183   if (!buffer)
184     return -1;
185   if (!buffer_map_.count(buffer)) {
186     buffer_map_[buffer] = static_cast<int32_t>(buffers_.size());
187     buffers_.emplace_back(buffer);
188   }
189   return buffer_map_[buffer];
190 }

192 int32_t Scene::RegisterImage(graphics::Image *image) {
193   if (image->Format() == graphics::IMAGE_FORMAT_R8G8B8A8_UNORM) {
194     if (!sdr_image_map_.count(image)) {
195       sdr_image_map_[image] = static_cast<int32_t>(sdr_images_.size());
196       sdr_images_.emplace_back(image);
197     }
198     return sdr_image_map_[image];
199   }
200   if (image->Format() == graphics::IMAGE_FORMAT_R32G32B32A32_SFLOAT) {
201     if (!hdr_image_map_.count(image)) {
202       hdr_image_map_[image] = static_cast<int32_t>(hdr_images_.size());
203       hdr_images_.emplace_back(image);
204     }
205     return hdr_image_map_[image] + 0x1000000;
206   }
207   return -1;
208 }

210 int32_t Scene::RegisterHitGroup(const InstanceHitGroups &hit_group) {
211   if (!hit_group_map_.count(hit_group)) {
212     hit_group_map_[hit_group] = static_cast<int32_t>(hit_groups_.size());
213     hit_groups_.emplace_back(hit_group);
214     pipeline_dirty_ = true;
215   }
216   return hit_group_map_[hit_group];

219 void Scene::UpdatePipeline(Camera *camera) {
220   if (!software_tracing_ && (callable_shaders_.empty() || callable_shaders_[0] != camera->Shader()))
221     pipeline_dirty_ = true;
222   graphics::CpuProfileScope registration_profile("scene_registration");
223   for (auto &[entity, status] : entities_) {
224     status.keep = false;
225   }
226 
227   std::vector<Entity *> ordered_entities;
228   for (auto *entity : scene_.GetEntityOrder()) {
229     const auto &status = scene_.GetEntities().at(entity);
230     auto rt_entity = DedicatedCast(entity);
231     if (rt_entity) {
232       entities_[rt_entity].keep = true;
233       entities_[rt_entity].active = status.active;
234       ordered_entities.push_back(rt_entity);
235     }
236   }
237 
238   std::set<Entity *> to_delete_entities;
239 
240   for (auto &[entity, status] : entities_) {
241     if (!status.keep) {
242       pipeline_dirty_ = true;
243       to_delete_entities.insert(entity);
244     }
245   }
246 
247   for (auto entity : to_delete_entities) {
248     entities_.erase(entity);
249   }
250 
251   if (software_tracing_)
252     software_pipeline_->ClearInstances();
253   instances_.clear();
254   instance_metadatas_.clear();
255   light_metadatas_.clear();
256 
257   preprocess_cmd_context_.reset();
258   core_->GraphicsCore()->CreateCommandContext(&preprocess_cmd_context_);
259 
260   bool &pipeline_dirty = pipeline_dirty_;
261 
262   buffers_.clear();
263   buffer_map_.clear();
264 
265   sdr_images_.clear();
266   sdr_image_map_.clear();
267 
268   hdr_images_.clear();
269   hdr_image_map_.clear();
270   RegisterImage(core_->GetImage("white"));
271   RegisterImage(core_->GetImage("white_hdr"));
272 
273   if (pipeline_dirty) {
274     hit_groups_.clear();
275     hit_group_map_.clear();
276     callable_shaders_.clear();
277     callable_shader_map_.clear();
278     if (!software_tracing_)
279       RegisterCallableShader(camera->Shader());
280   }
281 
282   graphics::GpuProfileScope geometry_light_profile(preprocess_cmd_context_.get(), "light_geometry");
283   for (auto *entity : ordered_entities) {
284     if (entities_.at(entity).active) {
285       entity->Update(this);
286     }
287   }
288 
289   geometry_light_profile.End();
290   registration_profile.End();
291   graphics::CpuProfileScope metadata_profile("scene_metadata");
292   if (!software_tracing_) {
293     if (!tlas_) {
294       core_->GraphicsCore()->CreateTopLevelAccelerationStructure(instances_, &tlas_);
295     } else {
296       tlas_->UpdateInstances(instances_);
297     }
298   }
299 
300   if (!instance_metadata_buffer_ ||
301       sizeof(InstanceMetadata) * instance_metadatas_.size() > instance_metadata_buffer_->Size()) {
302     instance_metadata_buffer_.reset();
303     core_->GraphicsCore()->CreateBuffer(sizeof(InstanceMetadata) * std::max<size_t>(1, instance_metadatas_.size()),
304                                         graphics::BUFFER_TYPE_STATIC, &instance_metadata_buffer_);
305   }
306   if (!instance_metadatas_.empty())
307     instance_metadata_buffer_->UploadData(instance_metadatas_.data(),
308                                           sizeof(InstanceMetadata) * instance_metadatas_.size());
309 
310   if (!light_selector_buffer_ ||
311       sizeof(uint32_t) + sizeof(float) * light_metadatas_.size() > light_selector_buffer_->Size()) {
312     core_->GraphicsCore()->CreateBuffer(sizeof(uint32_t) + sizeof(float) * light_metadatas_.size(),
313                                         graphics::BUFFER_TYPE_STATIC, &light_selector_buffer_);
314   }
315   if (!light_metadatas_buffer_ || sizeof(LightMetadata) * light_metadatas_.size() > light_metadatas_buffer_->Size()) {
316     core_->GraphicsCore()->CreateBuffer(sizeof(LightMetadata) * std::max<size_t>(1, light_metadatas_.size()),
317                                         graphics::BUFFER_TYPE_STATIC, &light_metadatas_buffer_);
318   }
319   if (!light_metadatas_.empty())
320     light_metadatas_buffer_->UploadData(light_metadatas_.data(), sizeof(LightMetadata) * light_metadatas_.size());
321   uint32_t light_count = static_cast<uint32_t>(light_metadatas_.size());
322   light_selector_buffer_->UploadData(&light_count, sizeof(uint32_t), 0);
323 
324   if (buffers_.empty())
325     RegisterBuffer(camera->Buffer());
326 
327   if (pipeline_dirty_ || buffers_.size() > buffer_capacity_ || sdr_images_.size() > sdr_image_capacity_ ||
328       hdr_images_.size() > hdr_image_capacity_ || (!software_tracing_ && !rt_program_)) {
329     if (!software_tracing_) {
330       rt_program_.reset();
331       core_->GraphicsCore()->CreateRayTracingProgram(&rt_program_);
332       miss_shader_indices_ = {0, 1};
333       callable_shader_indices_.resize(callable_shaders_.size());
334       std::iota(callable_shader_indices_.begin(), callable_shader_indices_.end(), 0);
335       hit_group_indices_.resize(hit_groups_.size() * 2);
336       std::iota(hit_group_indices_.begin(), hit_group_indices_.end(), 0);
337       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_WRITABLE_IMAGE, 1);
338       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_WRITABLE_IMAGE, 1);
339       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_ACCELERATION_STRUCTURE, 1);
340       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_UNIFORM_BUFFER, 1);
341       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
342       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
343       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, buffers_.size());
344       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
345       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
346       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
347       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_IMAGE, sdr_images_.size());
348       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_IMAGE, hdr_images_.size());
349       rt_program_->AddResourceBinding(graphics::RESOURCE_TYPE_SAMPLER, 2);
350 
351       rt_program_->AddRayGenShader(raygen_shader_.get());
352       rt_program_->AddMissShader(default_miss_shader_.get());
353       rt_program_->AddMissShader(shadow_miss_shader_.get());
354       for (const auto &hit_group : hit_groups_) {
355         rt_program_->AddHitGroup(hit_group.render_group);
356         rt_program_->AddHitGroup(hit_group.shadow_group);
357       }
358       for (const auto &callable_shader : callable_shaders_) {
359         rt_program_->AddCallableShader(callable_shader);
360       }
361       rt_program_->Finalize(miss_shader_indices_, hit_group_indices_, callable_shader_indices_);
362     }
363 
364     if (buffers_.size() > buffer_capacity_ || !gather_light_power_program_) {
365       gather_light_power_program_.reset();
366       core_->GraphicsCore()->CreateComputeProgram(gather_light_power_shader_.get(), &gather_light_power_program_);
367       gather_light_power_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, 1);
368       gather_light_power_program_->AddResourceBinding(graphics::RESOURCE_TYPE_WRITABLE_STORAGE_BUFFER, 1);
369       gather_light_power_program_->AddResourceBinding(graphics::RESOURCE_TYPE_STORAGE_BUFFER, buffers_.size());
370       gather_light_power_program_->Finalize();
371     }
372 
373     buffer_capacity_ = buffers_.size();
374     sdr_image_capacity_ = sdr_images_.size();
375     hdr_image_capacity_ = hdr_images_.size();
376     pipeline_dirty_ = false;
377   }
378 
379   while (buffers_.size() < buffer_capacity_) {
380     buffers_.push_back(camera->Buffer());
381   }
382 
383   while (sdr_images_.size() < sdr_image_capacity_) {
384     sdr_images_.push_back(core_->GetImage("white"));
385   }
386 
387   while (hdr_images_.size() < hdr_image_capacity_) {
388     hdr_images_.push_back(core_->GetImage("white_hdr"));
389   }
390 
391   metadata_profile.End();
392   if (software_tracing_)
393     software_pipeline_->Update(preprocess_cmd_context_.get(), buffers_, sdr_images_.size(), hdr_images_.size(), camera);
394 
395   graphics::CpuProfileScope light_record_profile("light_selection_record");
396   graphics::GpuProfileScope light_selection_profile(preprocess_cmd_context_.get(), "light_selection");
397   preprocess_cmd_context_->CmdBindComputeProgram(gather_light_power_program_.get());
398   preprocess_cmd_context_->CmdBindResources(0, {light_metadatas_buffer_.get()}, graphics::BIND_POINT_COMPUTE);
399   preprocess_cmd_context_->CmdBindResources(1, {light_selector_buffer_.get()}, graphics::BIND_POINT_COMPUTE);
400   preprocess_cmd_context_->CmdBindResources(2, buffers_, graphics::BIND_POINT_COMPUTE);
401   if (!light_metadatas_.empty())
402     preprocess_cmd_context_->CmdDispatch((light_metadatas_.size() + 63) / 64, 1, 1);
403 
404   blelloch_metadatas_.clear();
405   BlellochScanMetadata metadata{4, 4, static_cast<uint32_t>(light_metadatas_.size())};
406   uint32_t group_size = 64;
407   while (metadata.element_count > 1) {
408     blelloch_metadatas_.push_back(metadata);
409     metadata = {metadata.offset + (group_size - 1) * metadata.stride, metadata.stride * group_size,
410                 metadata.element_count / group_size};
411   }
412   if (!blelloch_metadatas_.empty()) {
413     auto blelloch_scan_up_program = core_->GetComputeProgram("blelloch_scan_up");
414     auto blelloch_scan_down_program = core_->GetComputeProgram("blelloch_scan_down");
415     if (!blelloch_metadata_buffer_ ||
416         sizeof(BlellochScanMetadata) * blelloch_metadatas_.size() > blelloch_metadata_buffer_->Size()) {
417       core_->GraphicsCore()->CreateBuffer(sizeof(BlellochScanMetadata) * blelloch_metadatas_.size(),
418                                           graphics::BUFFER_TYPE_STATIC, &blelloch_metadata_buffer_);
419     }
420     blelloch_metadata_buffer_->UploadData(blelloch_metadatas_.data(),
421                                           sizeof(BlellochScanMetadata) * blelloch_metadatas_.size());
422     preprocess_cmd_context_->CmdBindComputeProgram(blelloch_scan_up_program);
423     preprocess_cmd_context_->CmdBindResources(0, {light_selector_buffer_.get()}, graphics::BIND_POINT_COMPUTE);
424     for (size_t i = 1; i < blelloch_metadatas_.size(); i++) {
425       preprocess_cmd_context_->CmdBindResources(1, {blelloch_metadata_buffer_->Range(sizeof(BlellochScanMetadata) * i)},
426                                                 graphics::BIND_POINT_COMPUTE);
427       preprocess_cmd_context_->CmdDispatch((blelloch_metadatas_[i].element_count + 63) / 64, 1, 1);
428     }
429     preprocess_cmd_context_->CmdBindComputeProgram(blelloch_scan_down_program);
430     for (size_t i = blelloch_metadatas_.size() - 1; i < blelloch_metadatas_.size(); i--) {
431       if (blelloch_metadatas_[i].element_count <= group_size) {
432         continue;
433       }
434       preprocess_cmd_context_->CmdBindResources(1, {blelloch_metadata_buffer_->Range(sizeof(BlellochScanMetadata) * i)},
435                                                 graphics::BIND_POINT_COMPUTE);
436       preprocess_cmd_context_->CmdDispatch((blelloch_metadatas_[i].element_count + 63) / 64, 1, 1);
437     }
438   }
439   light_selection_profile.End();
440   light_record_profile.End();
441   graphics::CpuProfileScope preprocess_submit_profile("preprocess_submit");
442   core_->GraphicsCore()->SubmitCommandContext(preprocess_cmd_context_.get());
443 }
