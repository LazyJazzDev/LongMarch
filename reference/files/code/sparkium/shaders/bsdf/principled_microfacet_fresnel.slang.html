36 Spectrum bsdf_microfacet_ggx_eval_reflect_fresnel(const FresnelBsdf bsdf,
37                                                   const float3 N,
38                                                   const float3 I,
39                                                   const float3 omega_in,
40                                                   inout float pdf,
41                                                   const float alpha_x,
42                                                   const float alpha_y,
43                                                   const float cosNO,
44                                                   const float cosNI) {
45   if (!(cosNI > 0 && cosNO > 0)) {
46     pdf = 0.0f;
47     return make_float3(0);
48   }
49 
50   /* get half vector */
51   float3 m = normalize(omega_in + I);
52   float alpha2 = alpha_x * alpha_y;
53   float D, G1o, G1i;
54 
55   if (alpha_x == alpha_y) {
56     /* isotropic
57      * eq. 20: (F*G*D)/(4*in*on)
58      * eq. 33: first we calculate D(m) */
59     float cosThetaM = dot(N, m);
60     float cosThetaM2 = cosThetaM * cosThetaM;
61     float cosThetaM4 = cosThetaM2 * cosThetaM2;
62     float tanThetaM2 = (1 - cosThetaM2) / cosThetaM2;
63 
64     D = alpha2 / (PI * cosThetaM4 * (alpha2 + tanThetaM2) * (alpha2 + tanThetaM2));
65 
66     /* eq. 34: now calculate G1(i,m) and G1(o,m) */
67     G1o = 2 / (1 + safe_sqrtf(1 + alpha2 * (1 - cosNO * cosNO) / (cosNO * cosNO)));
68     G1i = 2 / (1 + safe_sqrtf(1 + alpha2 * (1 - cosNI * cosNI) / (cosNI * cosNI)));
69   } else {
70     /* anisotropic */
71     float3 X, Y, Z = N;
72     make_orthonormals_tangent(Z, bsdf.T, X, Y);
73 
74     /* distribution */
75     float3 local_m = make_float3(dot(X, m), dot(Y, m), dot(Z, m));
76     float slope_x = -local_m.x / (local_m.z * alpha_x);
77     float slope_y = -local_m.y / (local_m.z * alpha_y);
78     float slope_len = 1 + slope_x * slope_x + slope_y * slope_y;
79 
80     float cosThetaM = local_m.z;
81     float cosThetaM2 = cosThetaM * cosThetaM;
82     float cosThetaM4 = cosThetaM2 * cosThetaM2;
83 
84     D = 1 / ((slope_len * slope_len) * PI * alpha2 * cosThetaM4);
85 
86     /* G1(i,m) and G1(o,m) */
87     float tanThetaO2 = (1 - cosNO * cosNO) / (cosNO * cosNO);
88     float cosPhiO = dot(I, X);
89     float sinPhiO = dot(I, Y);
90 
91     float alphaO2 = (cosPhiO * cosPhiO) * (alpha_x * alpha_x) + (sinPhiO * sinPhiO) * (alpha_y * alpha_y);
92     alphaO2 /= cosPhiO * cosPhiO + sinPhiO * sinPhiO;
93 
94     G1o = 2 / (1 + safe_sqrtf(1 + alphaO2 * tanThetaO2));
95 
96     float tanThetaI2 = (1 - cosNI * cosNI) / (cosNI * cosNI);
97     float cosPhiI = dot(omega_in, X);
98     float sinPhiI = dot(omega_in, Y);
99 
100     float alphaI2 = (cosPhiI * cosPhiI) * (alpha_x * alpha_x) + (sinPhiI * sinPhiI) * (alpha_y * alpha_y);
101     alphaI2 /= cosPhiI * cosPhiI + sinPhiI * sinPhiI;
102 
103     G1i = 2 / (1 + safe_sqrtf(1 + alphaI2 * tanThetaI2));
104   }
105 
106   float G = G1o * G1i;
107 
108   /* eq. 20 */
109   float common_ = D * 0.25f / cosNO;
110 
111   Spectrum F = reflection_color(bsdf, omega_in, m);
112 
113   Spectrum out_ = F * G * common_;
114 
115   /* eq. 2 in distribution of visible normals sampling
116    * `pm = Dw = G1o * dot(m, I) * D / dot(N, I);` */
117 
118   /* eq. 38 - but see also:
119    * eq. 17 in http://www.graphics.cornell.edu/~bjw/wardnotes.pdf
120    * `pdf = pm * 0.25 / dot(m, I);` */
121   pdf = G1o * common_;
122 
123   return out_;
124 }

144 int bsdf_microfacet_ggx_sample_fresnel(const FresnelBsdf bsdf,
145                                        float3 Ng,
146                                        float3 I,
147                                        float randu,
148                                        float randv,
149                                        inout Spectrum eval,
150                                        inout float3 omega_in,
151                                        inout float pdf) {
152   float alpha_x = bsdf.alpha_x;
153   float alpha_y = bsdf.alpha_y;
154 
155   float3 N = bsdf.N;
156   int label;
157 
158   float cosNO = dot(N, I);
159   if (cosNO > 0) {
160     float3 X, Y, Z = N;
161 
162     if (alpha_x == alpha_y)
163       MakeOrthonormals(Z, X, Y);
164     else
165       make_orthonormals_tangent(Z, bsdf.T, X, Y);
166 
167     /* importance sampling with distribution of visible normals. vectors are
168      * transformed to local space before and after */
169     float3 local_I = make_float3(dot(X, I), dot(Y, I), cosNO);
170     float3 local_m;
171     float G1o;
172 
173     local_m = microfacet_sample_stretched(local_I, alpha_x, alpha_y, randu, randv, false, G1o);
174 
175     float3 m = X * local_m.x + Y * local_m.y + Z * local_m.z;
176     float cosThetaM = local_m.z;
177 
178     /* reflection or refraction? */
179     float cosMO = dot(m, I);
180     label = LABEL_REFLECT | LABEL_GLOSSY;
181 
182     if (cosMO > 0) {
183       /* eq. 39 - compute actual reflected direction */
184       omega_in = 2 * cosMO * m - I;
185 
186       if (dot(Ng, omega_in) > 0) {
187         if (alpha_x * alpha_y <= 1e-7f) {
188           /* some high number for MIS */
189           pdf = 1e6f;
190           eval = make_float3(1e6f) * reflection_color(bsdf, omega_in, m);
191 
192           label = LABEL_REFLECT | LABEL_SINGULAR;
193         } else {
194           /* microfacet normal is visible to this ray */
195           /* eq. 33 */
196           float alpha2 = alpha_x * alpha_y;
197           float D, G1i;
198 
199           if (alpha_x == alpha_y) {
200             /* isotropic */
201             float cosThetaM2 = cosThetaM * cosThetaM;
202             float cosThetaM4 = cosThetaM2 * cosThetaM2;
203             float tanThetaM2 = 1 / (cosThetaM2)-1;
204 
205             /* eval BRDF*cosNI */
206             float cosNI = dot(N, omega_in);
207 
208             D = alpha2 / (PI * cosThetaM4 * (alpha2 + tanThetaM2) * (alpha2 + tanThetaM2));
209 
210             /* eq. 34: now calculate G1(i,m) */
211             G1i = 2 / (1 + safe_sqrtf(1 + alpha2 * (1 - cosNI * cosNI) / (cosNI * cosNI)));
212           } else {
213             /* anisotropic distribution */
214             float3 local_m = make_float3(dot(X, m), dot(Y, m), dot(Z, m));
215             float slope_x = -local_m.x / (local_m.z * alpha_x);
216             float slope_y = -local_m.y / (local_m.z * alpha_y);
217             float slope_len = 1 + slope_x * slope_x + slope_y * slope_y;
218 
219             float cosThetaM = local_m.z;
220             float cosThetaM2 = cosThetaM * cosThetaM;
221             float cosThetaM4 = cosThetaM2 * cosThetaM2;
222 
223             D = 1 / ((slope_len * slope_len) * PI * alpha2 * cosThetaM4);
224 
225             /* calculate G1(i,m) */
226             float cosNI = dot(N, omega_in);
227 
228             float tanThetaI2 = (1 - cosNI * cosNI) / (cosNI * cosNI);
229             float cosPhiI = dot(omega_in, X);
230             float sinPhiI = dot(omega_in, Y);
231 
232             float alphaI2 = (cosPhiI * cosPhiI) * (alpha_x * alpha_x) + (sinPhiI * sinPhiI) * (alpha_y * alpha_y);
233             alphaI2 /= cosPhiI * cosPhiI + sinPhiI * sinPhiI;
234 
235             G1i = 2 / (1 + safe_sqrtf(1 + alphaI2 * tanThetaI2));
236           }
237 
238           /* see eval function for derivation */
239           float common_ = (G1o * D) * 0.25f / cosNO;
240           pdf = common_;
241 
242           Spectrum F = reflection_color(bsdf, omega_in, m);
243 
244           eval = G1i * common_ * F;
245         }
246       } else {
247         eval = make_float3(0);
248         pdf = 0.0f;
249       }
250     }
251   } else {
252     label = LABEL_REFLECT | LABEL_GLOSSY;
253   }
254   return label;
255 }
