3 void microfacet_beckmann_sample_slopes(const float cos_theta_i,
4                                        const float sin_theta_i,
5                                        float randu,
6                                        float randv,
7                                        inout float slope_x,
8                                        inout float slope_y,
9                                        inout float G1i) {
10   /* special case (normal incidence) */
11   if (cos_theta_i >= 0.99999f) {
12     const float r = sqrt(-log(randu));
13     const float phi = 2.0 * PI * randv;
14     slope_x = r * cos(phi);
15     slope_y = r * sin(phi);
16     G1i = 1.0f;
17     return;
18   }
19 
20   /* precomputations */
21   const float tan_theta_i = sin_theta_i / cos_theta_i;
22   const float inv_a = tan_theta_i;
23   const float cot_theta_i = 1.0f / tan_theta_i;
24   const float erf_a = fast_erff(cot_theta_i);
25   const float exp_a2 = exp(-cot_theta_i * cot_theta_i);
26   const float SQRT_PI_INV = 0.56418958354f;
27   const float Lambda = 0.5f * (erf_a - 1.0f) + (0.5f * SQRT_PI_INV) * (exp_a2 * inv_a);
28   const float G1 = 1.0f / (1.0f + Lambda); /* masking */
29 
30   G1i = G1;
31   /* Based on paper from Wenzel Jakob
32    * An Improved Visible Normal Sampling Routine for the Beckmann Distribution
33    *
34    * http://www.mitsuba-renderer.org/~wenzel/files/visnormal.pdf
35    *
36    * Reformulation from OpenShadingLanguage which avoids using inverse
37    * trigonometric functions.
38    */
39 
40   /* Sample slope X.
41    *
42    * Compute a coarse approximation using the approximation:
43    *   exp(-ierf(x)^2) ~= 1 - x * x
44    *   solve y = 1 + b + K * (1 - b * b)
45    */
46   float K = tan_theta_i * SQRT_PI_INV;
47   float y_approx = randu * (1.0f + erf_a + K * (1 - erf_a * erf_a));
48   float y_exact = randu * (1.0f + erf_a + K * exp_a2);
49   float b = K > 0 ? (0.5f - sqrt(K * (K - y_approx + 1.0f) + 0.25f)) / K : y_approx - 1.0f;
50 
51   /* Perform newton step to refine toward the true root. */
52   float inv_erf = fast_ierff(b);
53   float value = 1.0f + b + K * exp(-inv_erf * inv_erf) - y_exact;
54   /* Check if we are close enough already,
55    * this also avoids NaNs as we get close to the root.
56    */
57   if (abs(value) > 1e-6f) {
58     b -= value / (1.0f - inv_erf * tan_theta_i); /* newton step 1. */
59     inv_erf = fast_ierff(b);
60     value = 1.0f + b + K * exp(-inv_erf * inv_erf) - y_exact;
61     b -= value / (1.0f - inv_erf * tan_theta_i); /* newton step 2. */
62     /* Compute the slope from the refined value. */
63     slope_x = fast_ierff(b);
64   } else {
65     /* We are close enough already. */
66     slope_x = inv_erf;
67   }
68   slope_y = fast_ierff(2.0f * randv - 1.0f);
69 }

71 void microfacet_ggx_sample_slopes(const float cos_theta_i,
72                                   const float sin_theta_i,
73                                   float randu,
74                                   float randv,
75                                   inout float slope_x,
76                                   inout float slope_y,
77                                   inout float G1i) {
78   /* special case (normal incidence) */
79   if (cos_theta_i >= 0.99999f) {
80     const float r = sqrt(randu / (1.0f - randu));
81     const float phi = 2.0 * PI * randv;
82     slope_x = r * cos(phi);
83     slope_y = r * sin(phi);
84     G1i = 1.0f;
85 
86     return;
87   }
88 
89   /* precomputations */
90   const float tan_theta_i = sin_theta_i / cos_theta_i;
91   const float G1_inv = 0.5f * (1.0f + safe_sqrtf(1.0f + tan_theta_i * tan_theta_i));
92 
93   G1i = 1.0f / G1_inv;
94 
95   /* sample slope_x */
96   const float A = 2.0f * randu * G1_inv - 1.0f;
97   const float AA = A * A;
98   const float tmp = 1.0f / (AA - 1.0f);
99   const float B = tan_theta_i;
100   const float BB = B * B;
101   const float D = safe_sqrtf(BB * (tmp * tmp) - (AA - BB) * tmp);
102   const float slope_x_1 = B * tmp - D;
103   const float slope_x_2 = B * tmp + D;
104   slope_x = (A < 0.0f || slope_x_2 * tan_theta_i > 1.0f) ? slope_x_1 : slope_x_2;
105 
106   /* sample slope_y */
107   float S;
108 
109   if (randv > 0.5f) {
110     S = 1.0f;
111     randv = 2.0f * (randv - 0.5f);
112   } else {
113     S = -1.0f;
114     randv = 2.0f * (0.5f - randv);
115   }
116 
117   const float z = (randv * (randv * (randv * 0.27385f - 0.73369f) + 0.46341f)) /
118                   (randv * (randv * (randv * 0.093073f + 0.309420f) - 1.000000f) + 0.597999f);
119   slope_y = S * z * safe_sqrtf(1.0f + (slope_x) * (slope_x));
120 }

122 float3 microfacet_sample_stretched(const float3 omega_i,
123                                    const float alpha_x,
124                                    const float alpha_y,
125                                    const float randu,
126                                    const float randv,
127                                    bool beckmann,
128                                    inout float G1i) {
129   /* 1. stretch omega_i */
130   float3 omega_i_ = make_float3(alpha_x * omega_i.x, alpha_y * omega_i.y, omega_i.z);
131   omega_i_ = normalize(omega_i_);
132 
133   /* get polar coordinates of omega_i_ */
134   float costheta_ = 1.0f;
135   float sintheta_ = 0.0f;
136   float cosphi_ = 1.0f;
137   float sinphi_ = 0.0f;
138 
139   if (omega_i_.z < 0.99999f) {
140     costheta_ = omega_i_.z;
141     sintheta_ = safe_sqrtf(1.0f - costheta_ * costheta_);
142 
143     float invlen = 1.0f / sintheta_;
144     cosphi_ = omega_i_.x * invlen;
145     sinphi_ = omega_i_.y * invlen;
146   }
147 
148   /* 2. sample P22_{omega_i}(x_slope, y_slope, 1, 1) */
149   float slope_x, slope_y;
150 
151   if (beckmann) {
152     microfacet_beckmann_sample_slopes(costheta_, sintheta_, randu, randv, slope_x, slope_y, G1i);
153   } else {
154     microfacet_ggx_sample_slopes(costheta_, sintheta_, randu, randv, slope_x, slope_y, G1i);
155   }
156 
157   /* 3. rotate */
158   float tmp = cosphi_ * slope_x - sinphi_ * slope_y;
159   slope_y = sinphi_ * slope_x + cosphi_ * slope_y;
160   slope_x = tmp;
161 
162   /* 4. unstretch */
163   slope_x = alpha_x * slope_x;
164   slope_y = alpha_y * slope_y;
165 
166   /* 5. compute normal */
167   return normalize(make_float3(-slope_x, -slope_y, 1.0f));
168 }
