44 float fresnel_dielectric(float eta,
45                          const float3 N,
46                          const float3 I,
47                          inout float3 R,
48                          inout float3 T,
49                          inout bool is_inside) {
50   float cos = dot(N, I), neta;
51   float3 Nn;
52 
53   // check which side of the surface we are on
54   if (cos > 0) {
55     // we are on the outside of the surface, going in
56     neta = 1 / eta;
57     Nn = N;
58     is_inside = false;
59   } else {
60     // we are inside the surface
61     cos = -cos;
62     neta = eta;
63     Nn = -N;
64     is_inside = true;
65   }
66 
67   // compute reflection
68   R = (2 * cos) * Nn - I;
69 
70   float arg = 1 - (neta * neta * (1 - (cos * cos)));
71   if (arg < 0) {
72     T = make_float3(0.0f, 0.0f, 0.0f);
73     return 1;  // total internal reflection
74   } else {
75     float dnp = max(sqrt(arg), 1e-7f);
76     float nK = (neta * cos) - dnp;
77     T = -(neta * I) + (nK * Nn);
78     // compute Fresnel terms
79     float cosTheta1 = cos;  // N.R
80     float cosTheta2 = -dot(Nn, T);
81     float pPara = (cosTheta1 - eta * cosTheta2) / (cosTheta1 + eta * cosTheta2);
82     float pPerp = (eta * cosTheta1 - cosTheta2) / (eta * cosTheta1 + cosTheta2);
83     return 0.5f * (pPara * pPara + pPerp * pPerp);
84   }
85 }

87 float fresnel_dielectric_cos(float cosi, float eta) {
88   float c = abs(cosi);
89   float g = eta * eta - 1 + c * c;
90   if (g > 0) {
91     g = sqrt(g);
92     float A = (g - c) / (g + c);
93     float B = (c * (g + c) - 1) / (c * (g - c) + 1);
94     return 0.5f * A * A * (1 + B * B);
95   }
96   return 1.0f;
97 }
