file(GLOB_RECURSE DEMO_SOURCES CONFIGURE_DEPENDS "*.cpp" "*.h")
add_executable(${DEMO_NAME} ${DEMO_SOURCES})
target_include_directories(${DEMO_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/game_of_life_lib)
find_package(tinyfiledialogs CONFIG REQUIRED)
target_link_libraries(${DEMO_NAME} PRIVATE LongMarch tinyfiledialogs::tinyfiledialogs)
if(WIN32)
    target_link_libraries(${DEMO_NAME} PRIVATE comdlg32 ole32 shell32)
endif()
PACK_SHADER_CODE(${DEMO_NAME})
LONGMARCH_DEPLOY_RUNTIME_DEPENDENCIES(${DEMO_NAME})
