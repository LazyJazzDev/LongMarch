function(embed_shaders target directory output)
  file(GLOB_RECURSE shaders CONFIGURE_DEPENDS "${directory}/*.slang")
  get_filename_component(parent "${output}" DIRECTORY)
  file(MAKE_DIRECTORY "${parent}")
  add_custom_command(OUTPUT "${output}"
    COMMAND "${Python3_EXECUTABLE}" "${MOBILE}/embed_shaders.py" "${directory}" "${output}"
    DEPENDS ${shaders} "${MOBILE}/embed_shaders.py" VERBATIM)
  target_sources(${target} PRIVATE "${output}")
endfunction()

  embed_shaders(harmony_engine "${ROOT}/code/${pair}" "${CMAKE_CURRENT_BINARY_DIR}/${pair}/built_in_shaders.inl")

  list(FILTER game_sources EXCLUDE REGEX "/(main|ai_player|ai_benchmark|game_rules)\\.cpp$|/[^/]*_lib/")

  target_include_directories(harmony_${game} PRIVATE "${ROOT}/demo/${game}" "${ROOT}/demo/${game}/${game}_lib" "${ROOT}/demo/gol/game_of_life_lib")
  target_compile_definitions(harmony_${game} PRIVATE LONGMARCH_HOSTED_GAMES GLFW_INCLUDE_NONE)
  embed_shaders(harmony_${game} "${ROOT}/demo/${game}" "${CMAKE_CURRENT_BINARY_DIR}/game_${game}/built_in_shaders.inl")
  set_source_files_properties("${ROOT}/demo/${game}/application/application.cpp" PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}/game_${game}")
