find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)

file(GLOB GRAPHICS "${ROOT}/code/grassland/graphics/*.cpp")

list(FILTER BACKEND EXCLUDE REGEX "/vulkan_(window|imgui_assets)\\.cpp$")

  "${LONGMARCH_HEADERS}" "${LONGMARCH_HEADERS}/eigen3" "${LONGMARCH_HEADERS}/stb"
  "${mikktspace_SOURCE_DIR}" "${ROOT}/demo/2048/2048_lib")

  FMT_HEADER_ONLY NDEBUG LONGMARCH_ASSETS_DIR="")

  find_package(Vulkan REQUIRED)
  target_link_libraries(harmony_engine PUBLIC Vulkan::Vulkan)

if(LONGMARCH_PREPARE)
  find_path(DXC_INCLUDE_DIR dxc/dxcapi.h HINTS /usr/local/include REQUIRED)
  find_library(DXC_LIBRARY dxcompiler HINTS /usr/local/lib REQUIRED)
  target_include_directories(harmony_engine PRIVATE "${DXC_INCLUDE_DIR}")
  target_link_libraries(harmony_engine PUBLIC "${DXC_LIBRARY}")

  target_compile_definitions(harmony_engine PRIVATE LONGMARCH_OFFLINE_SHADERS)

function(embed_shaders target directory output)
  file(GLOB_RECURSE shaders CONFIGURE_DEPENDS "${directory}/*.hlsl" "${directory}/*.hlsli")
  get_filename_component(parent "${output}" DIRECTORY)
  file(MAKE_DIRECTORY "${parent}")
  add_custom_command(OUTPUT "${output}"
    COMMAND "${Python3_EXECUTABLE}" "${MOBILE}/embed_shaders.py" "${directory}" "${output}"
    DEPENDS ${shaders} "${MOBILE}/embed_shaders.py" VERBATIM)
  target_sources(${target} PRIVATE "${output}")
endfunction()
foreach(pair "grassland/graphics" "sparkium/shaders")
  embed_shaders(harmony_engine "${ROOT}/code/${pair}" "${CMAKE_CURRENT_BINARY_DIR}/${pair}/built_in_shaders.inl")
endforeach()

  file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS "${ROOT}/demo/${game}/*.cpp")
  list(FILTER game_sources EXCLUDE REGEX "/(main|ai_player|ai_benchmark|game_rules)\\.cpp$|/[^/]*_lib/")
  add_library(harmony_${game} STATIC ${game_sources})
  target_link_libraries(harmony_${game} PUBLIC harmony_engine freetype)
  target_include_directories(harmony_${game} PRIVATE "${ROOT}/demo/${game}" "${ROOT}/demo/${game}/${game}_lib" "${ROOT}/demo/gol/game_of_life_lib")
  target_compile_definitions(harmony_${game} PRIVATE LONGMARCH_HOSTED_GAMES GLFW_INCLUDE_NONE)
  embed_shaders(harmony_${game} "${ROOT}/demo/${game}" "${CMAKE_CURRENT_BINARY_DIR}/game_${game}/built_in_shaders.inl")
  set_source_files_properties("${ROOT}/demo/${game}/application/application.cpp" PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}/game_${game}")
endforeach()
